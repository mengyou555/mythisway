# practice_llm.py : Fake Client สำหรับโหมด MOCK ของแบบฝึกหัด Part 3
# ใช้ร่วมกันทุกโจทย์ ไม่ต้องมี API Key ก็รันได้
import os
from dataclasses import dataclass

MODEL = os.environ.get("CLAUDE_MODEL", "claude-opus-5-5")


@dataclass
class TextBlock:
    text: str
    type: str = "text"


@dataclass
class ToolUseBlock:
    id: str
    name: str
    input: dict
    type: str = "tool_use"


@dataclass
class Usage:
    input_tokens: int
    output_tokens: int


@dataclass
class FakeMessage:
    content: list
    stop_reason: str
    usage: Usage
    role: str = "assistant"
    model: str = "mock"


class _FakeMessages:
    def __init__(self, script):
        self.script = list(script)  # คำตอบที่เตรียมไว้ เรียงตามลำดับ
        self.calls = []             # เก็บ kwargs ทุกครั้งที่ถูกเรียก ไว้ตรวจใน test

    def create(self, **kwargs):
        # copy รายการ messages ไว้ เพราะโปรแกรมหลักจะต่อท้าย list เดิมไปเรื่อยๆ
        self.calls.append({**kwargs, "messages": list(kwargs.get("messages", []))})
        if self.script:
            return self.script.pop(0)
        # สคริปต์หมดแล้ว: ตอบ end_turn อย่างปลอดภัย
        return FakeMessage([TextBlock("[MOCK] สคริปต์หมดแล้ว")], "end_turn", Usage(100, 10))


class FakeClient:
    def __init__(self, script):
        self.messages = _FakeMessages(script)

    @property
    def calls(self):
        return self.messages.calls


def make_client(script):
    """มี ANTHROPIC_API_KEY ใช้ Claude จริง ถ้าไม่มีใช้ FakeClient (ไม่พิมพ์ Key ออกมาเด็ดขาด)"""
    if os.environ.get("ANTHROPIC_API_KEY"):
        import anthropic
        return anthropic.Anthropic()
    return FakeClient(script)


def mode_of(client) -> str:
    return "MOCK" if isinstance(client, FakeClient) else "REAL"


# ---------- ตัวช่วยสร้างคำตอบจำลองให้สั้นลง ----------
def tool_turn(*calls, usage=(812, 96)):
    """calls คือ tuple (id, name, input) หนึ่งตัวหรือหลายตัว (หลายตัว = Parallel Tool Calls)"""
    blocks = [ToolUseBlock(id=i, name=n, input=inp) for i, n, inp in calls]
    return FakeMessage(blocks, "tool_use", Usage(*usage))


def final_turn(text, usage=(1530, 188)):
    return FakeMessage([TextBlock(text)], "end_turn", Usage(*usage))

BRANCHES = ["เชียงใหม่", "ขอนแก่น", "ภูเก็ต"]

    "strict": True,
    "input_schema": {
        "type": "object",
        "properties": {
            "branch": {"type": "string", "enum": BRANCHES, "description": "ชื่อสาขา"},

    },

def get_branch_sales(branch: str, quarter: str) -> str:
    # ตรวจซ้ำในโค้ด แม้ schema จะมี enum แล้ว (ข้อมูลอาจมาจากที่อื่นที่ไม่ใช่ strict)
    if quarter not in QUARTERS:
        raise ValueError(f"ไม่มีข้อมูลไตรมาส {quarter} มีเฉพาะ {', '.join(QUARTERS)}")
    path = DOCS_DIR / f"sales_{quarter.lower()}_2026.md"
    text = path.read_text(encoding="utf-8")
    m = re.search(rf"^- สาขา{re.escape(branch)}: ([\d,]+) บาท$", text, re.MULTILINE)
    if m is None:
        raise LookupError(f"ไม่พบสาขา {branch} ในไฟล์ {path.name} สาขาที่มีคือ {', '.join(BRANCHES)}")
    result = {"branch": branch, "quarter": quarter,
              "sales_baht": int(m.group(1).replace(",", "")), "source": path.name}
    return json.dumps(result, ensure_ascii=False)


TOOL_FUNCTIONS = {"get_branch_sales": get_branch_sales}


def run_tool(name: str, tool_input: dict) -> tuple[str, bool]:
    """รัน Tool ตามชื่อ คืน (content, is_error) เสมอ ไม่ปล่อยให้ exception หลุดออกไป"""
    func = TOOL_FUNCTIONS.get(name)
    if func is None:
        return f"Error: ไม่รู้จัก tool ชื่อ {name} ใช้ได้เฉพาะ {', '.join(TOOL_FUNCTIONS)}", True
    try:
        return func(**tool_input), False
    except (ValueError, LookupError, FileNotFoundError, TypeError) as e:
        return f"Error: {e}", True

def read_doc(name: str) -> str:
    target = (DOCS_DIR / name).resolve()
    # Path Sandbox: ต้องอยู่ใน docs/ และเป็นไฟล์ .md เท่านั้น
    if not target.is_relative_to(DOCS_DIR) or target.suffix != ".md":
        raise PermissionError(f"อ่านได้เฉพาะไฟล์ .md ในโฟลเดอร์ docs/ เท่านั้น ({name} ไม่อนุญาต)")
    if not target.exists():
        files = ", ".join(sorted(p.name for p in DOCS_DIR.glob("*.md")))
        raise FileNotFoundError(f"ไม่พบไฟล์ {name} ไฟล์ที่มีคือ: {files}")
    return target.read_text(encoding="utf-8")


def calculator(a: float, b: float, op: str) -> str:
    if op == "div" and b == 0:
        raise ZeroDivisionError("หารด้วยศูนย์ไม่ได้ ตรวจค่า b อีกครั้ง")
    ops = {"add": a + b, "sub": a - b, "mul": a * b}
    result = a / b if op == "div" else ops[op]
    return json.dumps({"a": a, "b": b, "op": op, "result": result})

    def log(self, step, event, *, stop_reason=None, input_tokens=0, output_tokens=0,
            tool=None, tool_input=None, is_error=False, ms=0):
        row = {"run_id": self.run_id, "step": step, "event": event, "stop_reason": stop_reason,
               "input_tokens": input_tokens, "output_tokens": output_tokens, "tool": tool,
               "input": tool_input, "is_error": is_error, "ms": round(ms)}
        with self.path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(row, ensure_ascii=False) + "\n")


def execute(block) -> dict:
    """รัน tool_use หนึ่งบล็อก แล้วคืน dict ของ tool_result พร้อมเวลาที่ใช้"""
    start = time.perf_counter()
    content, is_error = run_tool(block.name, block.input)
    result = {"type": "tool_result", "tool_use_id": block.id, "content": content}
    if is_error:
        result["is_error"] = True
    return {"result": result, "ms": (time.perf_counter() - start) * 1000}


def run_agent(client, question: str, run_id: str, max_steps: int = 8) -> dict:
    tracer = Tracer(run_id)
    messages = [{"role": "user", "content": question}]
    totals = {"input_tokens": 0, "output_tokens": 0, "tool_calls": 0}

    def finish(answer, steps):
        tracer.log(steps, "final", input_tokens=totals["input_tokens"],
                   output_tokens=totals["output_tokens"])
        return {"answer": answer, "steps": steps, **totals, "trace": str(tracer.path.name)}

    for step in range(1, max_steps + 1):
        start = time.perf_counter()
        response = client.messages.create(model=MODEL, max_tokens=16000, system=SYSTEM,
                                          tools=TOOLS, messages=messages)
        usage = response.usage
        totals["input_tokens"] += usage.input_tokens
        totals["output_tokens"] += usage.output_tokens
        tracer.log(step, "llm_call", stop_reason=response.stop_reason,
                   input_tokens=usage.input_tokens, output_tokens=usage.output_tokens,
                   ms=(time.perf_counter() - start) * 1000)
        print(f"[step {step}] stop_reason={response.stop_reason}")

        if response.stop_reason == "end_turn":
            return finish("".join(b.text for b in response.content if b.type == "text"), step)
        if response.stop_reason == "max_tokens":
            return finish("คำตอบถูกตัดเพราะเกิน max_tokens ลองถามให้แคบลง", step)
        if response.stop_reason == "refusal":
            return finish("Claude ปฏิเสธคำขอนี้ จึงหยุดการทำงาน", step)
        if response.stop_reason != "tool_use":
            return finish(f"หยุดเพราะ stop_reason ที่ไม่รองรับ: {response.stop_reason}", step)

        messages.append({"role": "assistant", "content": response.content})
        tool_blocks = [b for b in response.content if b.type == "tool_use"]
        # รันทุก tool พร้อมกัน pool.map คืนผลตามลำดับเดิมของ tool_blocks
        with ThreadPoolExecutor(max_workers=4) as pool:
            outputs = list(pool.map(execute, tool_blocks))
        for block, out in zip(tool_blocks, outputs):
            res = out["result"]
            flag = res.get("is_error", False)
            tracer.log(step, "tool_call", tool=block.name, tool_input=block.input,
                       is_error=flag, ms=out["ms"])
            print(f"   -> {block.name}({json.dumps(block.input, ensure_ascii=False)})"
                  f" is_error={flag} | {res['content'][:60].replace(chr(10), ' ')}")
        totals["tool_calls"] += len(tool_blocks)
        results = [out["result"] for out in outputs]
        # ส่งผลทุก tool กลับใน user message เดียว
        messages.append({"role": "user", "content": results})
        print(f"   user message: {len(results)} tool_result,"
              f" is_error={[r.get('is_error', False) for r in results]}")

    return finish(f"หยุดเพราะครบจำนวนรอบสูงสุด ({max_steps} รอบ)", max_steps)


def cost_usd(model: str, input_tokens: int, output_tokens: int) -> float:
    price_in, price_out = PRICES[model]
    return input_tokens * price_in / 1_000_000 + output_tokens * price_out / 1_000_000

แนวคิด
  • Loop มีทางออกครบทุกทาง: for step in range(1, max_steps + 1) รับประกันว่าจบเสมอ และทุก stop_reason (end_turn, max_tokens, refusal, ค่าที่ไม่รู้จัก) มีทางออกพร้อมข้อความที่บอกผู้ใช้ว่าเกิดอะไรขึ้น ทุกทางออกผ่าน finish() จึงมี event final ใน Trace เสมอ
  • Parallel แต่ผลเรียงลำดับ: pool.map(execute, tool_blocks) รัน Tool พร้อมกัน (มีประโยชน์มากเมื่อ Tool ต้องรอเครือข่ายหรือดิสก์) แต่คืนผลตามลำดับเดิม ทำให้ log อ่านง่าย และ tool_result ทั้งหมดถูกรวมเป็น user message เดียว
  • Error เป็นข้อมูล ไม่ใช่ความล้มเหลว: run_tool แปลงทุก exception เป็นข้อความที่บอกวิธีแก้ และ execute ใส่ is_error เฉพาะเมื่อ error Claude จึงแก้ตัวเองได้ภายในรอบเดียว
  • Trace แบบ append-only: เขียน JSONL ทีละบรรทัดทันทีที่เกิด event ถ้าโปรแกรมพังกลางทาง ข้อมูลก่อนหน้ายังอยู่ครบ และใช้ schema เดียวกับ Part 1 ทำให้ใช้คำสั่ง jq ชุดเดิมได้
  • ค่าใช้จ่ายคำนวณจาก usage จริง: รวม input_tokens กับ output_tokens จากทุกการเรียก แล้วคูณราคาต่อ 1M token แยกกัน Run 1 ใช้ 7,038/722 tokens ประมาณ $0.042592 ตัวเลขนี้คือสิ่งที่ใช้ตัดสินว่า Agent คุ้มกว่า Workflow หรือไม่
  • ทดสอบเพดานด้วยสคริปต์ที่ไม่รู้จบ: การจงใจสร้างสถานการณ์ผิดปกติ (LOOP_SCRIPT) แล้วดูว่า Guardrail ทำงาน สำคัญกว่าการทดสอบแค่กรณีที่ทุกอย่างราบรื่น

def save_note(filename: str, content: str, notes_dir: Path) -> str:
    # ตรวจชื่อไฟล์ใน Python เพราะ strict schema ไม่ได้ตรวจ pattern ให้
    if not FILENAME_RE.fullmatch(filename):
        raise ValueError(f"ชื่อไฟล์ {filename!r} ไม่ถูกต้อง ใช้ได้แค่ a-z 0-9 _ - และลงท้าย .md")
    notes_dir.mkdir(parents=True, exist_ok=True)
    target = notes_dir / filename
    if target.exists():
        raise FileExistsError(f"มีไฟล์ {filename} อยู่แล้ว ไม่เขียนทับ ให้ตั้งชื่อใหม่")
    target.write_text(content, encoding="utf-8")
    return f"บันทึกแล้ว: notes/{filename} ({len(content)} ตัวอักษร)"


def wrap(source: str, text: str) -> str:
    """ห่อผลของ tool เพื่อบอก Claude ว่านี่คือข้อมูล ไม่ใช่คำสั่ง"""
    return f'<document source="{source}">\n{text}\n</document>'

def run_agent(client, question, approver=deny_all, notes_dir=NOTES_DIR,
              max_steps=MAX_STEPS, max_total_tokens=MAX_TOTAL_TOKENS):
    messages = [{"role": "user", "content": question}]
    events, security = [], []
    used = 0

    def run_one(block, step):
        name, tool_input = block.name, block.input
        if name not in ALLOWED_TOOLS:  # ชั้นที่ 1: Allowlist
            return f"Error: tool {name} ไม่อยู่ในรายการที่อนุญาต", True
        if name in WRITE_TOOLS:
            if not FILENAME_RE.fullmatch(tool_input.get("filename", "")):  # ตรวจก่อนรบกวนผู้ใช้
                return f"Error: ชื่อไฟล์ {tool_input.get('filename')!r} ไม่ถูกต้อง", True
            approved = approver(name, tool_input)  # ชั้นที่ 2: Human-in-the-loop
            events.append({"step": step, "event": "approval", "tool": name,
                           "filename": tool_input["filename"], "approved": approved})
            if not approved:
                return "Error: ผู้ใช้ไม่อนุมัติการบันทึกไฟล์นี้ ห้ามลองใหม่ด้วยชื่ออื่น", True
            try:
                return save_note(tool_input["filename"], tool_input["content"], notes_dir), False
            except (ValueError, FileExistsError) as e:
                return f"Error: {e}", True
        content, is_error = research_tools.run_tool(name, tool_input)  # ชั้นที่ 3: Path Sandbox อยู่ใน read_doc
        if is_error:
            return content, True
        if any(s in content.lower() for s in SUSPICIOUS):
            security.append({"step": step, "source": tool_input.get("name", name),
                             "reason": "พบข้อความที่พยายามสั่ง Agent ในผลของ tool"})
        return wrap(tool_input.get("name", name), content), False  # ชั้นที่ 4: ห่อเป็นข้อมูล

    def done(answer, steps, stopped_by):
        return {"answer": answer, "steps": steps, "stopped_by": stopped_by, "tokens": used,
                "events": events, "security_events": security}

    for step in range(1, max_steps + 1):
        response = client.messages.create(model=MODEL, max_tokens=16000, system=SYSTEM,
                                          tools=TOOLS, messages=messages)
        used += response.usage.input_tokens + response.usage.output_tokens
        events.append({"step": step, "event": "llm_call", "stop_reason": response.stop_reason})
        if response.stop_reason == "end_turn":
            return done("".join(b.text for b in response.content if b.type == "text"), step, "end_turn")
        if response.stop_reason != "tool_use":
            return done(f"หยุดเพราะ stop_reason={response.stop_reason}", step, response.stop_reason)
        if used > max_total_tokens:  # ชั้นที่ 5: Token Budget
            return done(f"หยุดเพราะใช้ token เกินงบ ({used} > {max_total_tokens})", step, "token_budget")
        messages.append({"role": "assistant", "content": response.content})
        results = []
        for block in response.content:
            if block.type != "tool_use":
                continue
            content, is_error = run_one(block, step)
            events.append({"step": step, "event": "tool_call", "tool": block.name,
                           "input": block.input, "is_error": is_error})
            result = {"type": "tool_result", "tool_use_id": block.id, "content": content}
            if is_error:
                result["is_error"] = True
            results.append(result)
        messages.append({"role": "user", "content": results})
    return done(f"หยุดเพราะครบจำนวนรอบสูงสุด ({max_steps} รอบ)", max_steps, "max_steps")
