  throw new Error('พังแบบ async');

    2. Request ค้างไปเรื่อย ๆ จน Client Timeout เพราะไม่มีใครส่ง Response

สถานการณ์

ร้านซ่อมมือถือของสมชายเขียนใบรับซ่อมบนกระดาษ ลูกค้าโทรมาถามสถานะบ่อยจนช่างไม่มีเวลาซ่อม เจ้าของร้านอยากได้ API เล็ก ๆ ที่หน้าร้านใช้เพิ่มใบรับซ่อม ช่างใช้เปลี่ยนสถานะงาน และหน้าเว็บใช้ดึงสถานะไปแสดงให้ลูกค้าดู รุ่นแรกนี้ขอให้เก็บข้อมูลใน Memory ก่อน แต่ต้องมีโครงสร้างที่พร้อมต่อยอด ทุก Error ต้องตอบเป็น JSON รูปแบบเดียวกัน และทุก Request ต้องมีรหัสติดตาม (Request ID) เพื่อให้ไล่ปัญหาจาก Log ได้

สิ่งที่ต้องทำ

  3. แยกไฟล์เป็น src/app.js (export createApp() ที่ยังไม่ listen) และ src/server.js (อ่าน PORT แปลงเป็นตัวเลข ตรวจความถูกต้อง แล้ว listen)

// report.js — สรุปยอดขายร้านกาแฟจากไฟล์ JSON รายวัน
import { readFile, writeFile, mkdir } from 'node:fs/promises';
import path from 'node:path';
import os from 'node:os';

// 1) รับวันที่จาก argv ถ้าไม่ส่งมาให้ใช้ค่าเริ่มต้น
const date = process.argv[2] ?? '2026-10-20';

// 2) สร้าง path จากโฟลเดอร์ของสคริปต์ ไม่ใช่โฟลเดอร์ที่เราเปิด Terminal อยู่
const inputFile = path.join(import.meta.dirname, 'data', `sales-${date}.json`);
const outputDir = path.join(import.meta.dirname, 'reports');
const outputFile = path.join(outputDir, `report-${date}.txt`);

// 3) อ่านไฟล์และแปลง JSON พร้อมจัดการ Error ที่เจอบ่อย
let sales;
try {
  const text = await readFile(inputFile, 'utf8');
  sales = JSON.parse(text);
} catch (err) {
  if (err.code === 'ENOENT') {
    console.error(`ไม่พบไฟล์ยอดขายของวันที่ ${date}: ${path.basename(inputFile)}`);
  } else if (err instanceof SyntaxError) {
    console.error(`ไฟล์ ${path.basename(inputFile)} ไม่ใช่ JSON ที่ถูกต้อง: ${err.message}`);
  } else {
    console.error('อ่านไฟล์ไม่สำเร็จ:', err.message);
  }
  process.exitCode = 1; // ให้ Shell รู้ว่าทำงานไม่สำเร็จ
}

if (sales) {
  // 4) รวมจำนวนแก้วและยอดเงินแยกตามเมนู
  const byMenu = {};
  for (const item of sales) {
    byMenu[item.menu] ??= { qty: 0, amount: 0 };
    byMenu[item.menu].qty += item.qty;
    byMenu[item.menu].amount += item.qty * item.price;
  }

  // 5) เรียงเมนูจากยอดเงินมากไปน้อย
  const rows = Object.entries(byMenu).sort((a, b) => b[1].amount - a[1].amount);
  const total = rows.reduce((sum, [, v]) => sum + v.amount, 0);
  const [bestMenu] = rows[0];

  // 6) สร้างข้อความรายงาน ใช้ os.EOL เป็นตัวขึ้นบรรทัด
  const lines = [
    `รายงานยอดขายวันที่ ${date}`,
    ...rows.map(([menu, v]) => `- ${menu}: ${v.qty} ชิ้น ${v.amount} บาท`),
    `รวมทั้งหมด: ${total} บาท`,
    `เมนูขายดีที่สุด: ${bestMenu}`,
  ];
  const report = lines.join(os.EOL) + os.EOL;

  // 7) สร้างโฟลเดอร์ reports (ถ้ามีอยู่แล้วก็ไม่ Error) แล้วเขียนไฟล์
  await mkdir(outputDir, { recursive: true });
  await writeFile(outputFile, report, 'utf8');

  console.log(report.trimEnd());
  console.log(`บันทึกรายงานที่ ${path.relative(process.cwd(), outputFile)}`);
}

แนวคิด
  • Path อิงตำแหน่งสคริปต์: path.join(import.meta.dirname, 'data', ...) ทำให้ผลไม่ขึ้นกับ process.cwd() การทดสอบข้อ 2 ที่รันจาก /tmp แล้วยังได้ Exit Code 0 พิสูจน์เรื่องนี้ ถ้าเขียน 'data/sales.json' ตรง ๆ จะเจอ ENOENT ทันทีที่รันจากโฟลเดอร์อื่น
  • แยก Error ตามชนิด: err.code === 'ENOENT' คือไม่พบไฟล์ err instanceof SyntaxError คือ JSON เสีย และกรณีอื่น เช่น EACCES (ไม่มีสิทธิ์อ่าน) ใช้ข้อความกลาง ผู้ใช้จึงรู้ว่าต้องแก้อะไร แทนที่จะเห็น Stack Trace ยาว ๆ
  • Exit Code: ใช้ process.exitCode = 1 ให้ Shell และระบบอัตโนมัติอย่าง cron หรือ CI รู้ว่าล้มเหลว โดยไม่ตัดการทำงานกลางคันแบบ process.exit(1)
  • รวมยอดด้วย Object: ??= สร้างช่องเก็บของเมนูใหม่เฉพาะครั้งแรก แล้ว Object.entries + sort เรียงตามยอดเงิน วิธีนี้วนข้อมูลรอบเดียว (O(n)) ใช้ได้แม้ไฟล์มีหลายพันรายการ
  • เขียนไฟล์อย่างปลอดภัย: mkdir(..., { recursive: true }) ไม่ Error ถ้าโฟลเดอร์มีอยู่แล้ว และ os.EOL ให้ตัวขึ้นบรรทัดที่ถูกต้องตามระบบปฏิบัติการ

// middleware/request-id.js — แปะรหัสประจำ Request เพื่อใช้ไล่ Log
import { randomUUID } from 'node:crypto';

export function requestId(req, res, next) {
  req.id = randomUUID();
  res.setHeader('X-Request-Id', req.id);
  next(); // ส่งต่อให้ Middleware ตัวถัดไป
}
// middleware/logger.js — พิมพ์ Method, URL, Status Code และเวลาที่ใช้ของทุก Request
export function logger(req, res, next) {
  const start = performance.now();
  // Event 'finish' เกิดหลังส่ง Response ครบแล้ว จึงรู้ Status Code จริง
  res.on('finish', () => {
    const ms = (performance.now() - start).toFixed(1);
    console.log(`${req.method} ${req.originalUrl} ${res.statusCode} ${ms}ms id=${req.id.slice(0, 8)}`);
  });
  next();
}
// middleware/errors.js — 404 และ Error-handling Middleware ที่ตอบ JSON รูปแบบเดียวกันทั้งระบบ
export class HttpError extends Error {
  constructor(status, error, message, details) {
    super(message);
    this.status = status;
    this.error = error;
    this.details = details;
  }
}

export function notFound(req, res) {
  res.status(404).json({ error: 'NotFound', message: `ไม่พบ ${req.method} ${req.originalUrl}` });
}

// ต้องมี 4 พารามิเตอร์ Express จึงจะรู้ว่าเป็น Error Handler
export function errorHandler(err, req, res, next) {
  // Body ที่เป็น JSON เสีย express.json() จะส่ง Error ที่มี status 400 มาให้
  if (err.type === 'entity.parse.failed') {
    return res.status(400).json({ error: 'ValidationError', message: 'Body ไม่ใช่ JSON ที่ถูกต้อง' });
  }
  if (err instanceof HttpError) {
    const body = { error: err.error, message: err.message };
    if (err.details) body.details = err.details;
    return res.status(err.status).json(body);
  }
  console.error(`[${req.id}]`, err); // Error ที่ไม่คาดคิด เก็บรายละเอียดไว้ใน Log ฝั่ง Server
  res.status(500).json({ error: 'InternalServerError', message: 'เกิดข้อผิดพลาดภายในระบบ' });
}

// routes/tickets.js — Router ของใบรับซ่อม เก็บข้อมูลไว้ในหน่วยความจำ
import express from 'express';
import { setTimeout as sleep } from 'node:timers/promises';
import { HttpError } from '../middleware/errors.js';

const STATUSES = ['waiting', 'repairing', 'done'];

// ข้อมูลตัวอย่าง (หายเมื่อปิด Server เพราะอยู่ใน Memory)
const tickets = [
  { id: 1, customer: 'สมศรี', device: 'iPhone 13', problem: 'จอแตก', status: 'repairing' },
  { id: 2, customer: 'วิชัย', device: 'Galaxy A54', problem: 'แบตเสื่อม', status: 'waiting' },
];
let nextId = 3;

// ตรวจ Body ของใบรับซ่อมใหม่ คืนค่า Object ของ Error แยกตามช่อง
function validateTicket(body) {
  const errors = {};
  for (const field of ['customer', 'device', 'problem']) {
    const value = body?.[field];
    if (typeof value !== 'string' || value.trim() === '') {
      errors[field] = 'ต้องเป็นข้อความและห้ามว่าง';
    } else if (value.length > 100) {
      errors[field] = 'ยาวได้ไม่เกิน 100 ตัวอักษร';
    }
  }
  return errors;
}

function findTicket(idParam) {
  const id = Number(idParam);
  const ticket = tickets.find((t) => t.id === id);
  if (!ticket) throw new HttpError(404, 'NotFound', `ไม่พบใบรับซ่อม id ${idParam}`);
  return ticket;
}

export const ticketsRouter = express.Router();

// GET /api/tickets?status=waiting
ticketsRouter.get('/', (req, res) => {
  const { status } = req.query;
  if (status !== undefined && !STATUSES.includes(status)) {
    throw new HttpError(400, 'ValidationError', `status ต้องเป็น ${STATUSES.join(', ')}`);
  }
  const items = status ? tickets.filter((t) => t.status === status) : tickets;
  res.json({ count: items.length, items });
});

// GET /api/tickets/stats — จำลองงานแบบ async เช่นการถามฐานข้อมูล
ticketsRouter.get('/stats', async (req, res) => {
  await sleep(50);
  const stats = Object.fromEntries(STATUSES.map((s) => [s, tickets.filter((t) => t.status === s).length]));
  res.json({ shop: req.app.locals.shopName, total: tickets.length, ...stats });
});

// GET /api/tickets/:id
ticketsRouter.get('/:id', (req, res) => {
  res.json(findTicket(req.params.id));
});

// POST /api/tickets
ticketsRouter.post('/', (req, res) => {
  const errors = validateTicket(req.body);
  if (Object.keys(errors).length > 0) {
    throw new HttpError(400, 'ValidationError', 'ข้อมูลใบรับซ่อมไม่ถูกต้อง', errors);
  }
  const ticket = {
    id: nextId++,
    customer: req.body.customer.trim(),
    device: req.body.device.trim(),
    problem: req.body.problem.trim(),
    status: 'waiting',
  };
  tickets.push(ticket);
  res.status(201).location(`/api/tickets/${ticket.id}`).json(ticket);
});

// PATCH /api/tickets/:id/status
ticketsRouter.patch('/:id/status', async (req, res) => {
  const ticket = findTicket(req.params.id);
  const { status } = req.body ?? {};
  if (!STATUSES.includes(status)) {
    throw new HttpError(400, 'ValidationError', `status ต้องเป็น ${STATUSES.join(', ')}`);
  }
  ticket.status = status;
  res.json(ticket);
});

// app.js — ประกอบ Express App แต่ยังไม่ listen เพื่อให้นำไปเทสต์ได้
import express from 'express';
import { requestId } from './middleware/request-id.js';
import { logger } from './middleware/logger.js';
import { notFound, errorHandler } from './middleware/errors.js';
import { ticketsRouter } from './routes/tickets.js';

export function createApp({ shopName = 'ร้านซ่อมมือถือ' } = {}) {
  const app = express();
  app.disable('x-powered-by');
  app.locals.shopName = shopName;

  // ลำดับสำคัญ: requestId ต้องมาก่อน logger เพราะ logger ใช้ req.id
  app.use(requestId);
  app.use(logger);
  app.use(express.json({ limit: '10kb' }));

  app.get('/health', (req, res) => res.json({ status: 'ok', shop: shopName }));
  app.use('/api/tickets', ticketsRouter);

  // Route ทดสอบ: throw ใน async handler แล้ว Express 5 ส่งต่อให้ errorHandler เอง
  app.get('/api/boom', async () => {
    throw new Error('ฐานข้อมูลจำลองล่ม');
  });

  app.use(notFound); // ต้องอยู่หลังทุก Route
  app.use(errorHandler); // ต้องอยู่ท้ายสุด
  return app;
}

แนวคิด
  • แยก app กับ server: createApp() คืน Express App ที่ยังไม่ listen ส่วน server.js ดูแลเรื่อง Port และ Environment ทำให้นำ App ไปเทสต์ด้วย supertest ได้ทันทีโดยไม่ต้องเปิด Port จริง และส่ง Config อย่าง shopName เข้าไปได้โดยไม่ผูกกับ process.env
  • ลำดับ Middleware คือ Pipeline: requestId ต้องมาก่อน logger เพราะ Logger ใช้ req.id ส่วน notFound อยู่หลังทุก Route เพื่อจับเฉพาะ Request ที่ไม่มี Route ไหนรับ และ errorHandler อยู่ท้ายสุดเพื่อรับ Error จากทุกจุด
  • Logger รอ Event finish: ตอน Middleware ทำงาน Handler ยังไม่ตัดสินว่าจะตอบ Status อะไร จึงต้องฟัง res.on('finish') ซึ่งเกิดหลังส่ง Response ครบ ใช้ performance.now() วัดเวลาละเอียดระดับเศษของมิลลิวินาที
  • throw แทนการตอบ Error เอง: Handler ทุกตัว throw new HttpError(...) แล้วให้ errorHandler แปลงเป็น JSON ที่จุดเดียว รูปแบบ { error, message, details? } จึงเหมือนกันทั้งระบบ Express 5 ส่งต่อ Error ให้ทั้งจาก Handler ปกติและ async handler เช่น PATCH และ /api/boom
  • ไม่เปิดเผยรายละเอียดภายใน: Error ที่ไม่ใช่ HttpError ตอบ Client แค่ "เกิดข้อผิดพลาดภายในระบบ" แต่ Log Stack Trace ฝั่ง Server พร้อม Request ID ทีมงานจึงเอา Header X-Request-Id ที่ลูกค้าแจ้งมาค้นใน Log ได้ตรงตัว
  • ลำดับ Route: /stats ถูกประกาศก่อน /:id ไม่อย่างนั้น GET /api/tickets/stats จะไปตกที่ /:id แล้วได้ 404 "ไม่พบใบรับซ่อม id stats"
ไอเดียต่อยอด
  1. เปลี่ยนการตรวจข้อมูลด้วยมือเป็น zod 4 Schema พร้อม Middleware validate(schema, 'body') และใช้ z.enum(['waiting', 'repairing', 'done']) แทน Array STATUSES
  2. ย้ายข้อมูลจาก Array ใน Memory ไปเก็บใน SQLite ด้วย better-sqlite3 ให้ข้อมูลอยู่รอดหลัง Restart และเพิ่มคอลัมน์ created_at กับ updated_at
  3. เขียน Test ด้วย node:test + supertest ให้ครอบคลุม 201, 400, 404 และ 500 โดยเพิ่มฟังก์ชัน resetTickets() หรือย้ายข้อมูลเข้าไปใน createApp() เพื่อให้แต่ละ Test เริ่มจากข้อมูลชุดเดิม
  4. เปลี่ยน Logger เป็น pino + pino-http ให้ได้ Log แบบ JSON ที่ค้นหาได้ และส่ง genReqId ให้ใช้ค่า req.id เดียวกับ Header X-Request-Id
  5. เพิ่มการแจ้งเตือนเมื่อสถานะเปลี่ยนเป็น done โดยใช้ EventEmitter (ticketEvents.emit('done', ticket)) แล้วให้ Listener แยกเป็นผู้ส่ง SMS หรือข้อความ ทำให้ Route ไม่ต้องรู้รายละเอียดการแจ้งเตือน

// db.js — เปิดฐานข้อมูล SQLite สร้างตาราง และใส่ข้อมูลเวิร์กช็อปตัวอย่าง
import Database from 'better-sqlite3';
import { mkdirSync } from 'node:fs';
import path from 'node:path';

export function openDb(file = ':memory:') {
  if (file !== ':memory:') {
    // better-sqlite3 ไม่สร้างโฟลเดอร์ให้ ต้องสร้างเองก่อนเปิดไฟล์
    mkdirSync(path.dirname(file), { recursive: true });
  }
  const db = new Database(file);
  db.pragma('journal_mode = WAL');
  db.pragma('foreign_keys = ON');
  db.exec(`
    CREATE TABLE IF NOT EXISTS workshops (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      title TEXT NOT NULL,
      seats INTEGER NOT NULL CHECK (seats > 0)
    );
    CREATE TABLE IF NOT EXISTS registrations (
      id INTEGER PRIMARY KEY AUTOINCREMENT,
      workshop_id INTEGER NOT NULL REFERENCES workshops(id) ON DELETE CASCADE,
      name TEXT NOT NULL,
      email TEXT NOT NULL,
      created_at TEXT NOT NULL DEFAULT (datetime('now')),
      UNIQUE (workshop_id, email)
    );
  `);
  // ใส่ข้อมูลตัวอย่างเฉพาะตอนที่ตารางยังว่าง
  const { n } = db.prepare('SELECT COUNT(*) AS n FROM workshops').get();
  if (n === 0) {
    const insert = db.prepare('INSERT INTO workshops (title, seats) VALUES (@title, @seats)');
    insert.run({ title: 'Node.js สำหรับมือใหม่', seats: 2 });
    insert.run({ title: 'Express 5 REST API', seats: 30 });
  }
  return db;
}

// app.js — สร้าง Express App จาก db ที่ส่งเข้ามา (ทดสอบด้วย ':memory:' ได้)
import express from 'express';
import helmet from 'helmet';
import { z } from 'zod';
import { idParams, registrationBody } from './schemas.js';

class HttpError extends Error {
  constructor(status, error, message, details) {
    super(message);
    Object.assign(this, { status, error, details });
  }
}

// Middleware ตรวจข้อมูล: ผ่านแล้วเก็บค่าที่แปลงแล้วไว้ที่ req.valid
function validate(schema, source) {
  return (req, res, next) => {
    const result = schema.safeParse(req[source]);
    if (!result.success) {
      throw new HttpError(400, 'ValidationError', `ข้อมูลใน ${source} ไม่ถูกต้อง`,
        z.flattenError(result.error).fieldErrors);
    }
    req.valid = { ...req.valid, [source]: result.data };
    next();
  };
}

export function createApp({ db }) {
  // เตรียม Prepared Statement ครั้งเดียว ใช้ซ้ำได้ทุก Request
  const stmt = {
    workshop: db.prepare(`
      SELECT w.id, w.title, w.seats, COUNT(r.id) AS registered
      FROM workshops w LEFT JOIN registrations r ON r.workshop_id = w.id
      WHERE w.id = ? GROUP BY w.id`),
    insertReg: db.prepare(
      'INSERT INTO registrations (workshop_id, name, email) VALUES (@workshopId, @name, @email)'),
    getReg: db.prepare('SELECT id, workshop_id AS workshopId, name, email, created_at AS createdAt FROM registrations WHERE id = ?'),
    deleteReg: db.prepare('DELETE FROM registrations WHERE id = ? AND workshop_id = ?'),
  };

  function loadWorkshop(id) {
    const w = stmt.workshop.get(id);
    if (!w) throw new HttpError(404, 'NotFound', `ไม่พบเวิร์กช็อป id ${id}`);
    return { ...w, seatsLeft: w.seats - w.registered };
  }

  // ตรวจที่นั่งและบันทึกใน Transaction เดียว ป้องกันการจองเกินจำนวนที่นั่ง
  const register = db.transaction((workshopId, name, email) => {
    const w = loadWorkshop(workshopId);
    if (w.seatsLeft <= 0) throw new HttpError(409, 'Conflict', `เวิร์กช็อป "${w.title}" เต็มแล้ว`);
    const info = stmt.insertReg.run({ workshopId, name, email });
    return stmt.getReg.get(info.lastInsertRowid);
  });

  const app = express();
  app.use(helmet());
  app.use(express.json({ limit: '10kb' }));

  app.get('/api/workshops/:id', validate(idParams, 'params'), (req, res) => {
    res.json(loadWorkshop(req.valid.params.id));
  });

  app.post('/api/workshops/:id/registrations',
    validate(idParams, 'params'), validate(registrationBody, 'body'),
    (req, res) => {
      const { id } = req.valid.params;
      const { name, email } = req.valid.body;
      try {
        const reg = register(id, name, email);
        res.status(201).location(`/api/workshops/${id}/registrations/${reg.id}`).json(reg);
      } catch (err) {
        if (err.code === 'SQLITE_CONSTRAINT_UNIQUE') {
          throw new HttpError(409, 'Conflict', `อีเมล ${email} ลงทะเบียนเวิร์กช็อปนี้แล้ว`);
        }
        throw err;
      }
    });

  app.delete('/api/workshops/:id/registrations/:regId', (req, res) => {
    const info = stmt.deleteReg.run(Number(req.params.regId), Number(req.params.id));
    if (info.changes === 0) throw new HttpError(404, 'NotFound', `ไม่พบการลงทะเบียน id ${req.params.regId}`);
    res.status(204).end();
  });

  app.use((req, res) => {
    res.status(404).json({ error: 'NotFound', message: `ไม่พบ ${req.method} ${req.originalUrl}` });
  });

  app.use((err, req, res, next) => {
    if (err instanceof HttpError) {
      return res.status(err.status).json({ error: err.error, message: err.message, ...(err.details && { details: err.details }) });
    }
    if (err.type === 'entity.too.large') {
      return res.status(413).json({ error: 'PayloadTooLarge', message: 'Body ใหญ่เกิน 10kb' });
    }
    if (err.type === 'entity.parse.failed') {
      return res.status(400).json({ error: 'ValidationError', message: 'Body ไม่ใช่ JSON ที่ถูกต้อง' });
    }
    console.error(err);
    res.status(500).json({ error: 'InternalServerError', message: 'เกิดข้อผิดพลาดภายในระบบ' });
  });

  return app;
}

// server.js — เปิดฐานข้อมูลจากไฟล์จริงแล้ว listen พร้อมปิดอย่างนุ่มนวลเมื่อกด Ctrl+C
import path from 'node:path';
import { openDb } from './db.js';
import { createApp } from './app.js';

const port = Number(process.env.PORT ?? 3000);
const dbFile = path.resolve(import.meta.dirname, '..', process.env.DB_FILE ?? 'data/workshops.db');

const db = openDb(dbFile);
const server = createApp({ db }).listen(port, () => {
  console.log(`Workshop API พร้อมที่ http://localhost:${port} (db: ${path.basename(dbFile)})`);
});

for (const signal of ['SIGINT', 'SIGTERM']) {
  process.on(signal, () => {
    console.log(`ได้รับ ${signal} กำลังปิด Server`);
    server.close(() => {
      db.close();
      process.exit(0);
    });
  });
}

import { describe, it, beforeEach, afterEach } from 'node:test';
import assert from 'node:assert/strict';
import request from 'supertest';
import { openDb } from '../src/db.js';
import { createApp } from '../src/app.js';

describe('Workshop Registration API', () => {
  let db;
  let app;

  // สร้างฐานข้อมูลใหม่ทุกเทสต์ เทสต์จึงไม่กระทบกัน
  beforeEach(() => {
    db = openDb(':memory:');
    app = createApp({ db });
  });
  afterEach(() => db.close());

  const somchai = { name: 'สมชาย ใจดี', email: 'Somchai@Example.com' };

  it('GET เวิร์กช็อปที่มีอยู่ ได้ 200 พร้อม seatsLeft', async () => {
    const res = await request(app).get('/api/workshops/1').expect(200);
    assert.equal(res.body.title, 'Node.js สำหรับมือใหม่');
    assert.equal(res.body.seatsLeft, 2);
  });

  it('ลงทะเบียนสำเร็จ ได้ 201, Location และอีเมลตัวเล็ก', async () => {
    const res = await request(app).post('/api/workshops/1/registrations').send(somchai).expect(201);
    assert.equal(res.headers.location, `/api/workshops/1/registrations/${res.body.id}`);
    assert.equal(res.body.email, 'somchai@example.com');
  });

  it('อีเมลผิดรูปแบบ ได้ 400 พร้อม details.email', async () => {
    const res = await request(app).post('/api/workshops/1/registrations')
      .send({ name: 'สมศรี', email: 'not-an-email' }).expect(400);
    assert.equal(res.body.error, 'ValidationError');
    assert.ok(Array.isArray(res.body.details.email));
  });

  it('id ไม่ใช่ตัวเลข ได้ 400', async () => {
    const res = await request(app).get('/api/workshops/abc').expect(400);
    assert.ok(res.body.details.id);
  });

  it('ลงทะเบียนอีเมลเดิมซ้ำ ได้ 409', async () => {
    await request(app).post('/api/workshops/1/registrations').send(somchai).expect(201);
    const res = await request(app).post('/api/workshops/1/registrations')
      .send({ ...somchai, email: 'somchai@example.com' }).expect(409);
    assert.match(res.body.message, /ลงทะเบียนเวิร์กช็อปนี้แล้ว/);
  });

  it('ที่นั่งเต็ม ได้ 409 และจำนวนคนไม่เกินที่นั่ง', async () => {
    await request(app).post('/api/workshops/1/registrations').send({ name: 'A', email: 'a@example.com' }).expect(201);
    await request(app).post('/api/workshops/1/registrations').send({ name: 'B', email: 'b@example.com' }).expect(201);
    const res = await request(app).post('/api/workshops/1/registrations')
      .send({ name: 'C', email: 'c@example.com' }).expect(409);
    assert.match(res.body.message, /เต็มแล้ว/);
    const w = await request(app).get('/api/workshops/1').expect(200);
    assert.equal(w.body.seatsLeft, 0);
  });

  it('ยกเลิกการลงทะเบียน ได้ 204 แล้วลบซ้ำได้ 404', async () => {
    const created = await request(app).post('/api/workshops/1/registrations').send(somchai).expect(201);
    await request(app).delete(created.headers.location).expect(204);
    await request(app).delete(created.headers.location).expect(404);
  });

  it('เวิร์กช็อปที่ไม่มีอยู่ ได้ 404', async () => {
    const res = await request(app).post('/api/workshops/99/registrations').send(somchai).expect(404);
    assert.equal(res.body.message, 'ไม่พบเวิร์กช็อป id 99');
  });

  it('Body ใหญ่เกิน 10kb ได้ 413', async () => {
    await request(app).post('/api/workshops/1/registrations')
      .send({ name: 'x'.repeat(20_000), email: 'big@example.com' }).expect(413);
  });

  it('มี Security Header จาก helmet และไม่มี X-Powered-By', async () => {
    const res = await request(app).get('/api/workshops/2').expect(200);
    assert.equal(res.headers['x-content-type-options'], 'nosniff');
    assert.equal(res.headers['x-powered-by'], undefined);
  });
});

แนวคิด
  • Dependency Injection ของฐานข้อมูล: createApp({ db }) รับฐานข้อมูลจากภายนอก Server จริงส่งไฟล์ data/workshops.db ส่วน Test ส่ง ':memory:' ที่สร้างใหม่ใน beforeEach ทุก Test จึงเริ่มจากสถานะเดียวกัน ไม่ทิ้งข้อมูลค้างให้ Test อื่น และไม่ต้องลบไฟล์หลังเทสต์
  • ตรวจข้อมูลที่ขอบระบบ: Middleware validate(schema, source) เรียก safeParse กับ params และ body ก่อนถึง Handler แล้วเก็บค่าที่แปลงแล้วไว้ที่ req.valid Handler จึงเชื่อได้ว่า id เป็นจำนวนเต็มบวก และ email เป็นตัวพิมพ์เล็กที่ถูกรูปแบบ (ใน Express 5 req.query เป็น Getter ที่เขียนทับไม่ได้ จึงเลือกเก็บผลไว้ใน Property ใหม่)
  • Transaction กันจองเกิน: db.transaction() ห่อการนับที่นั่งว่างและ INSERT ไว้ด้วยกัน ถ้าตรวจแล้วเต็มจะ throw ทำให้ Rollback และ better-sqlite3 เป็นแบบ Synchronous จึงไม่มี Request อื่นแทรกระหว่างสองคำสั่งนี้ได้ภายใน Process เดียวกัน
  • ให้ฐานข้อมูลเป็นด่านสุดท้าย: UNIQUE (workshop_id, email) กันอีเมลซ้ำแม้โค้ดฝั่ง JavaScript จะพลาด เราแค่จับ err.code === 'SQLITE_CONSTRAINT_UNIQUE' แล้วแปลงเป็น 409 แทนที่จะปล่อยให้เป็น 500
  • Prepared Statement ทั้งหมด: ทุกคำสั่งใช้ ? หรือ @name และเตรียมครั้งเดียวตอนสร้าง App จึงปลอดภัยจาก SQL Injection และเร็วกว่าการ prepare ใหม่ทุก Request
  • Security เบื้องต้น: helmet() ใส่ Security Header ชุดใหญ่และลบ X-Powered-By ให้ ส่วน express.json({ limit: '10kb' }) ปฏิเสธ Body ใหญ่ด้วย 413 ก่อนที่จะกิน Memory
  • Graceful Shutdown: เมื่อได้ SIGINT หรือ SIGTERM จะหยุดรับ Connection ใหม่ด้วย server.close() รอ Request ที่ค้างอยู่จบ แล้วค่อย db.close() ไฟล์ฐานข้อมูลจึงไม่ค้างอยู่ในสถานะที่เขียนไม่เสร็จ
