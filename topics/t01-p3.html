import socket

with socket.socket(socket.AF_INET, socket.SOCK_STREAM) as srv:
    # อนุญาตให้ใช้ Port เดิมซ้ำได้ทันที ไม่ต้องรอ TIME-WAIT หมด
    srv.setsockopt(socket.SOL_SOCKET, socket.SO_REUSEADDR, 1)
    srv.bind(("127.0.0.1", 9000))
    srv.listen()
    conn, addr = srv.accept()
    with conn:
        data = conn.recv(1024)
        conn.sendall(data.upper())

    รันคำสั่ง nc -zv 127.0.0.1 9002 แล้วได้ผลลัพธ์ทันทีดังนี้ ข้อใดอธิบายสถานการณ์ได้ถูกต้อง

    nc: connect to 127.0.0.1 port 9002 (tcp) failed: Connection refused
    1. Firewall ทิ้ง Packet SYN ไปเงียบ ๆ โดยไม่ตอบอะไรกลับมา
    2. เครื่องแปลงชื่อ 127.0.0.1 เป็น IP ไม่ได้ เพราะ DNS ล่ม
    3. เครื่องปลายทางได้รับ SYN แล้วตอบ RST กลับมา เพราะไม่มีโปรแกรมใด Listen ที่ Port 9002
    4. 3-way Handshake สำเร็จแล้ว แต่โปรแกรมปลายทางปิด Connection ทันทีหลังจากนั้น

10.0.1.0/24 dev d0 proto kernel scope link src 10.0.1.10
10.0.2.0/24 via 10.0.1.254 dev d0
    1. 10.0.1.200 เพราะ Default Route อยู่บรรทัดแรกของตาราง
    2. 10.0.1.1 เพราะ 10.0.0.0/8 ครอบคลุมทุก IP ที่ขึ้นต้นด้วย 10
    3. ไม่ผ่าน Next hop เลย ส่งออก d0 ตรง ๆ เพราะเป็นวง 10 เหมือนกัน
    4. 10.0.1.254 เพราะ 10.0.2.0/24 เป็น Route ที่ Prefix ยาวที่สุดที่ตรงกับปลายทาง
  9. กลาง

    ใน Routing Lab ของ Part 2 แนวคิดที่ 8 หลังเพิ่มกฎ iptables -t nat -A POSTROUTING -s 10.0.1.0/24 -o r-lan2 -j MASQUERADE ใน Namespace router แล้ว tcpdump ใน lan2 เห็น Source ของ ping เป็น 10.0.2.1 แทนที่จะเป็น 10.0.1.10 ข้อใดอธิบายได้ถูกต้อง

    1. ip_forward ใน router ถูกปิดไป ทำให้ router ตอบ ping แทน lan1
    2. MASQUERADE เปลี่ยน Source IP เป็น IP ของ Interface ขาออก (r-lan2) และจดการแปลงไว้ใน Connection Tracking เพื่อแปลงคำตอบกลับให้ lan1
    3. lan1 เปลี่ยน IP ของตัวเองเป็น 10.0.2.1 ชั่วคราวระหว่าง ping
    4. tcpdump แสดง IP ของ Default Gateway แทน Source จริงเสมอ
  10. ยาก

    บน Trunk แบบ 802.1Q ระหว่าง SW1 กับ R1 Frame ที่อยู่ใน Native VLAN ถูกส่งข้าม Trunk อย่างไร

    1. ใส่ Tag 802.1Q ที่มี VLAN ID 1 เสมอ ไม่ว่า Native VLAN จะเป็นเลขอะไร
    2. ถูกทิ้งที่ Trunk เพราะ Trunk รับเฉพาะ Frame ที่มี Tag
    3. ส่งแบบไม่มี Tag (Untagged) และฝั่งรับจะถือว่าเป็น Native VLAN ของตัวเอง
    4. ใส่ Tag ที่มี VLAN ID 4095 เพื่อบอกว่าเป็น Native VLAN
  11. ยาก

    Router 3 ตัวต่อกันเป็นสามเหลี่ยม Link R1–R2 เป็น 100 Mbps ส่วน R1–R3 และ R3–R2 เป็น 1 Gbps ทุกตัวตั้ง auto-cost reference-bandwidth 1000 ใน OSPF เมื่อนับเฉพาะ Cost ของ Link ระหว่าง Router เส้นทางที่ R1 ใช้ไปหา R2 คือข้อใด

    1. ผ่าน R3 (R1 → R3 → R2) Cost รวม 2
    2. ตรงจาก R1 ไป R2 Cost รวม 10 เพราะผ่าน Router น้อยกว่า
    3. แบ่งโหลดทั้งสองเส้นทางเท่า ๆ กัน เพราะ Cost เท่ากัน
    4. ตรงจาก R1 ไป R2 Cost รวม 1 เพราะ OSPF นับ Cost เป็นจำนวน Hop
  12. ยาก

    ผู้ดูแลเขียน Named Extended ACL SALES-IN แล้วใช้ขาเข้า (in) ที่ Interface Gi0/0.10 ของวง SALES แต่ลืมบรรทัดสุดท้าย permit ip any any ทราฟฟิกจาก 192.168.10.25 ไป Web Server ภายนอก 203.0.113.80 TCP 443 จะเป็นอย่างไร

    ip access-list extended SALES-IN
 permit tcp 192.168.10.0 0.0.0.255 host 192.168.20.10 eq 443
 deny ip 192.168.10.0 0.0.0.255 192.168.20.0 0.0.0.255
!
interface GigabitEthernet0/0.10
 ip access-group SALES-IN in
    1. ผ่าน เพราะไม่ตรงกับบรรทัด deny ที่เขียนไว้
    2. ผ่าน เพราะ ACL นี้ตรวจเฉพาะทราฟฟิกที่ไปวง HR เท่านั้น
    3. ผ่าน เพราะ ACL ขาเข้าไม่มีผลกับทราฟฟิกที่ออกไปอินเทอร์เน็ต
    4. ถูกทิ้ง เพราะไม่ตรงบรรทัดใดเลย จึงโดน Implicit Deny ท้าย ACL
  13. ยาก

    Server ตัวหนึ่งโหลดกฎ nftables ขาเข้าแบบ policy drop ที่ยอมรับเฉพาะ iif "lo", ip protocol icmp และ tcp dport 8080 แต่ลืมกฎ ct state established,related accept ผลที่เกิดขึ้นคือข้อใด

    1. Client ภายนอกเข้า Web App ที่ Port 8080 ไม่ได้เลย
    2. Client เข้า Web App ที่ 8080 ได้ แต่ Server เองเปิด Connection ออกไป เช่น curl หรือ apt update ไม่สำเร็จ เพราะคำตอบขาเข้าถูกทิ้ง
    3. ทุกอย่างใช้ได้ยกเว้น ping เพราะ ICMP ต้องมีกฎ established
    4. ไม่มีผลอะไร เพราะ nftables จำ Connection ขาออกและปล่อยคำตอบให้เองเสมอ
  14. ยาก

    Port Fa0/5 ของ Switch ตั้ง spanning-tree portfast และ spanning-tree bpduguard enable ไว้สำหรับเครื่องพนักงาน วันหนึ่งพนักงานนำ Switch ตัวเล็กมาเสียบที่ Port นี้ และ Switch ตัวเล็กส่ง BPDU ออกมา จะเกิดอะไรขึ้น

    1. Switch ตัวเล็กกลายเป็น Root Bridge ของทั้งวงทันที
    2. Port Fa0/5 เปลี่ยนเป็น Trunk อัตโนมัติเพื่อรับ VLAN ทั้งหมด
    3. Port Fa0/5 ถูกปิดเป็นสถานะ err-disabled ทันที
    4. PortFast ทำให้ Port ไม่สนใจ BPDU และส่งข้อมูลต่อตามปกติ

# ip_plan.py : ตรวจแผน Static IP ของวง LAN บ้าน 192.168.1.0/24
import argparse
import csv
import ipaddress
import sys

# ค่าของวง LAN บ้าน (ตรงกับ Sample Network ของบทเรียน)
LAN = ipaddress.ip_network("192.168.1.0/24")
POOL_START = ipaddress.ip_address("192.168.1.100")
POOL_END = ipaddress.ip_address("192.168.1.199")


def check_device(ip_text, seen):
    """ตรวจ IP หนึ่งตัว คืนค่าเป็นรายการปัญหา (ว่าง = ผ่าน)"""
    problems = []
    try:
        ip = ipaddress.ip_address(ip_text)
    except ValueError:
        return ["รูปแบบ IP ไม่ถูกต้อง"], None

    if ip not in LAN:
        problems.append(f"อยู่นอกวง {LAN}")
    elif ip == LAN.network_address:
        problems.append("เป็น Network Address ใช้กับอุปกรณ์ไม่ได้")
    elif ip == LAN.broadcast_address:
        problems.append("เป็น Broadcast Address ใช้กับอุปกรณ์ไม่ได้")
    elif POOL_START <= ip <= POOL_END:
        problems.append("ชนกับ DHCP Pool .100-.199")

    if ip in seen:
        problems.append(f"IP ซ้ำกับ {seen[ip]}")
    return problems, ip


def suggest_free(used, count):
    """หา IP ว่างนอก DHCP Pool เรียงจากน้อยไปมาก"""
    free = []
    if count == 0:
        return free
    for host in LAN.hosts():
        if POOL_START <= host <= POOL_END or host in used:
            continue
        free.append(str(host))
        if len(free) == count:
            break
    return free


def main():
    parser = argparse.ArgumentParser(description="ตรวจแผน Static IP")
    parser.add_argument("csv_file", nargs="?", default="devices.csv")
    args = parser.parse_args()

    seen = {}          # เก็บว่า IP ไหนถูกใช้โดยอุปกรณ์ชื่ออะไรแล้ว
    total = 0          # จำนวนอุปกรณ์ทั้งหมด
    bad = 0            # จำนวนอุปกรณ์ที่มีปัญหา
    try:
        f = open(args.csv_file, newline="", encoding="utf-8")
    except FileNotFoundError:
        print(f"ไม่พบไฟล์ {args.csv_file} ตรวจชื่อไฟล์และโฟลเดอร์อีกครั้ง")
        sys.exit(2)
    print(f"{'DEVICE':<12}{'IP':<16}RESULT")
    print("-" * 52)
    with f:
        for row in csv.DictReader(f):
            total += 1
            problems, ip = check_device(row["ip"], seen)
            status = "OK" if not problems else "ปัญหา: " + "; ".join(problems)
            print(f"{row['name']:<12}{row['ip']:<16}{status}")
            if problems:
                bad += 1
            # จำเฉพาะ IP ที่อยู่ในวงและยังไม่ถูกจอง
            if ip is not None and ip in LAN and ip not in seen:
                seen[ip] = row["name"]

    print("-" * 52)
    print(f"สรุป: ตรวจ {total} รายการ ผ่าน {total - bad} มีปัญหา {bad}")
    if bad:
        print("IP ว่างที่แนะนำ (นอก DHCP Pool):", ", ".join(suggest_free(seen, bad)))
    # Exit code 1 เมื่อมีปัญหา เพื่อให้ Script อื่นนำไปใช้ต่อได้
    sys.exit(1 if bad else 0)


if __name__ == "__main__":
    main()

# ใช้กับเครื่องของตัวเองเท่านั้น (127.0.0.0/8 และ ::1)
import argparse
import ipaddress
import json
import socket
import sys
from concurrent.futures import ThreadPoolExecutor
from datetime import datetime


def is_allowed_target(host):
    """อนุญาตเฉพาะ Loopback: 127.0.0.0/8 และ ::1"""
    try:
        ip = ipaddress.ip_address(host)
    except ValueError:
        return False          # ไม่รับชื่อโดเมน ป้องกันการชี้ไปเครื่องอื่น
    return ip.is_loopback


def check_port(host, port, timeout=0.5):
    """คืนค่า open, closed (ถูกปฏิเสธ), timeout หรือ error"""
    try:
        with socket.create_connection((host, port), timeout=timeout):
            return "open"
    except ConnectionRefusedError:
        return "closed"       # ได้ RST กลับมา: เครื่องอยู่ แต่ไม่มีโปรแกรมฟัง Port นี้
    except (socket.timeout, TimeoutError):
        return "timeout"      # ไม่มีคำตอบ: อาจโดน Firewall ทิ้ง Packet
    except OSError:
        return "error"        # ปัญหาอื่น เช่น เครื่องไม่รองรับ IPv6


def main():
    parser = argparse.ArgumentParser(description="ตรวจ Port ของ Service บนเครื่องตัวเอง")
    parser.add_argument("config", nargs="?", default="services.json")
    parser.add_argument("--host", help="เขียนทับ host ในไฟล์ config")
    parser.add_argument("--out", default="port-audit.json")
    args = parser.parse_args()

    with open(args.config, encoding="utf-8") as f:
        cfg = json.load(f)
    host = args.host or cfg["host"]

    # ด่านความปลอดภัย: ปฏิเสธทุกเป้าหมายที่ไม่ใช่เครื่องตัวเอง
    if not is_allowed_target(host):
        print(f"ปฏิเสธ: {host} ไม่ใช่ IP แบบ Loopback (127.0.0.0/8 หรือ ::1) โปรแกรมนี้ตรวจได้เฉพาะเครื่องของตัวเอง")
        sys.exit(2)

    services = cfg["services"]
    # ตรวจทุก Port พร้อมกันด้วย Thread เพื่อไม่ต้องรอทีละตัว
    with ThreadPoolExecutor(max_workers=8) as pool:
        states = list(pool.map(lambda s: check_port(host, s["port"]), services))

    results = []
    fails = 0
    print(f"ตรวจ {host} เวลา {datetime.now():%Y-%m-%d %H:%M:%S}")
    print(f"{'SERVICE':<11}{'PORT':>5}  {'EXPECT':<7}{'ACTUAL':<8}RESULT")
    for svc, state in zip(services, states):
        ok = state == svc["expect"]
        fails += 0 if ok else 1
        result = "PASS" if ok else "FAIL"
        print(f"{svc['name']:<11}{svc['port']:>5}  {svc['expect']:<7}{state:<8}{result}")
        results.append({**svc, "actual": state, "result": result})

    report = {
        "host": host,
        "checked_at": datetime.now().isoformat(timespec="seconds"),
        "fails": fails,
        "results": results,
    }
    with open(args.out, "w", encoding="utf-8") as f:
        json.dump(report, f, ensure_ascii=False, indent=2)
    print(f"สรุป: FAIL {fails} จาก {len(services)} รายการ บันทึกรายงานที่ {args.out}")
    sys.exit(1 if fails else 0)


if __name__ == "__main__":
    main()

#!/usr/bin/env bash
# verify.sh : ทดสอบนโยบาย Firewall ทีละข้อ แล้วพิมพ์ PASS/FAIL
pass=0; fail=0

check() {
    # $1 = คำอธิบาย, $2 = ผลที่คาดหวัง (allow/deny), ที่เหลือ = คำสั่งทดสอบ
    local desc="$1" expect="$2"; shift 2
    if "$@" > /dev/null 2>&1; then got="allow"; else got="deny"; fi
    if [ "$got" = "$expect" ]; then
        echo "PASS  $desc (ได้ $got)"; pass=$((pass + 1))
    else
        echo "FAIL  $desc (คาดว่า $expect แต่ได้ $got)"; fail=$((fail + 1))
    fi
}

<h1>Payroll Admin</h1>
อ่านค่า counter ให้เป็น
  • กฎ accept TCP 443 จับได้ 1 Packet 60 bytes: คือ SYN ตัวแรกของ curl เท่านั้น Packet ที่เหลือของ Connection นี้ (ACK, คำขอ HTTP) ถูกปล่อยโดยกฎ ct state established,related ซึ่งอยู่บรรทัดแรก นี่คือหลักฐานว่า Firewall ทำงานแบบ Stateful จริง
  • กฎ drop จับได้ 3 Packet 204 bytes: มาจาก SYN ไปยัง Port 8080 จำนวน 2 ตัว (ตัวละ 60 bytes เพราะ curl ส่ง SYN ซ้ำเมื่อไม่ได้คำตอบภายใน 1 วินาที และเรากำหนด -m 2) รวมกับ ICMP Echo Request ของ ping อีก 1 ตัว (84 bytes) รวมเป็น 60 + 60 + 84 = 204 bytes
  • ping ไป Router และ ping จาก HR ไม่ปรากฏใน counter เลย: ตัวแรกวิ่งผ่าน Hook input ส่วนตัวหลังเข้ามาทาง r-hr จึงไม่ตรงเงื่อนไข iifname "r-sales" และคำตอบที่กลับจาก SALES ถูกปล่อยด้วยกฎ established

จำนวน Packet ของกฎ drop อาจต่างไปเล็กน้อยบนเครื่องของคุณ ขึ้นกับจังหวะการส่ง SYN ซ้ำของ Kernel แต่กฎ accept ควรนับได้ 1 Packet ต่อการรัน verify.sh หนึ่งครั้ง
