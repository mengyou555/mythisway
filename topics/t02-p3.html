def test_average_works():

  6. ยาก

    AI เขียน Route ค้นหาโน้ตดังนี้ conn.execute(f"SELECT id, owner, body FROM notes WHERE owner = '{owner}'") เมื่อส่ง owner=' OR '1'='1 ระบบคืนโน้ตของทุกคน วิธีแก้ที่ถูกต้องและเป็นมาตรฐานที่สุดคือข้อใด

    3. ครอบคำสั่งด้วย try/except Exception: pass เพื่อไม่ให้ Error หลุดไปถึงผู้ใช้

เขียนฟังก์ชัน Python ชื่อ parse_minutes(text) แปลงข้อความเวลาเรียนภาษาไทยเป็นจำนวนนาที (int)
ตัวอย่างข้อความ: "45", "45 นาที", "1 ชม.", "1 ชม. 30 นาที"
"""โค้ดที่ AI เขียนให้ในคำตอบแรก (ยังไม่ได้ตรวจ)"""


def parse_minutes(text):
    # ตัดหน่วยออก แล้วแยกตัวเลขด้วยช่องว่าง
    text = text.replace("ชม.", "").replace("นาที", "")
    parts = text.split()
    if len(parts) == 2:
        return int(parts[0]) * 60 + int(parts[1])
    return int(parts[0])

"""API บันทึกโน้ตการเรียน (โค้ดที่ AI เขียน ยังไม่ผ่านการรีวิว)"""

import sqlite3

from flask import Flask, jsonify, request


def create_app(db_path="notes.db", secret_key=None):
    app = Flask(__name__)
    app.config["SECRET_KEY"] = "notes-secret-2026"
    password = "admin1234"
    app.config["ADMIN_PASSWORD"] = password

    def get_conn():
        conn = sqlite3.connect(db_path)
        conn.row_factory = sqlite3.Row
        return conn

    with get_conn() as conn:
        conn.execute(
            "CREATE TABLE IF NOT EXISTS notes "
            "(id INTEGER PRIMARY KEY, owner TEXT NOT NULL, body TEXT NOT NULL)"
        )

    @app.post("/api/notes")
    def add_note():
        data = request.get_json()
        with get_conn() as conn:
            cur = conn.execute(
                "INSERT INTO notes (owner, body) VALUES (?, ?)", (data["owner"], data["body"])
            )
        return jsonify({"id": cur.lastrowid}), 201

    @app.get("/api/notes")
    def list_notes():
        owner = request.args.get("owner", "")
        with get_conn() as conn:
            rows = conn.execute(f"SELECT id, owner, body FROM notes WHERE owner = '{owner}'")
            notes = [dict(row) for row in rows]
        return jsonify(notes)

    return app


if __name__ == "__main__":
    create_app().run(host="0.0.0.0", debug=True)

ข้อ 4

คำตอบ: restore คำสั่งเต็มคือ git restore duration.py ซึ่งนำเนื้อหาไฟล์จาก Index มาทับไฟล์ใน Working Directory เมื่อยังไม่ได้ git add เนื้อหาใน Index ก็คือเนื้อหาของ Commit ล่าสุด ไฟล์จึงกลับไปเหมือน Commit ล่าสุดทันที ระวังว่าการแก้ไขที่ยังไม่ Commit ของไฟล์นี้จะหายถาวร

ถ้าตอบ checkout ก็ถือว่าถูก เพราะ git checkout -- duration.py เป็นรูปแบบเก่าที่ได้ผลเหมือนกัน (Git ตอบว่า Updated 1 path from the index) แต่หลักสูตรนี้ใช้ git restore เพราะชื่อคำสั่งบอกหน้าที่ชัดกว่า และไม่ปนกับการสลับ Branch

เปรียบเทียบกับคำสั่งย้อนกลับแบบอื่นในแนวคิดที่ 4

  • git restore --staged duration.py ใช้เมื่อ git add ไปแล้ว คำสั่งนี้ถอดไฟล์ออกจาก Stage แต่การแก้ไขในไฟล์ยังอยู่
  • git reset --hard HEAD ทิ้งการแก้ไขที่ยังไม่ Commit ของทุกไฟล์ ใช้เมื่อตั้งใจจริงเท่านั้น
  • git revert <commit> ใช้กับการแก้ไขที่ Commit ไปแล้ว โดยสร้าง Commit ใหม่ที่กลับผลของ Commit นั้น History เดิมจึงยังอยู่ครบ

The current statement uses 0, and there are 1 supplied. ส่วน Placeholder แบบมีชื่อ เช่น :topic ต้องส่งค่าเป็น dict อย่าง {"topic": f"%{topic}%"} ถ้าใช้คู่กับ Tuple แบบในโจทย์ Python 3.12 จะขึ้น DeprecationWarning และตั้งแต่ Python 3.14 จะกลายเป็น sqlite3.ProgrammingError คำตอบของข้อนี้จึงเป็น ?

คำตอบ: -euo pipefail บรรทัดเต็มคือ set -euo pipefail ซึ่งรวม 3 Option ไว้ในบรรทัดเดียว (เขียนแยกเป็น set -e -u -o pipefail ก็ได้ผลเหมือนกัน)

  • -e หยุด Script ทันทีเมื่อคำสั่งใดคืน Exit Code ที่ไม่ใช่ 0 เช่น ถ้า pytest -q ล้ม Script จะไม่ไปถึงบรรทัด "ผ่านทุกด่าน"
  • -u ถือว่าการใช้ตัวแปรที่ไม่ได้ตั้งค่าเป็น Error (Bash แจ้ง BUILD_DIR: unbound variable) แทนที่จะได้ค่าว่างแบบเงียบ ๆ ช่วยกันบั๊กอย่าง rm -rf "$BUILD_DIR"/* ซึ่งถ้าลืมตั้งค่า BUILD_DIR จะกลายเป็น rm -rf /* และลบทุกไฟล์ที่ผู้ใช้มีสิทธิ์ลบ
  • -o pipefail ทำให้ Pipe ทั้งเส้นล้มถ้าคำสั่งใดใน Pipe ล้ม ปกติ Bash ดูแค่ Exit Code ของคำสั่งสุดท้าย เช่น pytest -q | tee test.log จะถือว่าสำเร็จเพราะ tee สำเร็จ แม้ pytest จะล้ม

เมื่อด่านใดล้ม Script จะคืน Exit Code ที่ไม่ใช่ 0 pre-commit Hook และ GitHub Actions จึงใช้ผลนี้ตัดสินได้ว่าจะให้ Commit หรือ Merge ผ่านหรือไม่

"""ตรวจฟังก์ชัน parse_minutes กับตาราง Acceptance Criteria"""

import sys

# ตาราง AC: (ข้อความที่ผู้ใช้พิมพ์, ผลที่ต้องได้) ถ้าต้องได้ Error ให้ใส่คลาสของ Error
CASES = [
    ("45", 45),
    ("45 นาที", 45),
    ("1 ชม.", 60),
    ("1 ชม. 30 นาที", 90),
    ("  20 นาที ", 20),
    ("-5 นาที", ValueError),
    ("", ValueError),
    ("abc", ValueError),
]


def run_case(func, text, expected):
    """รันหนึ่งกรณี คืนค่า (ผ่านหรือไม่, ผลที่ได้จริงเป็นข้อความ)"""
    try:
        got = func(text)
    except Exception as exc:  # จับทุก Error เพื่อรายงานผล ไม่ได้ซ่อน Error
        got_text = type(exc).__name__
        ok = isinstance(expected, type) and isinstance(exc, expected)
        return ok, got_text
    ok = not isinstance(expected, type) and got == expected
    return ok, repr(got)


def main() -> int:
    # เลือกว่าจะตรวจไฟล์ไหน: ai = โค้ดของ AI, fixed = ฉบับแก้แล้ว
    target = sys.argv[1] if len(sys.argv) > 1 else "ai"
    if target == "ai":
        from ch1_ai_parse import parse_minutes
    else:
        from ch1_parse import parse_minutes

    print(f"ตรวจ parse_minutes จาก: {target}")
    passed = 0
    for no, (text, expected) in enumerate(CASES, start=1):
        ok, got = run_case(parse_minutes, text, expected)
        want = expected.__name__ if isinstance(expected, type) else repr(expected)
        status = "PASS" if ok else "FAIL"
        print(f"[{status}] AC{no} {text!r:20} ได้ {got:12} ต้องได้ {want}")
        passed += ok
    print(f"ผ่าน {passed}/{len(CASES)}")
    # คืน exit code 1 ถ้ามีข้อไม่ผ่าน เพื่อใช้ใน Script อื่นได้
    return 0 if passed == len(CASES) else 1


if __name__ == "__main__":
    sys.exit(main())
  • CASES เก็บ AC เป็นข้อมูล ข้อที่ต้องได้ Error เก็บเป็นคลาส ValueError จึงแยกจากค่าปกติได้ด้วย isinstance(expected, type)
  • run_case จับทุก Exception เพื่อ "รายงาน" ไม่ใช่เพื่อซ่อน ข้อจะผ่านก็ต่อเมื่อ Error ที่เกิดเป็นชนิดที่คาดไว้ ข้อที่ต้องได้ ValueError แต่ได้ IndexError จึงนับเป็น FAIL
  • คำสั่ง Import อยู่ใน if ทำให้เลือกไฟล์ที่จะตรวจจาก Argument ได้ และ sys.exit(main()) คืน Exit Code 1 เมื่อมีข้อไม่ผ่าน Script อื่นอย่าง scripts/check.sh จึงนำไปใช้เป็นด่านตรวจได้

"""แปลงข้อความเวลาเรียนภาษาไทยเป็นจำนวนนาที (ฉบับแก้แล้ว)"""

import re

# รูปแบบที่รับ: "45", "45 นาที", "1 ชม.", "1 ชม. 30 นาที", "1 ชม. 30"
_PATTERN = re.compile(r"(?:(\d+)\s*ชม\.)?\s*(?:(\d+)\s*(?:นาที)?)?")


def parse_minutes(text: str) -> int:
    """คืนจำนวนนาทีเป็น int ถ้ารูปแบบผิดให้ยก ValueError"""
    cleaned = text.strip()
    if not cleaned:
        raise ValueError("ข้อความว่าง")
    if cleaned.startswith("-"):
        raise ValueError("นาทีต้องไม่ติดลบ")
    match = _PATTERN.fullmatch(cleaned)
    # ต้องตรงทั้งข้อความ และต้องมีตัวเลขอย่างน้อยหนึ่งส่วน
    if match is None or (match.group(1) is None and match.group(2) is None):
        raise ValueError(f"รูปแบบไม่ถูกต้อง: {text!r}")
    hours = int(match.group(1) or 0)
    minutes = int(match.group(2) or 0)
    return hours * 60 + minutes
  • (?:(\d+)\s*ชม\.)? คือส่วนชั่วโมงที่มีหรือไม่มีก็ได้ จุดหลัง ชม ต้องเขียนเป็น \. เพราะจุดเปล่า ๆ ใน Regular Expression แปลว่าอักขระอะไรก็ได้
  • (?:(\d+)\s*(?:นาที)?)? คือส่วนนาที คำว่า นาที จะใส่หรือไม่ใส่ก็ได้ ทั้ง "45" และ "45 นาที" จึงผ่าน
  • fullmatch บังคับให้รูปแบบตรงกับ "ทั้งข้อความ" ข้อความที่มีส่วนเกินหรือเรียงผิดลำดับจึงถูกปฏิเสธ นี่คือการกำหนดสิ่งที่ยอมรับไว้ชัดเจน (Allowlist) แล้วปฏิเสธทุกอย่างที่ไม่ตรง ซึ่งปลอดภัยกว่าการ replace ต่อกันหลายครั้ง
  • ตรวจข้อความว่างและเครื่องหมายลบก่อนจับรูปแบบ เพื่อให้ข้อความ Error บอกสาเหตุได้ตรงจุด

def total_minutes(sessions: list[tuple[str, int]]) -> int:
    """รวมนาทีของทุกรายการ"""
    return sum(minutes for _, minutes in sessions)


def top_topics(sessions: list[tuple[str, int]], n: int = 3) -> list[tuple[str, int]]:
    """คืนหัวข้อที่เรียนรวมนานที่สุด n อันดับ เป็นลิสต์ของ (หัวข้อ, นาทีรวม)"""
    if n < 1:
        raise ValueError("n ต้องมากกว่า 0")
    totals: dict[str, int] = {}

"""API บันทึกโน้ตการเรียน (แก้แล้วหลัง Security Review)"""

import os
import re
import sqlite3

from flask import Flask, jsonify, request

OWNER_PATTERN = re.compile(r"[a-z0-9_]{1,30}")
MAX_BODY = 500


def parse_note(payload: object) -> tuple[str, str]:
    """ตรวจข้อมูลโน้ต คืน (owner, body) หรือยก ValueError พร้อมข้อความภาษาไทย"""
    if not isinstance(payload, dict):
        raise ValueError("ต้องส่งข้อมูลเป็น JSON object")
    owner = payload.get("owner")
    body = payload.get("body")
    if not isinstance(owner, str) or not OWNER_PATTERN.fullmatch(owner):
        raise ValueError("owner ต้องเป็น a-z, 0-9 หรือ _ ยาว 1-30 ตัวอักษร")
    if not isinstance(body, str) or not body.strip():
        raise ValueError("body ต้องเป็นข้อความและห้ามว่าง")
    if len(body) > MAX_BODY:
        raise ValueError(f"body ยาวได้ไม่เกิน {MAX_BODY} ตัวอักษร")
    return owner, body.strip()


def create_app(db_path: str = "notes.db", secret_key: str | None = None) -> Flask:
    # Secret มาจากผู้เรียกหรือ Environment Variable เท่านั้น ไม่มีค่าในโค้ด
    key = secret_key or os.environ.get("SECRET_KEY")
    if not key:
        raise RuntimeError("ไม่พบ SECRET_KEY: ตั้งค่าใน Environment Variable ก่อนรันแอป")
    app = Flask(__name__)
    app.config["SECRET_KEY"] = key

    def get_conn() -> sqlite3.Connection:
        conn = sqlite3.connect(db_path)
        conn.row_factory = sqlite3.Row
        return conn

    with get_conn() as conn:
        conn.execute(
            "CREATE TABLE IF NOT EXISTS notes "
            "(id INTEGER PRIMARY KEY, owner TEXT NOT NULL, body TEXT NOT NULL)"
        )

    @app.post("/api/notes")
    def add_note():
        try:
            owner, body = parse_note(request.get_json(silent=True))
        except ValueError as exc:
            return jsonify({"error": str(exc)}), 400
        with get_conn() as conn:
            cur = conn.execute("INSERT INTO notes (owner, body) VALUES (?, ?)", (owner, body))
        return jsonify({"id": cur.lastrowid}), 201

    @app.get("/api/notes")
    def list_notes():
        owner = request.args.get("owner", "")
        if not OWNER_PATTERN.fullmatch(owner):
            return jsonify({"error": "owner ไม่ถูกต้อง"}), 400
        with get_conn() as conn:
            # Parameterized Query: ค่า owner ถูกส่งแยกจากคำสั่ง SQL
            rows = conn.execute(
                "SELECT id, owner, body FROM notes WHERE owner = ? ORDER BY id", (owner,)
            )
            notes = [dict(row) for row in rows]
        return jsonify(notes)

    return app
