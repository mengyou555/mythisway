ฟีเจอร์ทั้งหมด

  • หน้าแรก /: แสดงรายการล่าสุด 50 รายการ เรียงวันที่ใหม่ไปเก่า และฟอร์มเพิ่มรายการ ได้แก่ วันที่ (YYYY-MM-DD), ประเภท (income หรือ expense), หมวดหมู่, จำนวนเงินเป็นบาท และหมายเหตุที่ไม่บังคับกรอก
  • เก็บเงินเป็นสตางค์: ตาราง transactions(id, date, kind, category, amount_satang, note) มี CHECK ให้ kind เป็น income หรือ expense เท่านั้น และ amount_satang > 0 เงินเก็บเป็นจำนวนเต็มหน่วยสตางค์ (1 บาท = 100 สตางค์) เพื่อเลี่ยงปัญหาทศนิยมของ float และแสดงผลเป็น 1,250.50 บาท
  • ตรวจข้อมูลใน validation.py: วันที่ต้องผ่าน date.fromisoformat, หมวดหมู่ต้องอยู่ในรายการ อาหาร เดินทาง ที่พัก การเรียน บันเทิง เงินเดือน รายได้เสริม อื่น ๆ, จำนวนเงินต้องมากกว่า 0 มีทศนิยมไม่เกิน 2 ตำแหน่ง และไม่เกิน 1,000,000 บาท ถ้าข้อมูลผิดจะแสดงฟอร์มเดิมพร้อมข้อความ Error ภาษาไทยและ HTTP 400
  • หน้า /summary?month=2026-10: สรุปรายรับ รายจ่าย คงเหลือ และรายจ่ายแยกตามหมวดหมู่ของเดือนที่เลือก
  • ลบและส่งออก: POST /delete/<id> ลบรายการแล้ว Redirect กลับหน้าแรก (ไม่รับ GET) และ GET /export.csv?month=2026-10 ส่งออกไฟล์ CSV แบบ UTF-8 ที่มี Header date,kind,category,amount,note พร้อมกัน CSV Injection
  • ความปลอดภัย: SQL ทุกคำสั่งใช้ Parameter ?, อ่าน SECRET_KEY จาก Environment Variable และแอปไม่ยอมเริ่มถ้าไม่มีค่านี้, เปิด Autoescape ของ Jinja2 ตลอด และไม่มี debug=True ในโค้ด
  • Test 42 ข้อด้วย pytest 9.1.1: tests/test_validation.py 24 ข้อ, tests/test_db.py 6 ข้อ และ tests/test_routes.py 12 ข้อ ครอบคลุมการเพิ่ม แสดง สรุป ลบ ส่งออก ข้อมูลผิด SQL Injection และ XSS
  • Quality Gate: scripts/check.sh 4 ด่าน (ruff check, ruff format --check, pytest, pip-audit) เรียกจาก Hook .git/hooks/pre-commit และ .github/workflows/check.yml
  • เอกสาร: SPEC.md, AGENTS.md, CLAUDE.md (ชี้ไปที่ AGENTS.md), AI_LOG.md 10 รายการ และ README.md ที่บอกวิธีติดตั้ง รัน ทดสอบ และส่วนที่ใช้ AI

import pytest

from money import format_baht, parse_amount, to_csv_amount
from validation import validate_month, validate_transaction

VALID_FORM = {
    "date": "2026-10-02",
    "kind": "expense",
    "category": "อาหาร",
    "amount": "65",
    "note": "ข้าวกะเพรา",
}


@pytest.mark.parametrize(
    ("text", "satang"),
    [("1,250.50", 125050), ("65", 6500), ("0.5", 50), ("1000000", 100_000_000)],
)
def test_parse_amount_valid(text, satang):
    assert parse_amount(text) == satang


@pytest.mark.parametrize(
    ("text", "message"),
    [
        ("0", "มากกว่า 0"),
        ("-5", "ตัวเลขที่มากกว่า 0"),
        ("abc", "ตัวเลขที่มากกว่า 0"),
        ("๑๐๐", "ตัวเลขที่มากกว่า 0"),
        ("12.345", "ทศนิยมได้ไม่เกิน 2 ตำแหน่ง"),
        ("1000000.01", "ไม่เกิน 1,000,000 บาท"),
    ],
)
def test_parse_amount_invalid(text, message):
    with pytest.raises(ValueError, match=message):
        parse_amount(text)


def test_format_baht_and_csv_amount():
    assert format_baht(125050) == "1,250.50 บาท"
    assert format_baht(-1500) == "-15.00 บาท"
    assert to_csv_amount(2500000) == "25000.00"


def test_valid_form_returns_clean_data():

    assert errors == []
    assert data == {

import re

# 1 บาท = 100 สตางค์ เพดานต่อรายการคือ 1,000,000 บาท
MAX_SATANG = 1_000_000 * 100
# ตัวเลข 0-9 อย่างน้อย 1 ตัว ตามด้วยทศนิยมได้ (ใช้ [0-9] เพื่อไม่รับเลขไทยหรือเลขของภาษาอื่น)
AMOUNT_RE = re.compile(r"[0-9]+(\.[0-9]+)?")


def parse_amount(text: str) -> int:
    """แปลงข้อความ เช่น "1,250.50" เป็นสตางค์ 125050 ถ้าข้อมูลไม่ถูกต้องจะ raise ValueError"""
    cleaned = text.strip().replace(",", "")
    if not AMOUNT_RE.fullmatch(cleaned):
        raise ValueError("จำนวนเงินต้องเป็นตัวเลขที่มากกว่า 0 เช่น 65 หรือ 1250.50")
    whole, _, fraction = cleaned.partition(".")
    whole = whole.lstrip("0") or "0"
    if len(fraction) > 2:
        raise ValueError("จำนวนเงินมีทศนิยมได้ไม่เกิน 2 ตำแหน่ง")
    # ตรวจความยาวก่อนแปลงเป็น int เพื่อกันตัวเลขที่ยาวผิดปกติ
    if len(whole) > 7:
        raise ValueError("จำนวนเงินต้องไม่เกิน 1,000,000 บาท")
    satang = int(whole) * 100 + int(fraction.ljust(2, "0"))
    if satang == 0:
        raise ValueError("จำนวนเงินต้องมากกว่า 0")
    if satang > MAX_SATANG:
        raise ValueError("จำนวนเงินต้องไม่เกิน 1,000,000 บาท")
    return satang


def format_baht(satang: int) -> str:
    """แสดงสตางค์เป็นข้อความ เช่น 125050 เป็น "1,250.50 บาท" ค่าติดลบจะมีเครื่องหมาย - นำหน้า"""
    sign = "-" if satang < 0 else ""
    baht, rest = divmod(abs(satang), 100)
    return f"{sign}{baht:,}.{rest:02d} บาท"

def validate_transaction(form: Mapping[str, str]) -> tuple[dict, list[str]]:
    """ตรวจฟอร์มเพิ่มรายการ คืน (ข้อมูลที่ผ่านการตรวจ, รายการข้อความ Error)"""

    return month

import sqlite3
from contextlib import closing

SCHEMA = """
CREATE TABLE IF NOT EXISTS transactions (
    id INTEGER PRIMARY KEY,
    date TEXT NOT NULL,
    kind TEXT NOT NULL CHECK (kind IN ('income', 'expense')),
    category TEXT NOT NULL,
    amount_satang INTEGER NOT NULL CHECK (amount_satang > 0),
    note TEXT DEFAULT ''
)
"""


def get_conn(db_path: str) -> sqlite3.Connection:
    """เปิด Connection ที่คืนแถวเป็น sqlite3.Row เพื่อเรียกคอลัมน์ด้วยชื่อได้"""
    conn = sqlite3.connect(db_path)
    conn.row_factory = sqlite3.Row
    return conn


def init_db(db_path: str) -> None:
    """สร้างตาราง transactions ถ้ายังไม่มี"""
    with closing(get_conn(db_path)) as conn:
        conn.execute(SCHEMA)
        conn.commit()


def add_transaction(db_path: str, data: dict) -> int:
    """บันทึกรายการที่ผ่าน validate_transaction แล้ว คืน id ของแถวใหม่"""
    with closing(get_conn(db_path)) as conn:
        cur = conn.execute(
            "INSERT INTO transactions (date, kind, category, amount_satang, note)"
            " VALUES (?, ?, ?, ?, ?)",
            (data["date"], data["kind"], data["category"], data["amount_satang"], data["note"]),
        )
        conn.commit()
        return cur.lastrowid


def list_transactions(db_path: str, limit: int = 50) -> list[sqlite3.Row]:
    """คืนรายการล่าสุดเรียงวันที่ใหม่ไปเก่า ถ้าวันเดียวกันรายการที่เพิ่มทีหลังขึ้นก่อน"""
    with closing(get_conn(db_path)) as conn:
        return conn.execute(
            "SELECT * FROM transactions ORDER BY date DESC, id DESC LIMIT ?", (limit,)
        ).fetchall()


def delete_transaction(db_path: str, tx_id: int) -> bool:
    """ลบรายการตาม id คืน True ถ้าลบได้ และ False ถ้าไม่พบรายการนั้น"""
    with closing(get_conn(db_path)) as conn:
        cur = conn.execute("DELETE FROM transactions WHERE id = ?", (tx_id,))
        conn.commit()
        return cur.rowcount > 0

    """คืนทุกรายการของเดือน month (YYYY-MM) เรียงวันที่เก่าไปใหม่ ใช้กับการส่งออก CSV"""

def month_summary(db_path: str, month: str) -> dict:
    """สรุปรายรับ รายจ่าย คงเหลือ และรายจ่ายแยกหมวดของเดือน month หน่วยเป็นสตางค์"""
    with closing(get_conn(db_path)) as conn:
        totals = conn.execute(
            """
            SELECT
                COALESCE(SUM(CASE WHEN kind = 'income' THEN amount_satang END), 0) AS income,
                COALESCE(SUM(CASE WHEN kind = 'expense' THEN amount_satang END), 0) AS expense
            FROM transactions
            WHERE substr(date, 1, 7) = ?
            """,
            (month,),
        ).fetchone()
        by_category = conn.execute(
            """
            SELECT category, SUM(amount_satang) AS total
            FROM transactions
            WHERE kind = 'expense' AND substr(date, 1, 7) = ?
            GROUP BY category
            ORDER BY total DESC, category
            """,
            (month,),
        ).fetchall()
    return {
        "income": totals["income"],
        "expense": totals["expense"],
        "balance": totals["income"] - totals["expense"],

    หลัง Push ให้เปิดแท็บ Actions ของ Repository แล้วรอให้ Workflow quality-gate ทำงานจนจบ ถ้าผ่าน Badge ใน README จะแสดงสถานะผ่าน ถ้าในเครื่องผ่านแต่บน GitHub ล้ม ให้อ่าน Log ของด่านที่ล้มก่อนเสมอ สาเหตุที่พบบ่อยคือ pip-audit เจอช่องโหว่ที่เพิ่งประกาศ หรือมีไฟล์ที่ลืม git add

    Security

    ถ้าเผลอ Commit Secret ขึ้น GitHub ไปแล้ว ให้ถือว่า Secret นั้นรั่วทันที แม้จะลบออกใน Commit ถัดไป เพราะยังอยู่ในประวัติ Git ขั้นแรกคือเปลี่ยน (Rotate) Secret ตัวนั้นทิ้งก่อน แล้วจึงล้างประวัติ ห้ามสลับลำดับ เพราะระหว่างที่ล้างประวัติ ใครก็ตามที่ Clone หรือ Fork ไปแล้วยังใช้ค่าเดิมได้

  3. ตรวจ License ว่าใช้กับงานของเราได้ เช่น MIT, BSD และ Apache-2.0 ใช้ได้กว้าง ส่วน GPL มีเงื่อนไขเรื่องการเปิด Source Code
  4. ติดตั้งใน venv แบบ Pin เวอร์ชันใน requirements.txt แล้วรัน pip-audit -r requirements.txt
  5. บันทึกเหตุผลลง AI_LOG.md

ตัวอย่าง: python3 -m pip index versions flask-sqlite-helper-pro ตอบว่า No matching distribution found แปลว่าเป็นชื่อที่ AI แต่งขึ้น แต่ถ้าวันหนึ่งชื่อนี้มีอยู่จริงขึ้นมา ก็ยังต้องตรวจอายุและ Source Code ต่อ เพราะอาจเป็นแพ็กเกจที่ผู้โจมตีเพิ่งจดไว้ และไม่รันคำสั่งที่ AI ให้มาแบบ sudo pip install หรือ curl ... | bash โดยไม่อ่านก่อน

3. เล่าขั้นตอนรีวิว Diff ที่ AI ส่งมา และยกตัวอย่าง Patch ที่คุณเคยปฏิเสธ (Mid)

แนวคำตอบ: ผมรีวิวเป็น 3 ชั้น คืออ่านเอง ยืนยันด้วยเครื่องมือ แล้วตัดสินใจพร้อมบันทึกเหตุผล

  • ภาพรวม: งานนี้อยู่ใน Spec ไหม แล้วใช้ git apply --check และ --stat ดูว่าแตะไฟล์อะไรบ้าง มีไฟล์ใน tests/ หรือ requirements.txt ถูกแก้หรือไม่
  • ทีละ Hunk: ตรงตาม Acceptance Criteria ไหม จัดการกรณีขอบหรือยัง ชื่อและ Type Hint ชัดไหม มี Import ที่ไม่ใช้ไหม และด้านความปลอดภัย ได้แก่ SQL ที่ต่อ String, |safe, debug=True, host="0.0.0.0", shell=True และ Secret ในโค้ด
  • ยืนยันด้วยเครื่องมือ: ใส่ Patch ชั่วคราว รัน ruff ที่เปิดกฎ S รัน Test ทั้งชุด และถ้าสงสัยช่องโหว่ก็ลองโจมตีแอปของตัวเองบนเครื่องตัวเอง

ตัวอย่าง: ผมขอหน้าค้นหาแบบไม่อ้างอิง Spec แล้วได้ Patch 3 ไฟล์ที่ใช้ f-string ใน LIKE, เติม |safe ในหน้าแรก และเปิด run(debug=True, host="0.0.0.0") ruff จับได้ S608 กับ S104 Test เรื่อง XSS ล้ม 1 ข้อ และการทดลอง UNION SELECT ดึงโครงสร้างตารางออกมาได้จริง ผมจึงปฏิเสธ ถอดด้วย git apply -R แล้วนำฟีเจอร์กลับไปเริ่มที่ Spec และ Test ใหม่ ประเด็นที่ควรพูดคือเครื่องมือไม่ได้จับทุกอย่าง ruff ไม่เตือน debug=True ในรูป create_app().run(...) การอ่านของคนจึงยังจำเป็น

4. รู้ได้อย่างไรว่า Test ที่ AI เขียนมีประโยชน์จริง และจะกันไม่ให้ AI แก้ Test เพื่อให้ผ่านได้อย่างไร (Mid)

แนวคำตอบ: Test ที่ดีต้องล้มได้เมื่อโค้ดผิด ผมจึงตรวจ Test ทั้งก่อนและหลังมีโค้ด

  • ให้ AI เขียน Test จาก Acceptance Criteria ก่อนมีโค้ด แล้วรันให้เห็นว่าล้มด้วยเหตุผลที่ถูกต้อง เช่น No module named 'money' ไม่ใช่ล้มเพราะ Test พิมพ์ผิด
  • ปฏิเสธ Test ที่ไม่พิสูจน์อะไร เช่น assert ... is not None, Test ที่ไม่มี assert, Test ที่คัดลอกสูตรจากโค้ดจริงมาเทียบกับตัวเอง และ pytest.raises(Exception) ที่กว้างเกินไป
  • เติมกรณีที่ AI มักลืม เช่น ค่าขอบ ข้อมูลภาษาไทย เลขไทย และ Test ด้านความปลอดภัยอย่าง SQL Injection และ XSS
  • ลองทำให้โค้ดผิดโดยตั้งใจ แล้วดูว่ามี Test ล้มจริง เหมือนตอนใส่ inline-month.patch แล้ว Test ข้อ SQL Injection ล้มทันที
  • กันการแก้ Test ด้วยกติกาใน AGENTS.md และตรวจ git diff --stat -- tests/ ก่อนทุก Commit ถ้า AI คิดว่า Test ผิด ต้องอธิบายเหตุผลแล้วรอคนอนุมัติ

ตัวอย่าง: AI เสนอ assert parse_amount("65") is not None ซึ่งผ่านแม้ฟังก์ชันจะคืน 65 แทน 6500 ผมเปลี่ยนเป็นการเทียบค่าที่คาดหวังตรง ๆ และเพิ่มกรณี "๑๐๐" เพราะ int() ของ Python รับเลขไทยได้ ถ้าไม่มี Test นี้ ระบบจะรับข้อมูลที่ Spec ไม่ได้ตั้งใจให้รับ

แนวคำตอบ: ผมจะวาง 4 ชั้น คือกติกาที่ AI อ่านได้ การตรวจอัตโนมัติ การรีวิวโดยคน และนโยบายข้อมูล

  • Quality Gate: สคริปต์เดียวที่รัน ruff, ruff format, pytest และ pip-audit ใช้ทั้งใน pre-commit Hook และใน CI ของทุก Pull Request พร้อมตั้ง Branch Protection ให้ต้องผ่านก่อน Merge
  • การรีวิว: คนรีวิวทุก Pull Request และระบุในคำอธิบายว่าส่วนไหนใช้ AI พร้อมบันทึกการตัดสินใจแบบ AI_LOG.md
  • นโยบายข้อมูล: ใช้เฉพาะเครื่องมือที่บริษัทอนุมัติ ห้ามส่ง Secret ข้อมูลลูกค้า หรือโค้ดของบริษัทให้เครื่องมือที่ไม่ได้รับอนุญาต ตาม PDPA และสัญญากับลูกค้า และลบข้อมูลอ่อนไหวก่อนส่ง Log ให้ AI
