    ฟังก์ชัน parseQty ต้องห่อ (wrap) error เดิมไว้ เพื่อให้ผู้เรียกใช้ errors.Is(err, ErrNegative) แล้วได้ true จงเติม verb ของ fmt.Errorf ที่ถูกต้องในทั้งสองช่อง (ไฟล์ practice/fill/f04/main.go)

    package main

import (
	"errors"
	"fmt"
	"strconv"
)

// ErrNegative คือ sentinel error สำหรับค่าติดลบ
var ErrNegative = errors.New("value must not be negative")

func parseQty(s string) (int, error) {
	n, err := strconv.Atoi(s)
	if err != nil {
		// ห่อ error เดิมไว้ เพื่อให้ errors.Is / errors.As มองทะลุเข้าไปได้
		return 0, fmt.Errorf("parse %q: _____", s, err)
	}
	if n < 0 {
		return 0, fmt.Errorf("parse %q: _____", s, ErrNegative)
	}
	return n, nil
}

func main() {
	for _, in := range []string{"12", "-5", "x"} {
		n, err := parseQty(in)
		if err != nil {
			fmt.Println("error:", err, "| negative?", errors.Is(err, ErrNegative))
			continue
		}
		fmt.Println("ok:", n)
	}
}

    ServeMux ของ Go 1.22 ขึ้นไปรองรับ Pattern แบบมี Method และ Wildcard จงเติม Method ของ *http.Request ที่ดึงค่าของ {name} ออกมา และเติม Status Code ที่ Server ตอบกลับเมื่อส่ง POST ไปยัง Path ที่ลงทะเบียนไว้แค่ GET (ไฟล์ practice/fill/f09/main.go)

    package main

import (
	"fmt"
	"io"
	"net/http"
	"net/http/httptest"
)

func main() {
	mux := http.NewServeMux()
	mux.HandleFunc("GET /users/{name}", func(w http.ResponseWriter, r *http.Request) {
		// ดึงค่า wildcard {name} จาก path
		name := r._____("name")
		fmt.Fprintf(w, "hello %s\n", name)
	})
	srv := httptest.NewServer(mux)
	defer srv.Close()

	resp, err := http.Get(srv.URL + "/users/somchai")
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	defer resp.Body.Close()
	body, _ := io.ReadAll(resp.Body)
	fmt.Print(resp.StatusCode, " ", string(body))

	resp2, err := http.Post(srv.URL+"/users/somchai", "text/plain", nil)
	if err != nil {
		fmt.Println("error:", err)
		return
	}
	resp2.Body.Close()
	fmt.Println(resp2.StatusCode, resp2.Header.Get("Allow"))
}

package main

import (
	"errors"

// Entry คือข้อมูลของ request หนึ่งบรรทัดใน log
type Entry struct {
	Method    string
	Path      string
	Status    int
	LatencyMS int
}

// ErrBadFormat ใช้บอกว่าบรรทัดมีจำนวนช่องไม่ครบ
var ErrBadFormat = errors.New("bad format")

// parseLine แปลงหนึ่งบรรทัดเป็น Entry
// รูปแบบ: <time> <method> <path> <status> <latency_ms>
func parseLine(line string) (Entry, error) {
	f := strings.Fields(line)
	if len(f) != 5 {
		return Entry{}, fmt.Errorf("want 5 fields, got %d: %w", len(f), ErrBadFormat)
	}
	status, err := strconv.Atoi(f[3])
	if err != nil {
		return Entry{}, fmt.Errorf("status %q: %w", f[3], err)
	}
	lat, err := strconv.Atoi(f[4])
	if err != nil {
		return Entry{}, fmt.Errorf("latency %q: %w", f[4], err)
	}
	return Entry{Method: f[1], Path: f[2], Status: status, LatencyMS: lat}, nil
}

// statusClass แปลง status code เป็นกลุ่ม เช่น 404 -> "4xx"
func statusClass(code int) string {
	switch {
	case code >= 500:
		return "5xx"
	case code >= 400:
		return "4xx"
	case code >= 300:
		return "3xx"
	case code >= 200:
		return "2xx"
	default:
		return "other"
	}
}

func main() {
	path := "practice/ch1/access.log"
	if len(os.Args) > 1 {
		path = os.Args[1]
	}
	data, err := os.ReadFile(path)
	if err != nil {
		fmt.Fprintln(os.Stderr, "error:", err)
		os.Exit(2) // ไฟล์เปิดไม่ได้ = ปัญหาการใช้งาน
	}

	var entries []Entry
	badLines := 0
	for i, line := range strings.Split(string(data), "\n") {
		line = strings.TrimSpace(line)
		if line == "" || strings.HasPrefix(line, "#") {
			continue // ข้ามบรรทัดว่างและ comment
		}
		e, err := parseLine(line)
		if err != nil {
			badLines++
			fmt.Fprintf(os.Stderr, "skip line %d: %v\n", i+1, err)
			continue
		}
		entries = append(entries, e)
	}
	if len(entries) == 0 {
		fmt.Fprintln(os.Stderr, "error: no valid entries")
		os.Exit(2)
	}

	byClass := make(map[string]int)
	byPath := make(map[string]int)
	total := 0
	slowest := entries[0]
	for _, e := range entries {
		byClass[statusClass(e.Status)]++
		byPath[e.Path]++
		total += e.LatencyMS
		if e.LatencyMS > slowest.LatencyMS {
			slowest = e
		}
	}

	fmt.Printf("requests: %d (skipped %d bad lines)\n", len(entries), badLines)
	fmt.Println("by status class:")
	for _, k := range slices.Sorted(maps.Keys(byClass)) {
		fmt.Printf("  %s: %d\n", k, byClass[k])
	}
	fmt.Println("by path:")
	for _, k := range slices.Sorted(maps.Keys(byPath)) {
		fmt.Printf("  %-14s %d\n", k, byPath[k])
	}
	avg := float64(total) / float64(len(entries))
	fmt.Printf("avg latency: %.2f ms\n", avg)
	fmt.Printf("slowest: %s %s %d (%d ms)\n", slowest.Method, slowest.Path, slowest.Status, slowest.LatencyMS)

	errRate := float64(byClass["5xx"]) / float64(len(entries)) * 100
	fmt.Printf("5xx rate: %.1f%%\n", errRate)
	if errRate > 10 {
		fmt.Println("ALERT: 5xx rate is above 10%")
		os.Exit(1) // พบปัญหา
	}
}

แนวคิด
  • แยกงาน parse ออกเป็นฟังก์ชัน parseLine รับ string คืน (Entry, error) ตามแบบฉบับของ Go ทำให้ทดสอบง่าย และ main ไม่ต้องรู้รายละเอียดว่าแต่ละช่องคืออะไร
  • Error ที่บอกบริบท ห่อ error ด้วย %w พร้อมบอกว่าช่องไหนผิด เช่น latency "abc": ... แล้ว main เติมเลขบรรทัดอีกชั้น คนอ่าน log จึงหาบรรทัดเสียเจอทันที และถ้าต้องการแยกประเภทภายหลังก็ใช้ errors.Is(err, ErrBadFormat) ได้
  • ไม่ล่มเพราะข้อมูลเสียบางบรรทัด ข้ามบรรทัดที่ผิดแล้วนับไว้ เป็นพฤติกรรมที่ต้องการในงานวิเคราะห์ log จริง ข้อความ skip ออกทาง stderr ทำให้ stdout ยังเป็นรายงานสะอาดที่ redirect ลงไฟล์ได้
  • ผลลัพธ์ต้องเหมือนเดิมทุกครั้ง ลำดับการวน Map สุ่ม จึงเรียง key ด้วย slices.Sorted(maps.Keys(m)) ก่อนพิมพ์ ทำให้นำผลไปเทียบ (diff) หรือเขียนเทสต์ได้
  • ระวังการหารจำนวนเต็ม แปลงเป็น float64 ก่อนหาร Go ไม่แปลง Type ให้อัตโนมัติ การหาร int ด้วย int จะตัดทศนิยมทิ้ง
  • Exit Code ตามมาตรฐานของหลักสูตร 0 คือปกติ, 1 คือพบปัญหา (5xx เกินเกณฑ์), 2 คือใช้งานผิดหรือเปิดไฟล์ไม่ได้ ทำให้นำไปใช้ใน Script หรือ Cron Job ที่แจ้งเตือนเมื่อ Exit Code ไม่ใช่ 0 ได้ทันที

// FieldError บอกว่า field ไหนของ service ไหนผิด และผิดเพราะอะไร
type FieldError struct {
	Service string
	Field   string
	Reason  string
}

func (e *FieldError) Error() string {
	return fmt.Sprintf("%s.%s: %s", e.Service, e.Field, e.Reason)
}

// ErrDuplicateName ใช้เมื่อชื่อ service ซ้ำกัน
var ErrDuplicateName = errors.New("duplicate service name")

// Validate ตรวจ service หนึ่งตัว และคืน error ทุกข้อรวมกันด้วย errors.Join
func (s Service) Validate() error {
	name := s.Name
	if name == "" {
		name = "(no-name)"
	}
	var errs []error
	if strings.TrimSpace(s.Name) == "" {
		errs = append(errs, &FieldError{name, "name", "must not be empty"})
	}
	if s.Port < 1 || s.Port > 65535 {
		errs = append(errs, &FieldError{name, "port", fmt.Sprintf("%d is out of range 1-65535", s.Port)})
	}
	if s.Replicas < 1 || s.Replicas > 10 {
		errs = append(errs, &FieldError{name, "replicas", fmt.Sprintf("%d is out of range 1-10", s.Replicas)})
	}
	if !strings.HasPrefix(s.HealthPath, "/") {
		errs = append(errs, &FieldError{name, "health_path", fmt.Sprintf("%q must start with /", s.HealthPath)})
	}
	return errors.Join(errs...) // คืน nil ถ้าไม่มี error เลย
}

// Load อ่าน JSON array ของ Service จาก r
func Load(r io.Reader) ([]Service, error) {
	var list []Service
	dec := json.NewDecoder(r)
	dec.DisallowUnknownFields() // พิมพ์ชื่อ field ผิดจะถูกจับได้ทันที
	if err := dec.Decode(&list); err != nil {
		return nil, fmt.Errorf("decode config: %w", err)
	}
	return list, nil
}

// Result คือผลตรวจของ service หนึ่งตัว
type Result struct {
	Name     string   `json:"name"`
	OK       bool     `json:"ok"`
	Problems []string `json:"problems,omitempty"`
}

// ValidateAll ตรวจทุก service รวมถึงชื่อซ้ำ และคืนผลตามลำดับเดิมในไฟล์
func ValidateAll(list []Service) []Result {
	seen := make(map[string]bool)
	results := make([]Result, 0, len(list))
	for _, s := range list {
		r := Result{Name: s.Name, OK: true}
		var errs []error
		if err := s.Validate(); err != nil {
			errs = append(errs, err)
		}
		if s.Name != "" && seen[s.Name] {
			errs = append(errs, fmt.Errorf("%s.name: %w", s.Name, ErrDuplicateName))
		}
		seen[s.Name] = true
		if err := errors.Join(errs...); err != nil {
			r.OK = false
			r.Problems = strings.Split(err.Error(), "\n")
		}
		results = append(results, r)
	}
	return results
}

// Supplier คือสายการบินหนึ่งเจ้าที่เราจะขอราคา
type Supplier struct {
	Name    string
	BaseURL string
}

// Quote คือราคาที่ได้กลับมาจาก Supplier
type Quote struct {
	Route    string `json:"route"`
	PriceTHB int    `json:"price_thb"`
}

// Result คือผลของ Supplier หนึ่งเจ้า
type Result struct {
	Supplier string
	Quote    *Quote
	Attempts int
	Err      error
}

// StatusError ใช้เมื่อ Supplier ตอบกลับด้วย status ที่ไม่ใช่ 200
type StatusError struct{ Code int }

func (e *StatusError) Error() string { return fmt.Sprintf("status %d", e.Code) }

// retryable บอกว่า error นี้ควรลองใหม่หรือไม่: ลองใหม่เฉพาะ 5xx และ network error
func retryable(err error) bool {
	var se *StatusError
	if errors.As(err, &se) {
		return se.Code >= 500
	}
	return !errors.Is(err, context.DeadlineExceeded) && !errors.Is(err, context.Canceled)
}

// getOnce ยิง request หนึ่งครั้ง
func getOnce(ctx context.Context, c *http.Client, url string) (*Quote, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return nil, err
	}
	resp, err := c.Do(req)
	if err != nil {
		return nil, err
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		io.Copy(io.Discard, resp.Body) // อ่านทิ้งให้หมด เพื่อให้นำ connection กลับมาใช้ใหม่ได้
		return nil, &StatusError{Code: resp.StatusCode}
	}
	var q Quote
	if err := json.NewDecoder(resp.Body).Decode(&q); err != nil {
		return nil, fmt.Errorf("decode quote: %w", err)
	}
	return &q, nil
}

// fetchQuote ขอราคาจาก Supplier พร้อม Retry แบบ Backoff
func fetchQuote(ctx context.Context, c *http.Client, s Supplier, route string, retries int, backoff time.Duration) Result {
	res := Result{Supplier: s.Name}
	url := s.BaseURL + "/quotes/" + route
	for attempt := 1; attempt <= retries+1; attempt++ {
		res.Attempts = attempt
		q, err := getOnce(ctx, c, url)
		if err == nil {
			res.Quote, res.Err = q, nil
			return res
		}
		res.Err = err
		if !retryable(err) || attempt == retries+1 {
			break
		}
		// รอก่อนลองใหม่ แต่ถ้า context หมดเวลาให้เลิกทันที
		select {
		case <-time.After(backoff * time.Duration(attempt)):
		case <-ctx.Done():
			res.Err = ctx.Err()
			return res
		}
	}
	return res
}

// aggregate ขอราคาจากทุก Supplier พร้อมกัน ผลลัพธ์เรียงตามลำดับ Supplier เดิม
func aggregate(ctx context.Context, c *http.Client, sups []Supplier, route string, retries int) []Result {
	results := make([]Result, len(sups))
	var wg sync.WaitGroup
	for i, s := range sups {
		wg.Go(func() {
			// แต่ละ Goroutine เขียนลงช่อง index ของตัวเอง จึงไม่เกิด Data Race
			results[i] = fetchQuote(ctx, c, s, route, retries, 20*time.Millisecond)
		})
	}
	wg.Wait()
	return results
}

// cheapest คืนผลที่ราคาถูกที่สุด หรือ false ถ้าไม่มีใครตอบสำเร็จ
func cheapest(results []Result) (Result, bool) {
	var best Result
	found := false
	for _, r := range results {
		if r.Err != nil {
			continue
		}
		if !found || r.Quote.PriceTHB < best.Quote.PriceTHB {
			best, found = r, true
		}
	}
	return best, found
}

// describe แปลง error ให้อ่านง่ายและไม่มีเลข port ที่สุ่มทุกครั้ง
func describe(err error) string {
	var se *StatusError
	switch {
	case err == nil:
		return "-"
	case errors.Is(err, context.DeadlineExceeded):
		return "timeout"
	case errors.As(err, &se):
		return se.Error()
	default:
		return err.Error()
	}
}

// quoteHandler สร้าง Handler ของ Supplier ปลอม
func quoteHandler(price int, delay time.Duration, failFirst int32, alwaysStatus int) http.Handler {
	var calls atomic.Int32
	mux := http.NewServeMux()
	mux.HandleFunc("GET /quotes/{route}", func(w http.ResponseWriter, r *http.Request) {
		n := calls.Add(1)
		select {
		case <-time.After(delay):
		case <-r.Context().Done():
			return // client ยกเลิกแล้ว ไม่ต้องตอบ
		}
		if alwaysStatus != 0 {
			http.Error(w, "unavailable", alwaysStatus)
			return
		}
		if n <= failFirst {
			http.Error(w, "temporary error", http.StatusInternalServerError)
			return
		}
		w.Header().Set("Content-Type", "application/json")
		json.NewEncoder(w).Encode(Quote{Route: r.PathValue("route"), PriceTHB: price})
	})
	return mux
}

func main() {
	timeout := flag.Duration("timeout", 300*time.Millisecond, "เวลารวมสูงสุดในการรอราคา")
	retries := flag.Int("retries", 2, "จำนวนครั้งที่ลองใหม่เพิ่ม (0-5)")
	route := flag.String("route", "BKK-CNX", "เส้นทางบิน")
	flag.Parse()
	if *retries < 0 || *retries > 5 {
		fmt.Fprintln(os.Stderr, "-retries must be between 0 and 5")
		os.Exit(2)
	}

	// Supplier ปลอม 4 เจ้า ทำงานบนเครื่องเราเอง ไม่ต้องใช้อินเทอร์เน็ต
	servers := []struct {
		name string
		h    http.Handler
	}{
		{"fastair", quoteHandler(2890, 10*time.Millisecond, 0, 0)},
		{"budgetfly", quoteHandler(2450, 10*time.Millisecond, 1, 0)},
		{"slowjet", quoteHandler(1990, 2*time.Second, 0, 0)},
		{"brokenair", quoteHandler(0, 5*time.Millisecond, 0, http.StatusServiceUnavailable)},
	}
	var sups []Supplier
	for _, s := range servers {
		srv := httptest.NewServer(s.h)
		defer srv.Close()
		sups = append(sups, Supplier{Name: s.name, BaseURL: srv.URL})
	}

	ctx, cancel := context.WithTimeout(context.Background(), *timeout)
	defer cancel()
	client := &http.Client{Timeout: 2 * time.Second} // กันไว้อีกชั้น เผื่อลืมส่ง ctx

	start := time.Now()
	results := aggregate(ctx, client, sups, *route, *retries)
	elapsed := time.Since(start)

	tw := tabwriter.NewWriter(os.Stdout, 0, 0, 2, ' ', 0)
	fmt.Fprintln(tw, "SUPPLIER\tPRICE_THB\tATTEMPTS\tERROR")
	for _, r := range results {
		price := "-"
		if r.Quote != nil {
			price = fmt.Sprint(r.Quote.PriceTHB)
		}
		fmt.Fprintf(tw, "%s\t%s\t%d\t%s\n", r.Supplier, price, r.Attempts, describe(r.Err))
	}
	tw.Flush()
	fmt.Fprintf(os.Stderr, "elapsed: %v\n", elapsed.Round(10*time.Millisecond))

	best, ok := cheapest(results)
	if !ok {
		fmt.Println("no quotes available")
		os.Exit(1)
	}
	fmt.Printf("best for %s: %s %d THB\n", *route, best.Supplier, best.Quote.PriceTHB)
}

func TestFetchQuote(t *testing.T) {
	tests := []struct {
		name         string
		handler      http.Handler
		wantPrice    int
		wantAttempts int
		wantErr      error // ใช้กับ errors.Is (nil = ไม่ตรวจ)
		wantStatus   int   // ใช้กับ errors.As (0 = ไม่ตรวจ)
	}{
		{"ok first try", quoteHandler(2890, 0, 0, 0), 2890, 1, nil, 0},
		{"500 then 200", quoteHandler(2450, 0, 1, 0), 2450, 2, nil, 0},
		{"always 503", quoteHandler(0, 0, 0, 503), 0, 3, nil, 503},
		{"404 no retry", quoteHandler(0, 0, 0, 404), 0, 1, nil, 404},
		{"too slow", quoteHandler(1990, time.Second, 0, 0), 0, 1, context.DeadlineExceeded, 0},
	}
	for _, tc := range tests {
		t.Run(tc.name, func(t *testing.T) {
			srv := httptest.NewServer(tc.handler)
			defer srv.Close()
			ctx, cancel := context.WithTimeout(context.Background(), 100*time.Millisecond)
			defer cancel()

			r := fetchQuote(ctx, srv.Client(), Supplier{"test", srv.URL}, "BKK-CNX", 2, time.Millisecond)

			if r.Attempts != tc.wantAttempts {
				t.Errorf("attempts = %d, want %d", r.Attempts, tc.wantAttempts)
			}
			if tc.wantErr != nil && !errors.Is(r.Err, tc.wantErr) {
				t.Errorf("err = %v, want %v", r.Err, tc.wantErr)
			}
			if tc.wantStatus != 0 {
				var se *StatusError
				if !errors.As(r.Err, &se) || se.Code != tc.wantStatus {
					t.Errorf("err = %v, want status %d", r.Err, tc.wantStatus)
				}
			}
			if tc.wantPrice != 0 {
				if r.Err != nil || r.Quote.PriceTHB != tc.wantPrice {
					t.Fatalf("got quote %+v err %v, want price %d", r.Quote, r.Err, tc.wantPrice)
				}
			}
		})
	}
}

แนวคิด
  • Fan-out ด้วย Goroutine ต่อ Supplier จำนวน Supplier มีน้อยและรู้ล่วงหน้า จึงเปิด Goroutine หนึ่งตัวต่อหนึ่งเจ้าด้วย wg.Go เวลารวมจึงเท่ากับเจ้าที่ช้าที่สุด (ถูกตัดที่ 300 ms) ไม่ใช่ผลรวมของทุกเจ้า ถ้าจำนวนเป้าหมายมีมากหรือไม่จำกัด ควรเปลี่ยนเป็น Worker Pool ที่จำกัดจำนวน Worker เหมือนใน Mini Project ของ Part 4
  • เก็บผลตาม index แทนการใช้ Channel แต่ละ Goroutine เขียนลง results[i] ของตัวเองเท่านั้น ไม่มีสองตัวไหนเขียนช่องเดียวกัน จึงไม่ต้องใช้ Mutex และ wg.Wait() รับประกันว่าทุกการเขียนเสร็จก่อน main อ่าน ผลยังเรียงตามลำดับเดิมเสมอ ซึ่งเทสต์ TestAggregateOrderAndCheapest พิสูจน์โดยให้เจ้าแรกช้าที่สุด
  • context ตัวเดียวคุมเวลารวมของทุกงาน ทุก Request สร้างด้วย http.NewRequestWithContext(ctx, ...) เมื่อครบ 300 ms Request ที่ค้างอยู่ทั้งหมดถูกยกเลิกพร้อมกัน และการรอ Backoff ก็ select กับ ctx.Done() จึงไม่มีงานไหนเลยเวลา http.Client{Timeout} เป็นเพียงตาข่ายกันพลาดอีกชั้น
  • Retry อย่างมีเหตุผล ลองใหม่เฉพาะ 5xx และ Network Error ซึ่งมักหายได้เอง ไม่ลองใหม่กับ 4xx เพราะคำขอผิดตั้งแต่ต้น ลองกี่ครั้งก็ผิด และไม่ลองใหม่เมื่อ context หมดเวลา Backoff เพิ่มขึ้นตามรอบ (20 ms, 40 ms) เพื่อไม่ถล่ม Server ที่กำลังมีปัญหา
  • Error แบบมี Type *StatusError ให้ retryable และ describe ใช้ errors.As อ่าน Status Code ได้ตรง ๆ และ errors.Is(err, context.DeadlineExceeded) ทำงานได้แม้ error ถูก net/http ห่อไว้หลายชั้น ผลลัพธ์จึงอ่านง่ายและไม่มีเลข port ที่สุ่มทุกครั้ง
  • ทดสอบโดยไม่ต้องใช้อินเทอร์เน็ต httptest.NewServer เปิด Server จริงบน 127.0.0.1 ที่ port สุ่ม เทสต์จึงรันได้ทุกที่ รวมถึงใน CI และ Handler ปลอมเคารพ r.Context().Done() ทำให้ srv.Close() ไม่ต้องรอ Handler ที่ถูกยกเลิกไปแล้ว
ไอเดียต่อยอด
  1. เปลี่ยนจาก sync.WaitGroup เป็น errgroup.Group จาก golang.org/x/sync v0.24.0 แล้วใช้ g.SetLimit(n) จำกัดจำนวน Request พร้อมกัน เมื่อต้องรองรับ Supplier หลายสิบเจ้า
  2. ห่อ aggregate เป็น HTTP API GET /search/{route} ด้วย ServeMux ของ Go 1.22+ ที่ตอบ JSON ของทุกเจ้าและราคาที่ถูกที่สุด ใช้ r.Context() เป็น context ตั้งต้น เพื่อให้การค้นหาถูกยกเลิกทันทีเมื่อผู้ใช้ปิดหน้าเว็บ
  3. เพิ่ม Jitter (สุ่มเวลาบวกลบเล็กน้อย) ให้ Backoff และอ่าน Header Retry-After เมื่อ Supplier ตอบ 503 หรือ 429 เพื่อไม่ให้ Client ทุกตัวลองใหม่พร้อมกันจนเกิด Thundering Herd
  4. เพิ่ม Cache ในหน่วยความจำ (map[string]Quote ป้องกันด้วย sync.RWMutex พร้อมเวลาหมดอายุ) เพื่อตอบเส้นทางที่ถูกค้นบ่อยโดยไม่ต้องเรียก Supplier ทุกครั้ง แล้วเขียนเทสต์ด้วย -race ยืนยันว่าไม่มี Data Race
  5. เพิ่ม Circuit Breaker แบบง่ายต่อ Supplier เช่น ถ้าล้มเหลว 5 ครั้งติดกันให้ข้ามเจ้านั้น 30 วินาที และบันทึก log ทุกครั้งที่เปิดหรือปิดวงจรด้วย log/slog แบบ JSONHandler
