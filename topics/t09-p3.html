  • A ผิดเพราะ in กับ list ไล่ค้นทีละตัวเสมอ ไม่ว่า list จะเรียงหรือไม่ การเรียงจะช่วยก็ต่อเมื่อเปลี่ยนไปใช้ Binary search (เช่น โมดูล bisect) ซึ่งได้ O(log n) แต่ยังช้ากว่า set และเพิ่มต้นทุนการเรียง O(n log n)
  • B ผิดเพราะ tuple ก็ค้นแบบไล่ทีละตัว O(n) เหมือน list การเปลี่ยนค่าไม่ได้ (Immutable) ไม่ได้ทำให้ค้นเร็วขึ้น
  • C ผิดเพราะ for loop ที่เขียนเองก็คือการไล่ค้น O(n) แบบเดียวกับ in แต่ช้ากว่า เพราะรันเป็นโค้ด Python ทีละบรรทัด ส่วน in รันในภาษา C ภายใน Interpreter

"""สรุป Access Log ของเว็บเป็นรายงานบนจอและไฟล์ JSON"""

import argparse
import json
import logging
import sys
from collections import Counter
from datetime import datetime
from pathlib import Path
from typing import Any

logger = logging.getLogger(__name__)


def parse_line(line: str) -> dict[str, str | int] | None:
    """แยก 1 บรรทัดเป็น dict ถ้ารูปแบบผิดให้คืน None"""
    parts = line.split()
    if len(parts) != 6:
        return None
    timestamp, ip, method, path, status_text, size_text = parts
    if not status_text.isdigit() or not size_text.isdigit():
        return None
    try:
        datetime.fromisoformat(timestamp)
    except ValueError:
        return None
    return {
        "time": timestamp,
        "ip": ip,
        "method": method,
        "path": path,
        "status": int(status_text),
        "bytes": int(size_text),
    }


def read_records(log_path: Path) -> tuple[list[dict[str, str | int]], int]:
    """อ่านไฟล์ log ทั้งไฟล์ คืน (รายการที่อ่านได้, จำนวนบรรทัดที่เสีย)"""
    records: list[dict[str, str | int]] = []
    bad_lines = 0
    with log_path.open(encoding="utf-8") as f:
        for line_no, line in enumerate(f, start=1):
            line = line.strip()
            if not line:
                continue
            record = parse_line(line)
            if record is None:
                bad_lines += 1
                logger.warning("ข้ามบรรทัด %d: รูปแบบไม่ถูกต้อง -> %r", line_no, line)
                continue
            records.append(record)
    return records, bad_lines


def summarize(records: list[dict[str, str | int]], top: int) -> dict[str, Any]:
    """สร้างรายงานสรุปจากรายการที่อ่านได้"""
    if not records:
        raise ValueError("ไม่มีรายการที่อ่านได้เลย")
    status_counts = Counter(r["status"] for r in records)
    ip_counts = Counter(r["ip"] for r in records)
    errors = sum(count for status, count in status_counts.items() if status >= 400)
    failed_logins = Counter(
        r["ip"] for r in records if r["path"] == "/login" and r["status"] == 401
    )
    times = sorted(datetime.fromisoformat(str(r["time"])) for r in records)
    return {
        "total_requests": len(records),
        "first": times[0].isoformat(),
        "last": times[-1].isoformat(),
        "duration_minutes": round((times[-1] - times[0]).total_seconds() / 60, 1),
        "total_bytes": sum(int(r["bytes"]) for r in records),
        "status_counts": {str(k): v for k, v in sorted(status_counts.items())},
        "error_rate": round(errors / len(records), 4),
        "top_ips": [
            {"ip": ip, "count": count} for ip, count in ip_counts.most_common(top)
        ],
        "failed_logins": dict(failed_logins),
    }


def print_report(report: dict[str, Any]) -> None:
    """พิมพ์รายงานให้คนอ่าน"""
    print("=== รายงาน Access Log ===")
    print(f"ช่วงเวลา: {report['first']} ถึง {report['last']}")
    print(f"ระยะเวลา: {report['duration_minutes']} นาที")
    print(f"จำนวน Request: {report['total_requests']}")
    print(f"ข้อมูลที่ส่งออก: {report['total_bytes']:,} bytes")
    print(f"อัตรา Error (4xx/5xx): {report['error_rate']:.1%}")
    print("Status code:")
    for status, count in report["status_counts"].items():
        print(f"  {status}: {count}")
    print("IP ที่เรียกบ่อยที่สุด:")
    for rank, item in enumerate(report["top_ips"], start=1):
        print(f"  {rank}. {item['ip']:<15} {item['count']} ครั้ง")
    for ip, count in report["failed_logins"].items():
        if count >= 3:
            print(f"เตือน: {ip} login ผิด {count} ครั้ง ควรตรวจสอบ")


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description="สรุป Access Log เป็นรายงาน")
    parser.add_argument("logfile", type=Path, help="path ของไฟล์ log")
    parser.add_argument("--top", type=int, default=3, help="จำนวน IP ที่แสดง (ค่าเริ่มต้น 3)")
    parser.add_argument("--output", type=Path, help="path ของไฟล์ JSON ที่จะเขียน")
    return parser


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
    logging.basicConfig(level=logging.WARNING, format="%(levelname)s: %(message)s")
    try:
        records, bad_lines = read_records(args.logfile)
        report = summarize(records, args.top)
    except FileNotFoundError:
        print(f"ไม่พบไฟล์: {args.logfile}", file=sys.stderr)
        return 1
    except ValueError as e:
        print(f"สรุปไม่ได้: {e}", file=sys.stderr)
        return 1
    report["bad_lines"] = bad_lines
    print_report(report)
    if args.output:
        args.output.parent.mkdir(parents=True, exist_ok=True)
        with args.output.open("w", encoding="utf-8") as f:
            json.dump(report, f, ensure_ascii=False, indent=2)
            f.write("\n")
        print(f"บันทึกรายงานที่ {args.output}")
    return 0


if __name__ == "__main__":
    sys.exit(main())

"""ระบบสต็อกสินค้าขนาดเล็ก: Dataclass, Generator, Decorator และ Context Manager"""

import functools
import json
from collections.abc import Callable, Iterator
from contextlib import contextmanager
from dataclasses import asdict, dataclass, replace
from decimal import Decimal
from pathlib import Path
from typing import Any


class StockError(Exception):
    """ข้อผิดพลาดทั้งหมดของระบบสต็อก"""


class OutOfStockError(StockError):
    """ของในสต็อกไม่พอขาย"""


@dataclass
class Product:
    sku: str
    name: str
    price: Decimal
    qty: int = 0

    def __post_init__(self) -> None:
        # ตรวจข้อมูลทันทีที่สร้าง Object
        if not self.sku.startswith("SKU-"):
            raise ValueError(f"sku ต้องขึ้นต้นด้วย SKU-: {self.sku!r}")
        if self.price <= 0:
            raise ValueError("ราคาต้องมากกว่า 0")
        if self.price.as_tuple().exponent < -2:
            raise ValueError("ราคามีทศนิยมได้ไม่เกิน 2 ตำแหน่ง")
        if self.qty < 0:
            raise ValueError("จำนวนต้องไม่ติดลบ")


def audit(method: Callable[..., Any]) -> Callable[..., Any]:
    """Decorator: บันทึกการเรียก Method ที่ทำสำเร็จลงใน self.history"""

    @functools.wraps(method)
    def wrapper(self: "Inventory", *args: Any, **kwargs: Any) -> Any:
        result = method(self, *args, **kwargs)
        # ถ้า Argument เป็น Product ให้บันทึกแค่ sku เพื่อให้ประวัติอ่านง่าย
        detail = " ".join(a.sku if isinstance(a, Product) else str(a) for a in args)
        self.history.append(f"{method.__name__} {detail}")
        return result

    return wrapper


class Inventory:
    def __init__(self) -> None:
        self._items: dict[str, Product] = {}
        self.history: list[str] = []

    def __len__(self) -> int:
        return len(self._items)

    def get(self, sku: str) -> Product:
        try:
            return self._items[sku]
        except KeyError:
            raise StockError(f"ไม่พบสินค้า {sku}") from None

    @audit
    def add_product(self, product: Product) -> None:
        if product.sku in self._items:
            raise StockError(f"มีสินค้า {product.sku} อยู่แล้ว")
        self._items[product.sku] = product

    @audit
    def receive(self, sku: str, qty: int) -> int:
        if qty <= 0:
            raise ValueError("จำนวนที่รับเข้าต้องมากกว่า 0")
        product = self.get(sku)
        product.qty += qty
        return product.qty

    @audit
    def sell(self, sku: str, qty: int) -> Decimal:
        product = self.get(sku)
        if qty <= 0:
            raise ValueError("จำนวนที่ขายต้องมากกว่า 0")
        if qty > product.qty:
            raise OutOfStockError(f"{sku} เหลือ {product.qty} ชิ้น แต่สั่ง {qty} ชิ้น")
        product.qty -= qty
        return product.price * qty

    @property
    def total_value(self) -> Decimal:
        return sum((p.price * p.qty for p in self._items.values()), Decimal("0"))

    def low_stock(self, threshold: int) -> Iterator[Product]:
        """Generator: คืนสินค้าที่เหลือน้อยกว่า threshold ทีละชิ้น เรียงจากน้อยไปมาก"""
        for product in sorted(self._items.values(), key=lambda p: p.qty):
            if product.qty < threshold:
                yield product

    @contextmanager
    def transaction(self) -> Iterator["Inventory"]:
        """ทำหลายคำสั่งรวดเดียว ถ้ามี Exception ให้ย้อนสต็อกกลับทั้งหมด"""
        snapshot = {sku: replace(p) for sku, p in self._items.items()}
        history_len = len(self.history)
        try:
            yield self
        except Exception:
            self._items = snapshot
            del self.history[history_len:]
            raise

    def save(self, path: Path) -> None:
        rows = [asdict(p) | {"price": str(p.price)} for p in self._items.values()]
        path.parent.mkdir(parents=True, exist_ok=True)
        with path.open("w", encoding="utf-8") as f:
            json.dump(rows, f, ensure_ascii=False, indent=2)

    @classmethod
    def load(cls, path: Path) -> "Inventory":
        inventory = cls()
        with path.open(encoding="utf-8") as f:
            for row in json.load(f):
                product = Product(
                    row["sku"], row["name"], Decimal(row["price"]), row["qty"]
                )
                inventory._items[product.sku] = product
        return inventory


def main() -> None:
    inv = Inventory()
    inv.add_product(Product("SKU-001", "ปากกาเจล", Decimal("15.00"), 40))
    inv.add_product(Product("SKU-002", "สมุดปกแข็ง", Decimal("89.50"), 5))
    inv.add_product(Product("SKU-003", "แฟ้มเอกสาร", Decimal("25.00"), 2))
    print(f"มูลค่าสต็อกเริ่มต้น: {inv.total_value:,.2f} บาท")
    print(f"ขาย SKU-001 x 10 ได้เงิน {inv.sell('SKU-001', 10):,.2f} บาท")

    # ออร์เดอร์ที่มีหลายรายการ ถ้ารายการไหนพัง ต้องย้อนทั้งออร์เดอร์
    try:
        with inv.transaction():
            inv.sell("SKU-002", 2)
            inv.sell("SKU-003", 5)
    except OutOfStockError as e:
        print(f"ยกเลิกออร์เดอร์: {e}")
    print(f"SKU-002 หลังยกเลิก ยังเหลือ {inv.get('SKU-002').qty} ชิ้น")

    print("สินค้าใกล้หมด (น้อยกว่า 6 ชิ้น):")
    for product in inv.low_stock(6):
        print(f"  {product.sku} เหลือ {product.qty} ชิ้น  {product.name}")
    print(f"มูลค่าสต็อกปัจจุบัน: {inv.total_value:,.2f} บาท")
    print("ประวัติ:", inv.history)
    print("ชื่อ Method หลังใส่ Decorator:", Inventory.sell.__name__)

    path = Path("data/stock.json")
    inv.save(path)
    again = Inventory.load(path)
    print(f"โหลดกลับจาก {path}: {len(again)} รายการ มูลค่า {again.total_value:,.2f}")


if __name__ == "__main__":
    main()

  • เขียน Test ให้ตรวจ "สถานะหลังเกิด Error" ด้วย ไม่ใช่แค่ตรวจว่ามี Exception เช่น test_sell_more_than_stock_raises ตรวจต่อว่าสต็อกยังเป็น 5
