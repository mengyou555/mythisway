from pathlib import Path

import numpy as np
import pandas as pd

rng = np.random.default_rng(42)
n = 5000
region = rng.choice(["กรุงเทพฯ", "ภาคเหนือ", "ภาคอีสาน", "ภาคใต้"], n, p=[0.45, 0.18, 0.22, 0.15])
amount = rng.lognormal(6.5, 1.0, n).round(2)                 # บาท
hour = rng.integers(0, 24, n)
new_device = (rng.random(n) < 0.12).astype(int)
tx_24h = rng.poisson(3, n)
account_age_days = rng.integers(1, 1500, n)

         + 0.6 * np.log(amount / 600) - 0.0015 * account_age_days)

                   "account_age_days": account_age_days, "fraud": fraud})
Path("data").mkdir(exist_ok=True)
df.to_csv("data/transactions.csv", index=False)
print(df.shape, f"fraud rate = {df['fraud'].mean():.2%}")
print(df.groupby("region")["fraud"].agg(["count", "sum"]).to_string())

# โจทย์ที่ 3: ตรวจจับธุรกรรมผิดปกติ เลือก Threshold ตามต้นทุน และตรวจผลแยกตามภูมิภาค
import numpy as np
import pandas as pd
from sklearn.compose import ColumnTransformer
from sklearn.dummy import DummyClassifier
from sklearn.ensemble import HistGradientBoostingClassifier
from sklearn.inspection import permutation_importance
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import confusion_matrix, precision_score, recall_score
from sklearn.model_selection import StratifiedKFold, cross_val_predict, cross_val_score, train_test_split
from sklearn.pipeline import Pipeline
from sklearn.preprocessing import FunctionTransformer, OneHotEncoder, StandardScaler

COST_FN, COST_FP = 500, 20          # พลาดธุรกรรมโกง = เสียหาย 500 บาท, ตรวจซ้ำผิดตัว = ค่าแรง 20 บาท

df = pd.read_csv("data/transactions.csv")
df["night"] = (df["hour"] < 5).astype(int)
num = ["amount", "tx_24h", "account_age_days", "new_device", "night"]
X, y = df[num + ["region"]], df["fraud"]
X_train, X_test, y_train, y_test = train_test_split(X, y, test_size=0.2, stratify=y, random_state=42)
print(f"fraud ในข้อมูลฝึก {y_train.sum()} / {len(y_train)} | ข้อมูลทดสอบ {y_test.sum()} / {len(y_test)}")

pre = ColumnTransformer([
    ("log_amount", FunctionTransformer(np.log1p, feature_names_out="one-to-one"), ["amount"]),
    ("num", "passthrough", num[1:]),
    ("cat", OneHotEncoder(handle_unknown="ignore"), ["region"]),
])
models = {
    "Dummy": DummyClassifier(strategy="most_frequent"),
    "Logistic balanced": Pipeline([("pre", pre), ("scale", StandardScaler()),
                                   ("model", LogisticRegression(max_iter=1000, class_weight="balanced"))]),
    "HistGB": Pipeline([("pre", pre), ("model", HistGradientBoostingClassifier(random_state=42))]),
}
cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
print("\nmodel              | CV average precision | CV ROC-AUC")
for name, m in models.items():
    ap = cross_val_score(m, X_train, y_train, cv=cv, scoring="average_precision").mean()
    auc = cross_val_score(m, X_train, y_train, cv=cv, scoring="roc_auc").mean()
    print(f"{name:<18} | {ap:20.3f} | {auc:.3f}")

# เลือก Threshold จากผลทำนายแบบ Cross-validation บนข้อมูลฝึก (ไม่แตะข้อมูลทดสอบ)
best = models["Logistic balanced"]
oof = cross_val_predict(best, X_train, y_train, cv=cv, method="predict_proba")[:, 1]
thresholds = np.round(np.arange(0.05, 0.96, 0.05), 2)
costs = []
for t in thresholds:
    p = oof >= t
    costs.append(COST_FN * ((~p) & (y_train == 1)).sum() + COST_FP * (p & (y_train == 0)).sum())
t_best = thresholds[int(np.argmin(costs))]
print("\nต้นทุนบนข้อมูลฝึก (CV) ตาม Threshold:",
      {float(t): f"{c:,}" for t, c in zip(thresholds, costs) if t in (0.2, 0.3, 0.4, 0.45, 0.5, 0.6, 0.7)})
print(f"Threshold ที่ต้นทุนต่ำสุดบนข้อมูลฝึก: {t_best} (ต้นทุน {min(costs):,} บาท)")

# วัดผลบนข้อมูลทดสอบครั้งเดียว เทียบ Threshold 0.5 กับ Threshold ที่เลือก
best.fit(X_train, y_train)
proba = best.predict_proba(X_test)[:, 1]
for t in [0.5, t_best]:
    p = (proba >= t).astype(int)
    tn, fp, fn, tp = confusion_matrix(y_test, p).ravel()
    print(f"\nThreshold {t}: TP={tp} FP={fp} FN={fn} TN={tn}")
    print(f"  precision {precision_score(y_test, p):.3f} | recall {recall_score(y_test, p):.3f}"
          f" | ต้นทุน {COST_FN * fn + COST_FP * fp:,} บาท")

# ความเป็นธรรม: Recall แยกตามภูมิภาค ที่ Threshold ที่เลือก
res = X_test.assign(fraud=y_test, flag=(proba >= t_best).astype(int))
by_region = res[res["fraud"] == 1].groupby("region")["flag"].agg(fraud_cases="count", recall="mean")
print("\nRecall แยกตามภูมิภาค (ข้อมูลทดสอบ):")
print(by_region.round(3).to_string())

perm = permutation_importance(best, X_test, y_test, scoring="average_precision", n_repeats=10, random_state=42)
imp = pd.Series(perm.importances_mean, index=X_test.columns).sort_values(ascending=False)
print("\nPermutation importance (average precision ที่ลดลง):")
print(imp.round(3).to_string())
