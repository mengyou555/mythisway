Mini Project: Concurrent Website Health Checker (gohealth)

เป้าหมายของโปรเจกต์

สร้าง CLI ชื่อ gohealth ที่โฟลเดอร์ ~/projects/gohealth เป็น Module แยกชื่อ github.com/somchai/gohealth (เปลี่ยน somchai เป็นชื่อบัญชี GitHub ของคุณได้) โปรแกรมรับรายการ URL จากไฟล์หรือจาก Argument แล้วส่ง GET ไปยังทุก URL พร้อมกันผ่าน Worker Pool ที่จำกัดจำนวน Worker ไว้ จากนั้นรายงานผลเป็นตาราง JSON หรือ CSV และจบด้วย Exit Code ที่ CI อ่านเข้าใจ

โปรเจกต์นี้ใช้ Standard Library ล้วน ไม่มี Dependency ภายนอกแม้แต่ตัวเดียว ไฟล์ go.sum จึงไม่ถูกสร้างขึ้น เหตุผลคือ Go มีทุกอย่างที่ต้องใช้อยู่ในตัวแล้ว ตั้งแต่ net/http, context, sync ไปจนถึง encoding/json และ log/slog การไม่มี Dependency ทำให้ Build ได้ทุกที่ ตรวจความปลอดภัยง่าย และผู้สัมภาษณ์เห็นชัดว่าคุณเข้าใจภาษาเอง ไม่ได้พึ่ง Library

หัวใจของโปรเจกต์มี 5 ข้อ

  1. Concurrency ที่ควบคุมได้: ไม่ได้เปิด Goroutine หนึ่งตัวต่อหนึ่ง URL แบบไม่จำกัด แต่ใช้ Worker Pool ขนาด -workers เพื่อไม่ให้ยิง Request ใส่ปลายทางมากเกินไป
  2. ทุก Request มี Timeout: ใช้ context.WithTimeout ต่อหนึ่ง Request และผูกทั้งหมดไว้กับ context หลักที่สร้างจาก signal.NotifyContext กด Ctrl+C เมื่อไรทุกอย่างหยุดทันที
  3. Retry อย่างมีเหตุผล: ลองซ้ำเฉพาะ Network Error และ 5xx ซึ่งเป็นปัญหาชั่วคราว ไม่ลองซ้ำ 4xx เพราะลองกี่ครั้งก็ได้ผลเดิม และรอนานขึ้นทีละรอบ (Backoff)
  4. stdout สะอาด: ผลลัพธ์ออก stdout ส่วน Log ออก stderr จึงส่งต่อด้วย | ไปให้ jq หรือโปรแกรมอื่นได้ทันที
  5. ทดสอบได้โดยไม่ต้องมีอินเทอร์เน็ต: Test ทุกตัวใช้ httptest.NewServer จำลอง Server ที่ตอบ 200, 404, 500, 503 หรือตอบช้า และผ่าน go test -race
Security

ใช้ gohealth ตรวจเฉพาะเว็บที่คุณเป็นเจ้าของ หรือได้รับอนุญาตให้ทดสอบเท่านั้น การยิง Request จำนวนมากใส่ระบบของคนอื่นโดยไม่ได้รับอนุญาตอาจเข้าข่ายความผิดตาม พ.ร.บ.ว่าด้วยการกระทำความผิดเกี่ยวกับคอมพิวเตอร์ และอาจทำให้ระบบนั้นช้าหรือล่มได้ ตั้ง -workers และ -retries ให้ต่ำเสมอเมื่อตรวจระบบจริง ในบทเรียนนี้เราทดสอบกับ Server ในเครื่องตัวเอง (127.0.0.1) และใช้ https://go.dev กับ https://example.com เป็นเพียงตัวอย่างในไฟล์เท่านั้น

ภาพรวมการทำงาน (Architecture)

ดูแผนภาพนี้ก่อนเขียนโค้ด จะเข้าใจว่าข้อมูลไหลผ่านส่วนไหนบ้าง

  urls.txt / argument
          │
          ▼
  main.go ── flag.Parse ── ตรวจค่า flag ── loadURLs (checker.ParseTargets)
          │                                   │ ผิด → exit 2
          ▼
  signal.NotifyContext (Ctrl+C = ยกเลิกทุกอย่าง)
          │
          ▼
  checker.Run ── jobs channel ──┬─► worker 0 ─► Check ─► once (ctx + timeout) ─► HTTP GET
                                ├─► worker 1 ─► Check ─► Retry + Backoff เมื่อ 5xx / network error
                                └─► worker N ─► Check ─► results[index] = Result
          │ wg.Wait()
          ▼
  []Result (เรียงตาม input) ── report.Write ─► stdout (table | json | csv)
          │
          └─ report.Summarize ── down > 0 ? exit 1 : exit 0

package checker

import (
	"bufio"
	"context"
	"errors"
	"fmt"
	"io"
	"log/slog"
	"net"
	"net/http"
	"net/url"
	"strings"
	"sync"
	"syscall"
	"time"

// ErrInvalidURL ใช้กับ URL ที่รูปแบบไม่ถูกต้อง (ตรวจด้วย errors.Is ได้)
var ErrInvalidURL = errors.New("invalid url")

// Result คือผลการตรวจ URL หนึ่งรายการ (Struct Tag กำหนดชื่อ field ใน JSON)
type Result struct {
	URL        string `json:"url"`
	Up         bool   `json:"up"`
	StatusCode int    `json:"status_code"`
	LatencyMS  int64  `json:"latency_ms"`
	Attempts   int    `json:"attempts"`
	Error      string `json:"error,omitempty"`
}

// Config เก็บค่าที่ใช้ควบคุมการตรวจ
type Config struct {
	Workers   int           // จำนวน Worker ที่ทำงานพร้อมกัน
	Timeout   time.Duration // Timeout ต่อหนึ่ง Request
	Retries   int           // จำนวนครั้งที่ลองซ้ำเพิ่ม (ไม่นับครั้งแรก)
	ExpectMax int           // Status Code ที่ไม่เกินค่านี้ถือว่า UP
	Backoff   time.Duration // เวลารอพื้นฐานก่อน Retry (คูณด้วยรอบที่)
	UserAgent string
	Logger    *slog.Logger
}

// Checker ตรวจ URL ตาม Config โดยใช้ http.Client ตัวเดียวร่วมกันทุก Worker
type Checker struct {
	cfg    Config
	client *http.Client
}

// New สร้าง Checker และเติมค่าเริ่มต้นให้ field ที่ยังเป็น Zero Value
func New(cfg Config) *Checker {
	if cfg.Workers < 1 {
		cfg.Workers = 1
	}
	if cfg.Timeout <= 0 {
		cfg.Timeout = 3 * time.Second
	}
	if cfg.ExpectMax == 0 {
		cfg.ExpectMax = 399
	}
	if cfg.Backoff <= 0 {
		cfg.Backoff = 200 * time.Millisecond
	}
	if cfg.UserAgent == "" {
		cfg.UserAgent = "gohealth/1.0"
	}
	if cfg.Logger == nil {
		// ถ้าไม่ส่ง Logger มา ให้ทิ้ง Log ทั้งหมด (สะดวกตอนเขียน Test)
		cfg.Logger = slog.New(slog.DiscardHandler)
	}
	return &Checker{
		cfg: cfg,
		// Timeout ของ Client ยาวกว่า context 1 วินาที เป็นตาข่ายกันพลาดชั้นที่สอง
		client: &http.Client{Timeout: cfg.Timeout + time.Second},
	}
}

// Run ตรวจทุก URL ด้วย Worker Pool แล้วคืนผลเรียงตามลำดับเดิมของ input
func (c *Checker) Run(ctx context.Context, urls []string) []Result {
	type job struct {
		index int
		url   string
	}
	jobs := make(chan job)
	results := make([]Result, len(urls))

	workers := min(c.cfg.Workers, len(urls))
	var wg sync.WaitGroup
	for w := range workers {
		// wg.Go (Go 1.25+) เรียก Add(1) และ Done() ให้อัตโนมัติ
		wg.Go(func() {
			for j := range jobs {
				c.cfg.Logger.Debug("checking", "worker", w, "url", j.url)
				// แต่ละงานเขียนลงช่อง index ของตัวเอง จึงไม่เกิด Data Race
				results[j.index] = c.Check(ctx, j.url)
			}
		})
	}

	for i, u := range urls {
		jobs <- job{index: i, url: u}
	}
	close(jobs) // ผู้ส่งเป็นคนปิด Channel เพื่อบอก Worker ว่าหมดงานแล้ว
	wg.Wait()
	return results
}

// Check ตรวจ URL เดียว พร้อม Retry เมื่อเจอ Network Error หรือ 5xx
func (c *Checker) Check(ctx context.Context, rawURL string) Result {
	res := Result{URL: rawURL}
	for attempt := 1; ; attempt++ {
		res.Attempts = attempt
		code, latency, err := c.once(ctx, rawURL)
		res.StatusCode = code
		res.LatencyMS = latency.Milliseconds()

		if !retryable(code, err) || attempt > c.cfg.Retries || ctx.Err() != nil {
			c.finish(&res, err)
			return res
		}

		wait := c.cfg.Backoff * time.Duration(attempt)
		c.cfg.Logger.Debug("retrying", "url", rawURL, "attempt", attempt, "wait", wait)
		select {
		case <-ctx.Done():
			// ผู้ใช้กด Ctrl+C หรือ context หลักหมดเวลา: หยุดทันที ไม่รอ Backoff
			c.finish(&res, ctx.Err())
			return res
		case <-time.After(wait):
		}
	}
}

// once ส่ง GET หนึ่งครั้งภายใต้ Timeout ของตัวเอง
func (c *Checker) once(ctx context.Context, rawURL string) (int, time.Duration, error) {
	ctx, cancel := context.WithTimeout(ctx, c.cfg.Timeout)
	defer cancel()

	req, err := http.NewRequestWithContext(ctx, http.MethodGet, rawURL, nil)
	if err != nil {
		return 0, 0, err
	}
	req.Header.Set("User-Agent", c.cfg.UserAgent)

	start := time.Now()
	resp, err := c.client.Do(req)
	if err != nil {
		return 0, time.Since(start), err
	}
	defer resp.Body.Close()
	// อ่าน Body ทิ้ง (ไม่เกิน 1 MB) เพื่อให้ Connection กลับไปใช้ซ้ำได้
	_, _ = io.Copy(io.Discard, io.LimitReader(resp.Body, 1<<20))
	return resp.StatusCode, time.Since(start), nil
}

// finish สรุปว่า UP หรือ DOWN และเขียนข้อความ Error ที่อ่านง่าย
func (c *Checker) finish(res *Result, err error) {
	switch {
	case err != nil:
		res.Up = false
		res.StatusCode = 0
		res.Error = describe(err, c.cfg.Timeout)
	case res.StatusCode <= c.cfg.ExpectMax:
		res.Up = true
		res.Error = ""
	default:
		res.Up = false
		res.Error = fmt.Sprintf("HTTP %d %s", res.StatusCode, http.StatusText(res.StatusCode))
	}
}

// retryable ลองซ้ำเฉพาะ Network Error และ 5xx เท่านั้น (4xx ลองซ้ำก็ไม่หาย)
func retryable(code int, err error) bool {
	return err != nil || code >= 500
}

// describe แปลง error ยาว ๆ จาก net/http ให้เหลือข้อความสั้นที่คนอ่านเข้าใจ
func describe(err error, timeout time.Duration) string {
	var dnsErr *net.DNSError
	var netErr net.Error
	switch {
	case errors.Is(err, context.DeadlineExceeded):
		return fmt.Sprintf("timeout after %s", timeout)
	case errors.Is(err, context.Canceled):
		return "canceled"
	case errors.As(err, &dnsErr):
		return "dns lookup failed: " + dnsErr.Err
	case errors.Is(err, syscall.ECONNREFUSED):
		return "connection refused"
	case errors.As(err, &netErr) && netErr.Timeout():
		return fmt.Sprintf("timeout after %s", timeout)
	default:
		return err.Error()
	}
}

// ValidateURL ยอมรับเฉพาะ http/https ที่มี host
func ValidateURL(s string) error {
	u, err := url.Parse(s)
	if err != nil {
		return fmt.Errorf("%w %q: %v", ErrInvalidURL, s, err)
	}
	if u.Scheme != "http" && u.Scheme != "https" {
		return fmt.Errorf("%w %q: scheme must be http or https", ErrInvalidURL, s)
	}
	if u.Host == "" {
		return fmt.Errorf("%w %q: missing host", ErrInvalidURL, s)
	}
	return nil
}

// ParseTargets อ่าน URL ทีละบรรทัด ข้ามบรรทัดว่างและบรรทัดที่ขึ้นต้นด้วย #
func ParseTargets(r io.Reader) ([]string, error) {
	var urls []string
	sc := bufio.NewScanner(r)
	line := 0
	for sc.Scan() {
		line++
		s := strings.TrimSpace(sc.Text())
		if s == "" || strings.HasPrefix(s, "#") {
			continue
		}
		if err := ValidateURL(s); err != nil {
			return nil, fmt.Errorf("line %d: %w", line, err)
		}
		urls = append(urls, s)
	}
	if err := sc.Err(); err != nil {
		return nil, err
	}
	return urls, nil
}

		}
	}

		fs.PrintDefaults()
		fmt.Fprintf(stderr, "\nExit codes: 0 all up, 1 at least one down, 2 usage error\n")
	}

	if err := fs.Parse(args); err != nil {
		if errors.Is(err, flag.ErrHelp) {
			return exitOK // -h หรือ -help ไม่ถือว่าผิด
		}
		return exitUsage
	}
	if *showVersion {
		fmt.Fprintf(stdout, "gohealth %s\n", version)
		return exitOK
	}

	// Log ไปที่ stderr เสมอ เพื่อให้ stdout มีแต่ผลลัพธ์ (ส่งต่อด้วย | หรือ > ได้สะอาด)
	level := slog.LevelInfo
	if *verbose {
		level = slog.LevelDebug
	}
	logger := slog.New(slog.NewTextHandler(stderr, &slog.HandlerOptions{Level: level}))

	usageErr := func(msg string, a ...any) int {
		fmt.Fprintf(stderr, "gohealth: "+msg+"\n", a...)
		return exitUsage
	}
	switch {
	case *workers < 1 || *workers > 100:
		return usageErr("-workers must be between 1 and 100, got %d", *workers)
	case *timeout <= 0:
		return usageErr("-timeout must be positive, got %s", *timeout)
	case *retries < 0 || *retries > 10:
		return usageErr("-retries must be between 0 and 10, got %d", *retries)
	case !report.ValidFormat(*format):
		return usageErr("unknown -format %q (use %s)", *format, strings.Join(report.Formats, ", "))
	}

	urls, err := loadURLs(*file, fs.Args())
	if err != nil {
		return usageErr("%v", err)
	}
	if len(urls) == 0 {
		fs.Usage()
		return usageErr("no URLs given: use -file or pass URLs as arguments")
	}

	// Ctrl+C จะยกเลิก ctx และทุก Request ที่แตกออกมาจาก ctx นี้
	ctx, stop := signal.NotifyContext(context.Background(), os.Interrupt)
	defer stop()

	c := checker.New(checker.Config{
		Workers:   *workers,
		Timeout:   *timeout,
		Retries:   *retries,
		ExpectMax: *expectMax,
		Logger:    logger,
	})

	logger.Info("starting", "urls", len(urls), "workers", *workers, "timeout", *timeout, "retries", *retries)
	start := time.Now()
	results := c.Run(ctx, urls)
	sum := report.Summarize(results)
	logger.Info("finished", "up", sum.Up, "down", sum.Down, "elapsed", time.Since(start).Round(time.Millisecond))

	if err := report.Write(stdout, *format, results); err != nil {
		fmt.Fprintf(stderr, "gohealth: write output: %v\n", err)
		return exitUsage
	}
	if ctx.Err() != nil {
		logger.Warn("interrupted by user")
	}
	if sum.Down > 0 {
		return exitDown
	}
	return exitOK
}

// loadURLs รวม URL จากไฟล์ (ถ้ามี) กับ URL ที่พิมพ์ต่อท้ายคำสั่ง
func loadURLs(path string, args []string) ([]string, error) {
	var urls []string
	if path != "" {
		f, err := os.Open(path)
		if err != nil {
			return nil, err
		}
		defer f.Close()
		fromFile, err := checker.ParseTargets(f)
		if err != nil {
			return nil, fmt.Errorf("%s: %w", path, err)
		}
		urls = append(urls, fromFile...)
	}
	for _, a := range args {
		if err := checker.ValidateURL(a); err != nil {
			return nil, fmt.Errorf("argument: %w", err)
		}
		urls = append(urls, a)
	}
	return urls, nil
}

    • 200 OK: ต้อง UP และใช้ 1 ครั้ง
    • 404: ต้อง DOWN และไม่ Retry เพราะ 4xx เป็นความผิดของฝั่ง Client ลองใหม่ก็ได้ผลเดิม

    • Handler ตอบช้า: ต้องได้ timeout after 50ms ไม่ใช่รอ 2 วินาที
    • พอร์ตที่ปิดอยู่: ต้องได้ connection refused

func TestCheck(t *testing.T) {
	tests := []struct {
		name         string
		handler      func() http.HandlerFunc // สร้าง Handler ใหม่ทุกเคส เพื่อให้ตัวนับเริ่มที่ 0
		retries      int
		timeout      time.Duration
		wantUp       bool
		wantCode     int
		wantAttempts int
		wantErr      string // ข้อความที่ต้องมีใน Result.Error ("" = ต้องไม่มี error)
	}{
		{
			name: "200 ok",
			handler: func() http.HandlerFunc {
				return func(w http.ResponseWriter, r *http.Request) { w.WriteHeader(http.StatusOK) }
			},
			retries: 2, timeout: time.Second,
			wantUp: true, wantCode: 200, wantAttempts: 1,
		},
		{
			name: "404 is down and not retried",
			handler: func() http.HandlerFunc {
				return func(w http.ResponseWriter, r *http.Request) { http.NotFound(w, r) }
			},
			retries: 2, timeout: time.Second,
			wantUp: false, wantCode: 404, wantAttempts: 1, wantErr: "HTTP 404 Not Found",
		},
		{
			name: "500 then 200 succeeds on retry",
			handler: func() http.HandlerFunc {
				var calls atomic.Int64
				return func(w http.ResponseWriter, r *http.Request) {
					if calls.Add(1) == 1 {
						w.WriteHeader(http.StatusInternalServerError)
						return
					}
					w.WriteHeader(http.StatusOK)
				}
			},
			retries: 2, timeout: time.Second,
			wantUp: true, wantCode: 200, wantAttempts: 2,
		},
		{
			name: "always 503 exhausts retries",
			handler: func() http.HandlerFunc {
				return func(w http.ResponseWriter, r *http.Request) { w.WriteHeader(http.StatusServiceUnavailable) }
			},
			retries: 2, timeout: time.Second,
			wantUp: false, wantCode: 503, wantAttempts: 3, wantErr: "HTTP 503 Service Unavailable",
		},
		{
			name: "slow handler hits timeout",
			handler: func() http.HandlerFunc {
				return func(w http.ResponseWriter, r *http.Request) {
					// รอ 2 วินาที หรือจนกว่า Client จะยกเลิก Request
					select {
					case <-time.After(2 * time.Second):
					case <-r.Context().Done():
					}
				}
			},
			retries: 0, timeout: 50 * time.Millisecond,
			wantUp: false, wantCode: 0, wantAttempts: 1, wantErr: "timeout after 50ms",
		},
	}

	for _, tt := range tests {
		t.Run(tt.name, func(t *testing.T) {
			srv := httptest.NewServer(tt.handler())
			defer srv.Close()

			got := newTestChecker(tt.retries, tt.timeout).Check(context.Background(), srv.URL)

			if got.Up != tt.wantUp {
				t.Errorf("Up = %v, want %v", got.Up, tt.wantUp)
			}
			if got.StatusCode != tt.wantCode {
				t.Errorf("StatusCode = %d, want %d", got.StatusCode, tt.wantCode)
			}
			if got.Attempts != tt.wantAttempts {
				t.Errorf("Attempts = %d, want %d", got.Attempts, tt.wantAttempts)
			}
			if tt.wantErr == "" && got.Error != "" {
				t.Errorf("Error = %q, want empty", got.Error)
			}
			if !strings.Contains(got.Error, tt.wantErr) {
				t.Errorf("Error = %q, want it to contain %q", got.Error, tt.wantErr)
			}
		})
	}
}

func TestWriteTable(t *testing.T) {

	if err := Write(&buf, "table", sample); err != nil {
		t.Fatal(err)
	}

	if got := buf.String(); got != want {
		t.Errorf("table output mismatch\n got:\n%s\nwant:\n%s", got, want)
	}
}

func TestWriteJSON(t *testing.T) {
	var buf bytes.Buffer
	if err := Write(&buf, "json", sample); err != nil {
		t.Fatal(err)
	}
	// แปลงกลับเป็น Struct เพื่อยืนยันว่า JSON ถูกต้องครบทุก field
	var got []checker.Result
	if err := json.Unmarshal(buf.Bytes(), &got); err != nil {
		t.Fatalf("invalid JSON: %v\n%s", err, buf.String())
	}
	if len(got) != len(sample) || got[2] != sample[2] {
		t.Errorf("round trip mismatch: %+v", got)

	if strings.Contains(first, `"error"`) {
		t.Errorf("first item should not contain error key:\n%s", first)
	}
}

func TestWriteCSV(t *testing.T) {
	var buf bytes.Buffer
	if err := Write(&buf, "csv", sample); err != nil {
		t.Fatal(err)
	}

	if err != nil {
		t.Fatal(err)
	}

		t.Fatalf("got %d rows, want 4 (header + 3)", len(rows))

			t.Errorf("rows[%d][%d] = %q, want %q", tt.row, tt.col, got, tt.want)
		}
	}
}

    ระวัง
    • อย่าเขียน Test ที่ยิงไป https://go.dev หรือเว็บจริง Test จะพังเมื่อเน็ตหลุดหรือเว็บปลายทางช้า และ CI จะ "แดง" แบบสุ่มโดยที่โค้ดไม่ได้ผิด
    • ถ้าลืม defer srv.Close() Server จะค้างอยู่จนจบการรัน Test ทั้งชุด และกินพอร์ตไปเรื่อย ๆ
    • ถ้าตั้งชื่อไฟล์ผิดเป็น checker_tests.go คำสั่ง go test จะไม่เห็น Test เลย และขึ้นว่า [no test files]
    Best Practice
    • Test ที่ดีต้องเร็ว (ชุดนี้ใช้ไม่ถึง 0.2 วินาที) ไม่ขึ้นกับอินเทอร์เน็ต และให้ผลเหมือนเดิมทุกครั้ง ทั้งสามข้อทำให้ทีมกล้ารัน Test บ่อย ๆ
    • เพิ่ม Test ทุกครั้งที่เจอ Bug: เขียน Test ที่จับ Bug นั้นให้พังก่อน แล้วค่อยแก้โค้ดให้ผ่าน Bug เดิมจะไม่กลับมาอีก
    • ตั้งเป้า Coverage ที่สมเหตุสมผล เช่น 70–90% สำหรับ Logic หลัก ไม่จำเป็นต้องไล่ให้ถึง 100% เพราะบางบรรทัด (เช่น Error ที่แทบไม่เกิด) ทดสอบยากแต่ได้ประโยชน์น้อย

    • ตั้งชื่อเวอร์ชันตาม Semantic Versioning (v1.0.0, v1.1.0, v2.0.0) และสร้าง Git Tag ชื่อเดียวกัน เพื่อให้ go install github.com/somchai/gohealth@v1.0.0 ทำงานได้
    • อย่า commit ไฟล์ Binary ขึ้น Git ให้แนบไว้ในหน้า GitHub Releases แทน โปรเจกต์นี้กันไว้แล้วด้วย bin/ และ *.exe ใน .gitignore

  8. Step 8: ตั้ง CI ด้วย GitHub Actions

    CI (Continuous Integration) คือการให้ GitHub รันด่านตรวจคุณภาพจาก Step 7 ให้อัตโนมัติทุกครั้งที่ push หรือเปิด Pull Request ถ้าใครส่งโค้ดที่ไม่ผ่าน gofmt, go vet หรือ Test จะเห็นเครื่องหมายกากบาทสีแดงทันที ก่อนโค้ดนั้นจะถูก Merge

    # CI ของ gohealth: รันทุกครั้งที่ push หรือเปิด Pull Request
name: ci

on:
  push:
    branches: [main]
  pull_request:

jobs:
  test:
    runs-on: ubuntu-latest
    steps:
      - name: Checkout code
        uses: actions/checkout@v5

      - name: Set up Go
        uses: actions/setup-go@v6
        with:
          go-version: '1.27.2'

      - name: Check formatting (gofmt)
        run: |
          unformatted=$(gofmt -l .)
          if [ -n "$unformatted" ]; then
            echo "These files need gofmt:"
            echo "$unformatted"
            exit 1
          fi

      - name: Vet
        run: go vet ./...

      - name: Test with race detector
        run: go test -race -cover ./...

      - name: Build
        run: go build -ldflags "-s -w -X main.version=${{ github.sha }}" -o bin/gohealth .
    ส่วนความหมาย
    on: push / pull_requestรันเมื่อ push เข้า Branch main และเมื่อมี Pull Request ใด ๆ
    runs-on: ubuntu-latestใช้เครื่อง Ubuntu ของ GitHub ซึ่งมี gcc อยู่แล้ว จึงรัน -race ได้ทันที

    # gohealth

![ci](https://github.com/somchai/gohealth/actions/workflows/ci.yml/badge.svg)

Concurrent website health checker written in Go (standard library only).
ตรวจสถานะเว็บไซต์หลายเว็บพร้อมกันด้วย Worker Pool มี Timeout ต่อ Request, Retry พร้อม Backoff
และ Exit Code ที่ใช้ใน CI ได้ทันที

## Features

- ตรวจหลาย URL พร้อมกันด้วย Worker Pool (`-workers`)
- Timeout ต่อ Request ด้วย `context` และกด Ctrl+C เพื่อยกเลิกทุก Request ได้
- Retry เฉพาะ Network Error และ HTTP 5xx พร้อม Backoff
- ผลลัพธ์แบบ `table`, `json`, `csv` เรียงตามลำดับ Input เสมอ
- Log ไป stderr ผลลัพธ์ไป stdout ใช้ต่อ Pipe ได้
- ไม่มี Dependency ภายนอก

## Usage

```bash
gohealth [flags] [url ...]
gohealth -file urls.txt -workers 10 -timeout 2s -format json
```

| Flag | Default | Description |
|---|---|---|
| `-file` | | อ่าน URL จากไฟล์ (บรรทัดละ 1 URL, `#` คือ comment) |
| `-workers` | 5 | จำนวน Worker (1..100) |
| `-timeout` | 3s | Timeout ต่อ Request |
| `-retries` | 2 | จำนวนครั้งที่ลองซ้ำ (0..10) |
| `-format` | table | `table`, `json`, `csv` |
| `-expect-max` | 399 | Status Code ที่ไม่เกินค่านี้ถือว่า UP |
| `-verbose` | false | แสดง Debug Log |
| `-version` | false | แสดงเวอร์ชัน |

## Exit codes

| Code | Meaning |
|---|---|
| 0 | ทุก URL ปกติ (UP) |
| 1 | มีอย่างน้อย 1 URL ล่ม (DOWN) หรือผู้ใช้กด Ctrl+C |
| 2 | ใช้คำสั่งผิด, ไม่มี URL, อ่านไฟล์ไม่ได้ หรือ URL ผิดรูปแบบ |

คำถามสัมภาษณ์งาน Go ที่เจอบ่อยในตำแหน่ง Junior ถึง Mid-level Backend Developer ลองตอบด้วยตัวเองออกเสียงก่อน แล้วค่อยเปิดดูแนวคำตอบ ถ้าอ้างถึงโปรเจกต์ gohealth ของคุณได้ คำตอบจะน่าเชื่อถือขึ้นมาก

1. Goroutine ต่างจาก OS Thread อย่างไร และทำไม Go สร้าง Goroutine ได้เป็นแสนตัว

แนวคำตอบ: Goroutine คือหน่วยการทำงานที่ Go Runtime จัดการเอง ไม่ใช่ Thread ของระบบปฏิบัติการโดยตรง ประเด็นที่ควรพูดให้ครบ:

  • Stack เล็กและโตได้: Goroutine เริ่มด้วย Stack เพียงไม่กี่ KB (ปัจจุบันประมาณ 2 KB) และขยายหรือหดได้ตามต้องการ ส่วน OS Thread มักจอง Stack ระดับ MB ตั้งแต่แรก
  • Scheduler แบบ M:N: Runtime กระจาย Goroutine จำนวนมาก (G) ลงบน OS Thread จำนวนน้อย (M) ผ่าน Processor (P) ซึ่งมีจำนวนเท่ากับ GOMAXPROCS (ค่าเริ่มต้นคือจำนวน CPU ที่ใช้ได้)
  • สลับงานถูก: การสลับ Goroutine เกิดใน User Space ไม่ต้องเข้า Kernel เมื่อ Goroutine รอ I/O เช่น รอ Network Runtime จะพักมันไว้และเอาตัวอื่นมารันแทน
  • Concurrency ไม่ใช่ Parallelism: Concurrency คือการออกแบบให้หลายงานดำเนินไปพร้อมกันได้ Parallelism คือการรันพร้อมกันจริงบนหลาย Core Go ให้ทั้งสองอย่าง

ตัวอย่างจากงานจริง: "ใน gohealth ผมใช้ Worker 5 ตัวตรวจ URL เวลาส่วนใหญ่ของแต่ละ Worker คือรอ Network ระหว่างนั้น Runtime ก็ให้ Worker ตัวอื่นทำงานแทน จึงตรวจ 5 URL ได้ในเวลาใกล้เคียงกับ URL ที่ช้าที่สุดเพียงตัวเดียว"

จุดที่ทำให้คำตอบโดดเด่น: บอกว่าแม้ Goroutine จะถูก แต่ต้องจำกัดจำนวนเมื่อไปเรียกทรัพยากรภายนอก (เช่น ใช้ Worker Pool) และต้องมีทางให้มันจบเสมอ ไม่เช่นนั้นจะเกิด Goroutine Leak

2. Unbuffered Channel กับ Buffered Channel ต่างกันอย่างไร ใครควรเป็นคนปิด Channel

แนวคำตอบ:

  • Unbuffered (make(chan T)): การส่งจะรอจนกว่ามีผู้รับมารับ ทั้งสองฝั่งต้อง "นัดเจอกัน" ณ จุดเดียว จึงใช้ซิงโครไนซ์ได้ด้วย
  • Buffered (make(chan T, n)): ส่งได้ทันทีจนกว่า Buffer จะเต็ม ช่วยลดการรอเมื่อผู้ส่งและผู้รับทำงานเร็วไม่เท่ากัน แต่ไม่ใช่วิธีแก้ Deadlock
  • ผู้ส่งเป็นคนปิด เพราะผู้ส่งเท่านั้นที่รู้ว่าไม่มีข้อมูลเพิ่มแล้ว ถ้าผู้รับปิด ผู้ส่งที่ยังทำงานอยู่จะ panic: send on closed channel ถ้ามีผู้ส่งหลายตัว ให้ใช้ sync.WaitGroup รอทุกตัวเสร็จแล้วปิดจาก Goroutine แยก
  • การอ่านจาก Channel ที่ปิดแล้วได้ Zero Value ทันที ใช้ v, ok := <-ch หรือ for v := range ch เพื่อรู้ว่าจบแล้ว

ตัวอย่าง Deadlock ที่ผู้สัมภาษณ์ชอบถาม:

package main

func main() {
	ch := make(chan int)
	ch <- 1 // ไม่มีผู้รับเลย main รอตลอดไป
}
fatal error: all goroutines are asleep - deadlock!

goroutine 1 [chan send]:
main.main()
	/home/somchai/projects/deadlock/main.go:5 +0x28
exit status 2

แก้ได้สองแบบ: ให้มี Goroutine อื่นรับ หรือใช้ make(chan int, 1) แต่ควรอธิบายว่าแบบหลังแค่เลื่อนปัญหาถ้าส่งเกินขนาด Buffer

3. อธิบาย Slice ว่าทำงานอย่างไร (len, cap, Backing Array) และ Bug ที่เกิดจาก append

แนวคำตอบ: Slice เป็น Struct เล็ก ๆ 3 ช่อง คือ Pointer ไปยัง Backing Array, ความยาว (len) และความจุ (cap) การส่ง Slice เข้าฟังก์ชันจึงคัดลอกแค่ 3 ช่องนี้ แต่ยังชี้ไปที่ Array เดียวกัน

  • เมื่อ append แล้ว cap ยังพอ ข้อมูลจะถูกเขียนลง Backing Array เดิม ซึ่งอาจทับข้อมูลของ Slice อื่นที่แชร์ Array เดียวกัน
  • เมื่อ cap ไม่พอ Go จะจอง Array ใหม่ที่ใหญ่ขึ้นแล้วคัดลอกข้อมูลไป Slice ใหม่จึงไม่เกี่ยวกับของเดิมอีก
  • เพราะไม่รู้ล่วงหน้าว่าจะเกิดแบบไหน ต้องรับค่ากลับเสมอ: s = append(s, x)
  • วิธีป้องกัน: slices.Clone(s), copy หรือ Full Slice Expression a[:2:2] เพื่อจำกัด cap
package main

import "fmt"

func main() {
	a := []int{1, 2, 3, 4}
	b := a[:2]        // b ใช้ Backing Array เดียวกับ a
	b = append(b, 99) // cap ยังพอ จึงเขียนทับ a[2]
	fmt.Println(a, b)
	c := a[:2:2]      // จำกัด cap = 2
	c = append(c, 77) // cap ไม่พอ จึงจองที่ใหม่ a ไม่เปลี่ยน
	fmt.Println(a, c)
}
[1 2 99 4] [1 2 99]
[1 2 99 4] [1 2 77]

จุดที่ทำให้คำตอบโดดเด่น: พูดถึงการจองที่ล่วงหน้าด้วย make([]T, 0, n) เมื่อรู้ขนาด เพื่อลดการจองหน่วยความจำซ้ำ และใน gohealth ใช้ make([]Result, len(urls)) ให้แต่ละ Worker เขียนคนละ Index จึงไม่มี Data Race

4. Go จัดการ Error อย่างไร ทำไมไม่มี Exception และใช้ errors.Is กับ errors.As ต่างกันอย่างไร

แนวคำตอบ: ใน Go Error เป็นค่าธรรมดาที่ Implement Interface error (มี Method Error() string) ฟังก์ชันคืน Error เป็นค่าสุดท้าย และผู้เรียกต้องตรวจทันที แนวคิดคือทำให้ทุกเส้นทางที่อาจผิดพลาดมองเห็นได้ชัดในโค้ด ไม่มีการกระโดดข้ามหลายชั้นแบบ Exception

  • เพิ่มบริบทด้วย %w: fmt.Errorf("line %d: %w", n, err) Wrap Error เดิมไว้ข้างใน ถ้าใช้ %v สายจะขาดและตรวจย้อนไม่ได้
  • errors.Is(err, target): ตรวจว่าในสายมี Error ตัวนี้ หรือไม่ ใช้กับ Sentinel Error เช่น ErrInvalidURL, context.DeadlineExceeded, os.ErrNotExist
  • errors.As(err, &target): ตรวจว่าในสายมี Error ชนิดนี้ หรือไม่ แล้วดึงออกมาใช้ field เช่น var ne net.Error แล้วเรียก ne.Timeout()
  • ห้ามเทียบข้อความ Error ด้วย err.Error() == "..." เพราะข้อความเปลี่ยนได้ทุกเมื่อ
  • panic ใช้เฉพาะความผิดพลาดของโปรแกรมเมอร์ ที่ไม่ควรเกิด เช่น Index เกินขอบเขต ไม่ใช่สำหรับ Input ผิดหรือ Network ล่ม

ตัวอย่างจากงานจริง: "ใน gohealth ฟังก์ชัน ParseTargets Wrap ErrInvalidURL พร้อมเลขบรรทัด ส่วน main.go แค่แสดงข้อความและคืน Exit Code 2 และใน Test ผมใช้ errors.Is ยืนยันว่าการ Wrap ถูกต้อง ส่วนการแปลง Network Error เป็นข้อความสั้น ๆ ใช้ errors.Is(err, syscall.ECONNREFUSED) และ errors.As กับ *net.DNSError"

5. ออกแบบโปรแกรมที่เรียก HTTP ไปยัง 10,000 URL ให้เร็ว ปลอดภัย และยกเลิกได้ จะทำอย่างไร และทดสอบอย่างไร

แนวคำตอบ (คำถามระดับ Mid ที่วัดทั้ง Concurrency และ Engineering): ตอบเป็นข้อ ๆ ตามลำดับการออกแบบ

  1. จำกัด Concurrency ด้วย Worker Pool: ไม่สร้าง Goroutine 10,000 ตัวพร้อมกัน เพราะจะเปิด Connection มากเกินไปจนเครื่องเราหรือปลายทางรับไม่ไหว ใช้ Channel ของงานกับ Worker จำนวน N ที่ปรับได้ด้วย flag
  2. รักษาลำดับผลลัพธ์: ส่ง Index ไปพร้อมงาน แล้วให้ Worker เขียนผลลง results[i] ของ Slice ที่จองไว้ล่วงหน้า ไม่ต้องใช้ Mutex เพราะแต่ละตัวเขียนคนละช่อง และ wg.Wait() รับประกันว่าอ่านหลังทุกตัวเขียนเสร็จ
  3. Timeout หลายชั้น: context.WithTimeout ต่อ Request, http.Client{Timeout} เป็นตาข่ายสำรอง และ Root Context จาก signal.NotifyContext เพื่อให้ Ctrl+C ยกเลิกทุกอย่าง
  4. Retry อย่างมีวินัย: Retry เฉพาะ Network Error และ 5xx ไม่ Retry 4xx ใช้ Backoff ที่เพิ่มขึ้น (ถ้าจะให้ดีขึ้นอีกเพิ่ม Jitter) และหยุดทันทีเมื่อ ctx.Done()
  5. ใช้ทรัพยากรถูกต้อง: ใช้ http.Client ตัวเดียวร่วมกันเพื่อ Reuse Connection, อ่าน Body ทิ้งและ Close ทุกครั้ง ไม่เช่นนั้น Connection จะรั่ว
  6. แยก Output: ผลลัพธ์ไป stdout, Log ไป stderr ด้วย log/slog และใช้ Exit Code ที่มีความหมายสำหรับ CI
  7. ทดสอบ: Table-driven Test กับ httptest.NewServer จำลอง 200, 404, 500-then-200, 503 ตลอด และ Handler ที่ช้ากว่า Timeout รวมถึง Test ที่ทำให้ URL แรกตอบช้าที่สุดเพื่อพิสูจน์ว่าลำดับไม่สลับ รันทั้งหมดด้วย go test -race ใน CI

จุดที่ทำให้คำตอบโดดเด่น: พูดถึงความรับผิดชอบด้วย เช่น จำกัด Request ต่อ Host เพื่อไม่ให้กลายเป็นการโจมตีปลายทางโดยไม่ตั้งใจ ตั้ง User-Agent ที่บอกว่าเป็นใคร และตรวจเฉพาะระบบที่ได้รับอนุญาต แล้วปิดท้ายด้วย "ผมทำโปรเจกต์นี้จริงชื่อ gohealth ดูโค้ดและ CI ได้ที่ GitHub ของผม"
