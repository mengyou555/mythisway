ความสามารถ (Features)

  • คำสั่ง subnet: รับ CIDR เช่น 192.168.1.0/24 แล้วแสดง Network, Netmask, Wildcard, Broadcast, Host แรก/สุดท้าย, จำนวน Host ที่ใช้ได้ และแบ่ง Subnet ย่อยด้วย --new-prefix รองรับทั้ง IPv4 และ IPv6 (ต่อยอดจาก subnet_calc.py ในแนวคิดที่ 3)
  • คำสั่ง dns: Forward lookup (A/AAAA) ผ่าน socket.getaddrinfo, Reverse lookup (PTR) ผ่าน socket.gethostbyaddr และถาม DNS server ที่ระบุด้วย dig ผ่าน subprocess เช่น DNS Lab ของเราบน 127.0.0.1 Port 5300
  • คำสั่ง ports: ตรวจ TCP Port ด้วย socket.create_connection (Timeout 0.5 วินาที) เฉพาะเครื่องของตัวเอง คือ 127.0.0.0/8, ::1 และชื่อ localhost เป้าหมายอื่นจะถูกปฏิเสธพร้อมข้อความภาษาไทยและ Exit code 2
  • คำสั่ง report: รวบรวม Hostname, Interface จาก ip -j -4 addr, Default route จาก ip -j route show default, DNS server จาก /etc/resolv.conf และสถานะ Port แล้วเขียนเป็นไฟล์ reports/network-report-YYYYMMDD-HHMMSS.txt
  • Unit test ด้วย unittest (ไม่ต้องติดตั้ง pytest) ครอบคลุมการคำนวณ Subnet, Allowlist และการตรวจ Port ที่เปิดด้วย Server ชั่วคราว
  • เอกสาร: docs/home-network.md มีแผนผังเครือข่ายแบบ Mermaid (GitHub แสดงเป็นภาพให้อัตโนมัติ) และแบบ ASCII, docs/ip-plan.md มีตาราง IP Plan ของวง 192.168.1.0/24

    """คำนวณและแบ่ง Subnet ทั้ง IPv4 และ IPv6 ด้วยโมดูล ipaddress (ต่อยอดจาก subnet_calc.py)"""
import ipaddress


def describe(net):
    """คืนค่า dict ข้อมูลสำคัญของ Network หนึ่งวง (รองรับ IPv4 และ IPv6)"""
    info = {
        "Network": f"{net.network_address}/{net.prefixlen}",
        "Version": f"IPv{net.version}",
    }
    if net.version == 4:
        if net.num_addresses > 2:
            # Subnet ปกติ: ตัด Network address และ Broadcast ออก
            first, last = net.network_address + 1, net.broadcast_address - 1
            usable = net.num_addresses - 2
        else:
            # /31 (ลิงก์ Point-to-Point) และ /32 (Host เดียว) ใช้ได้ทุก Address
            first, last = net.network_address, net.broadcast_address
            usable = net.num_addresses
        info.update({
            "Netmask": str(net.netmask),
            "Wildcard": str(net.hostmask),
            "Broadcast": str(net.broadcast_address),
        })
    else:
        # IPv6 ไม่มี Broadcast จึงนับทุก Address ในวง
        first, last = net.network_address, net.broadcast_address
        usable = net.num_addresses
    info.update({
        "First host": str(first),
        "Last host": str(last),
        "Usable hosts": usable,
        "Private": net.is_private,
    })
    return info


def parse_network(cidr):
    """แปลงข้อความเป็น Network; strict=False ยอมรับ IP ที่มี Host bits เช่น 172.16.45.200/21"""
    return ipaddress.ip_network(cidr, strict=False)


def split(net, new_prefix, limit=16):
    """แบ่ง Network เป็น Subnet ย่อยตาม new_prefix คืนค่า (รายการ, จำนวนทั้งหมด)"""
    if new_prefix <= net.prefixlen or new_prefix > net.max_prefixlen:
        raise ValueError(
            f"new prefix ต้องอยู่ระหว่าง {net.prefixlen + 1} ถึง {net.max_prefixlen}")
    total = 2 ** (new_prefix - net.prefixlen)
    subs = []
    for sub in net.subnets(new_prefix=new_prefix):
        if len(subs) >= limit:
            break
        subs.append(sub)
    return subs, total


def format_info(net):
    """จัดข้อมูลของ Network เป็นข้อความหลายบรรทัด"""
    return "\n".join(f"{key:<13}: {value}" for key, value in describe(net).items())


def format_split(net, new_prefix, limit=16):
    """จัดผลการแบ่ง Subnet เป็นข้อความหลายบรรทัด"""
    subs, total = split(net, new_prefix, limit)
    lines = [f"แบ่ง {net} เป็น /{new_prefix} ได้ {total} วง (แสดง {len(subs)} วงแรก):"]
    for i, sub in enumerate(subs, 1):
        info = describe(sub)
        lines.append(f"{i:>3}. {info['Network']:<22} {info['First host']}"
                     f" - {info['Last host']} ({info['Usable hosts']})")
    return "\n".join(lines)

  3. Step 3: เขียน hnt/dnslookup.py สำหรับค้นหา DNS

    Module นี้มี 2 วิธีถาม DNS ซึ่ง Network Engineer ต้องแยกให้ออก วิธีแรกคือถามผ่าน Resolver ของระบบ ด้วย socket.getaddrinfo แบบเดียวกับที่ Browser และโปรแกรมทั่วไปทำ ผลจึงขึ้นกับ /etc/hosts และ /etc/resolv.conf ของเครื่อง วิธีที่สองคือถาม DNS server ที่ระบุตรงๆ ด้วย dig @server ซึ่งข้าม Resolver ของระบบ ใช้ตรวจว่า Server ตัวนั้นตอบอะไร

    เวลาผู้ใช้บอกว่า "เข้าเว็บไม่ได้" แล้ววิธีแรกล้มเหลวแต่วิธีที่สองได้คำตอบ แปลว่าปัญหาอยู่ที่การตั้งค่า Resolver ของเครื่อง ไม่ใช่ที่ DNS server นี่คือเหตุผลที่เครื่องมือนี้มีทั้งสองแบบ เราเรียก dig ด้วย subprocess.run แบบส่ง List ของ Argument (ไม่ใช้ shell=True) เพื่อกัน Command injection และตั้ง timeout ไว้เสมอ

    """ค้นหา DNS ด้วย socket และเรียก dig (ถ้ามี) ผ่าน subprocess"""
import ipaddress
import shutil
import socket
import subprocess


def forward(name):
    """หา A/AAAA ผ่าน Resolver ของระบบ คืนค่า list ของ (ชนิด, IP) โดยไม่ซ้ำ"""
    results = []
    for family, _, _, _, sockaddr in socket.getaddrinfo(name, None):
        kind = "A" if family == socket.AF_INET else "AAAA"
        item = (kind, sockaddr[0])
        if item not in results:
            results.append(item)
    return results


def reverse(ip):
    """หาชื่อจาก IP (PTR) คืนค่า None ถ้าหาไม่เจอ"""
    try:
        return socket.gethostbyaddr(ip)[0]
    except (socket.herror, socket.gaierror):
        return None


def is_ip(text):
    """ตรวจว่าข้อความเป็น IP address หรือไม่"""
    try:
        ipaddress.ip_address(text)
        return True
    except ValueError:
        return False


def dig_query(name, server, port=53, rtype="A", timeout=5):
    """ถาม DNS server ที่ระบุด้วย dig +short คืนค่า list ของคำตอบ
    คืน None ถ้าเครื่องไม่มี dig (ติดตั้งด้วย sudo apt install bind9-dnsutils)"""
    if shutil.which("dig") is None:
        return None
    cmd = ["dig", f"@{server}", "-p", str(port), "+short",
           "+time=2", "+tries=1", name, rtype]
    proc = subprocess.run(cmd, capture_output=True, text=True, timeout=timeout)
    return [line for line in proc.stdout.splitlines() if line.strip()]

    """ตรวจ TCP Port เฉพาะเครื่องของตัวเอง (127.0.0.0/8 และ ::1) เท่านั้น"""
import ipaddress
import socket

# ชื่อเครื่องที่อนุญาตโดยไม่ต้อง Resolve DNS
ALLOWED_NAMES = {"localhost": "127.0.0.1"}


def is_allowed_target(host):
    """อนุญาตเฉพาะ Loopback: 127.0.0.0/8, ::1 และชื่อ localhost
    ไม่ Resolve ชื่ออื่น เพื่อกันการใช้ชื่อ DNS พาไปสแกนเครื่องคนอื่น"""
    if host in ALLOWED_NAMES:
        return True
    try:
        return ipaddress.ip_address(host).is_loopback
    except ValueError:
        return False


def parse_ports(text):
    """แปลง '22,80,8000-8002' เป็น list ของเลข Port ที่ไม่ซ้ำและเรียงแล้ว"""
    ports = set()
    for part in text.split(","):
        part = part.strip()
        if not part:
            continue
        if "-" in part:
            start, end = (int(x) for x in part.split("-", 1))
            if end - start > 1024:
                raise ValueError("ช่วง Port กว้างเกิน 1024 พอร์ต")
            ports.update(range(start, end + 1))
        else:
            ports.add(int(part))
    for port in ports:
        if not 1 <= port <= 65535:
            raise ValueError(f"Port {port} ไม่อยู่ในช่วง 1-65535")
    return sorted(ports)


def check_port(host, port, timeout=0.5):
    """ลองเชื่อมต่อ TCP คืนค่า 'open', 'closed' หรือ 'timeout'"""
    if not is_allowed_target(host):
        raise PermissionError(f"ปฏิเสธเป้าหมาย {host}: ตรวจได้เฉพาะเครื่องตัวเอง")
    target = ALLOWED_NAMES.get(host, host)
    try:
        with socket.create_connection((target, port), timeout=timeout):
            return "open"
    except ConnectionRefusedError:
        return "closed"
    except (socket.timeout, TimeoutError):
        return "timeout"
    except OSError:
        return "closed"


def scan(host, ports, timeout=0.5):
    """ตรวจหลาย Port คืนค่า list ของ (port, สถานะ, ชื่อบริการ)"""
    results = []
    for port in ports:
        try:
            service = socket.getservbyport(port, "tcp")
        except OSError:
            service = "-"
        results.append((port, check_port(host, port, timeout), service))
    return results

def run_ip_json(args):

def dns_servers(path="/etc/resolv.conf"):
    """อ่านบรรทัด nameserver จาก resolv.conf"""
    try:
        lines = Path(path).read_text().splitlines()
    except OSError:
        return []
    return [line.split()[1] for line in lines
            if line.startswith("nameserver") and len(line.split()) > 1]

def cmd_dns(args):

        return 0 if name else 1

            print(f"{kind:<5}{args.name} -> {ip}")
    except socket.gaierror as err:
        print(f"หาชื่อ {args.name} ไม่เจอ: {err}", file=sys.stderr)
        return 1
    return 0

    def test_open_and_closed(self):
        # เปิด Server ชั่วคราวบน Port ว่างที่ระบบเลือกให้ (Port 0)
        server = socket.socket(socket.AF_INET, socket.SOCK_STREAM)

        port = server.getsockname()[1]
        try:

        finally:
            server.close()
        # ปิด Server แล้ว Port เดิมต้องกลายเป็น closed

Interview Questions

คำถามด้านล่างเป็นคำถามที่พบบ่อยในการสัมภาษณ์ตำแหน่ง Network Engineer ระดับ Junior ถึง Mid-level, IT Support และ NOC Engineer ลองตอบด้วยตัวเองออกเสียงก่อนเปิดดูแนวคำตอบ ผู้สัมภาษณ์ไม่ได้ต้องการคำตอบที่ท่องมา แต่อยากเห็นว่าคุณคิดเป็นขั้นตอนและอธิบายด้วยเหตุผลได้

1. ผู้ใช้แจ้งว่า "เข้าเว็บไม่ได้" คุณจะตรวจอะไรก่อน และตรวจตามลำดับอย่างไร Junior

แนวคำตอบที่ดี: เริ่มจากถามข้อมูลให้ชัดก่อน (เข้าไม่ได้ทุกเว็บหรือเว็บเดียว, เป็นคนเดียวหรือหลายคน, เริ่มเมื่อไร, มีอะไรเปลี่ยนไปบ้าง) จากนั้นไล่ตรวจแบบ Bottom-up ตาม OSI ทีละชั้น ทุกขั้นที่ผ่านจะตัดสาเหตุออกไปได้ครึ่งหนึ่ง

  1. Layer 1–2: สายหรือ Wi-Fi เชื่อมต่ออยู่ไหม Interface เป็น UP หรือไม่ (ip -br link หรือดูไอคอน Wi-Fi)

  6. DNS: dig example.com หรือ nslookup ได้คำตอบไหม ถ้า ping IP ได้แต่ Resolve ชื่อไม่ได้ ปัญหาคือ DNS (อาการที่พบบ่อยที่สุด)

3. TCP กับ UDP ต่างกันอย่างไร อธิบาย 3-way handshake และยกตัวอย่างว่าเมื่อไรควรใช้ UDP Junior

แนวคำตอบที่ดี: ทั้งสองเป็นโปรโตคอล Layer 4 ที่ใช้ Port แยกโปรแกรม แต่ต่างกันที่การรับประกันการส่ง

  • TCP เป็นแบบ Connection-oriented ต้องเปิด Connection ก่อน มีเลขลำดับ (Sequence number), ส่งซ้ำเมื่อข้อมูลหาย, เรียงลำดับให้ และควบคุมการไหลของข้อมูล (Flow control และ Congestion control) จึงเชื่อถือได้แต่มี Overhead มากกว่า ใช้กับ HTTP/1.1, HTTP/2, SSH, SMTP และฐานข้อมูล
  • UDP เป็นแบบ Connectionless ส่งได้ทันที ไม่รับประกันว่าถึงหรือถึงตามลำดับ Header เล็กเพียง 8 ไบต์ ใช้กับงานที่ต้องการความเร็วหรือรับข้อมูลหายได้บ้าง เช่น DNS query, VoIP, วิดีโอสด, เกมออนไลน์, DHCP และ QUIC ที่เป็นฐานของ HTTP/3

3-way handshake: Client ส่ง SYN พร้อม Sequence number เริ่มต้น, Server ตอบ SYN-ACK (รับทราบและส่ง Sequence ของตัวเอง), Client ตอบ ACK จากนั้น Connection อยู่ในสถานะ ESTABLISHED การปิดใช้ FIN/ACK ทั้งสองฝั่ง ฝั่งที่ปิดก่อนจะค้างใน TIME-WAIT ชั่วครู่

จุดที่ทำให้คำตอบโดดเด่น: อธิบายได้ว่า Connection refused คือปลายทางตอบ RST (เครื่องอยู่ แต่ไม่มีโปรแกรมฟัง Port นั้น) ส่วน Connection timed out คือไม่มีใครตอบเลย มักเกิดจาก Firewall Drop Packet หรือ Route ไม่ถึง และบอกได้ว่าดูด้วย sudo tcpdump -i lo -nn 'tcp port 9000' ในเครื่องตัวเองจะเห็น Flag [S], [S.], [.] ตามลำดับ

4. VLAN, Trunk และ Inter-VLAN Routing คืออะไร ถ้าเครื่องใน VLAN 10 คุยกับ VLAN 20 ไม่ได้ คุณจะตรวจอะไร Mid-level

แนวคำตอบที่ดี: VLAN แบ่ง Switch ตัวเดียวออกเป็นหลาย Broadcast domain ทาง Logic ช่วยแยกแผนก ลด Broadcast และเพิ่มความปลอดภัย Access port เป็นของ VLAN เดียวและส่ง Frame แบบไม่มี Tag ส่วน Trunk ส่งหลาย VLAN บนสายเดียว โดยใส่ Tag 802.1Q ขนาด 4 ไบต์ที่มี VLAN ID 12 บิต (ใช้ได้ 1–4094) Frame ของ Native VLAN วิ่งบน Trunk แบบไม่มี Tag

เพราะแต่ละ VLAN เป็นคนละ Subnet การข้าม VLAN จึงต้องผ่าน Layer 3 มี 2 แบบหลัก: Router-on-a-stick (Sub-interface บน Router เช่น Gi0/0.10 กับ encapsulation dot1Q 10) และ SVI บน Multilayer switch (interface vlan 10 และ ip routing) ซึ่งเร็วกว่าและนิยมในองค์กร

ลำดับการตรวจเมื่อ VLAN 10 คุยกับ VLAN 20 ไม่ได้:

  1. show vlan brief: Port ของเครื่องอยู่ถูก VLAN หรือไม่
  2. show interfaces trunk: Trunk ขึ้นไหม และ VLAN 10, 20 อยู่ใน Allowed list หรือไม่ Native VLAN สองฝั่งตรงกันไหม (ถ้าไม่ตรงจะเห็น %CDP-4-NATIVE_VLAN_MISMATCH)
  3. Sub-interface หรือ SVI มี IP ถูกต้องและสถานะ up/up (show ip interface brief)
  4. เครื่องลูกข่ายตั้ง Default Gateway เป็น IP ของ Sub-interface หรือ SVI ในวงของตัวเองหรือไม่ เช่น 192.168.10.1
  5. มี ACL หรือ Firewall กั้นระหว่าง VLAN หรือไม่ เช่น ACL SALES-IN ที่ตั้งใจอนุญาตเฉพาะ HTTPS ไปยัง Server

คำตอบระดับ Mid-level จะพูดถึง STP ด้วย ว่าเมื่อมีหลาย Switch ต่อกันเป็น Loop ต้องใช้ STP/RSTP ป้องกัน Broadcast storm และ Port ที่ต่อเครื่องลูกข่ายควรเปิด PortFast พร้อม BPDU Guard

5. OSPF เลือกเส้นทางอย่างไร และต่างจาก BGP อย่างไร ควรใช้แต่ละตัวเมื่อไร Mid-level

แนวคำตอบที่ดี: OSPF เป็น Link-state IGP ใช้ภายในองค์กร (Autonomous System เดียว) Router ทุกตัวแลก LSA จนมีแผนที่เครือข่ายเหมือนกัน แล้วคำนวณเส้นทางสั้นที่สุดด้วยอัลกอริทึม SPF (Dijkstra) ตัวชี้วัดคือ Cost = Reference bandwidth / Interface bandwidth ค่า Default ของ Reference คือ 100 Mbps ทำให้ลิงก์ 100 Mbps กับ 1 Gbps มี Cost เท่ากันคือ 1 จึงควรตั้ง auto-cost reference-bandwidth ให้เท่ากันทุก Router เช่น 1000 ส่วนการจะเป็น Neighbor กันได้ ต้องมี Area, Hello/Dead timer (Default 10/40 วินาทีบนลิงก์ Ethernet), Subnet และ Authentication ตรงกัน และ Router ID ต้องไม่ซ้ำ

BGP เป็น Path-vector EGP ใช้แลกเส้นทางระหว่าง AS เช่น ระหว่างองค์กรกับ ISP หรือระหว่าง ISP ด้วยกัน ทำงานบน TCP Port 179 เลือกเส้นทางตาม Attribute และนโยบาย (Policy) เช่น AS_PATH ที่สั้นกว่า มากกว่าความเร็วของลิงก์ แบ่งเป็น eBGP (ต่าง AS) และ iBGP (AS เดียวกัน) Private ASN อยู่ในช่วง 64512–65534
