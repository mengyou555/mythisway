  • A ผิดเพราะ user:// เป็นแค่โฟลเดอร์หนึ่งบนดิสก์ของผู้เล่น ไม่ใช่ Sandbox สคริปต์ที่โหลดเข้ามาใช้ API ของ Engine ได้ครบ รวมถึง FileAccess และ OS.execute() บนเครื่อง Desktop
  • B ผิดเพราะนามสกุลไฟล์ไม่ได้บอกว่าข้างในมีอะไร ไฟล์ .tres ที่ฝังสคริปต์ก็ยังเป็นไฟล์ .tres ที่ถูกต้องตามรูปแบบทุกประการ

func _ready() -> void:
    build_grid()

    # รอให้เกมเริ่มเดิน Frame แรกก่อน ผลการลบจึงเหมือนกันทุกครั้งที่รัน

    # queue_free() ยังไม่ลบทันที จำนวนจึงยังเท่าเดิม

    await get_tree().process_frame
    print("หลังจบ Frame เหลือ %d ดวง คะแนนรวม %d" % [get_child_count(), total_points()])


func build_grid() -> void:
    for row in rows:
        for col in columns:
            var star: GridStar = STAR_SCENE.instantiate()
            star.name = "Star_%d_%d" % [row, col]
            # ตั้งตำแหน่งและคะแนนให้เสร็จก่อน add_child() เสมอ
            star.position = origin + Vector2(col * spacing.x, row * spacing.y)
            if row == rows - 1:
                star.points = bottom_row_points
            add_child(star)
            print("%s ที่ (%d, %d) ได้ %d คะแนน" % [star.name, star.position.x, star.position.y, star.points])


func total_points() -> int:
    var total := 0
    for child in get_children():
        var star := child as GridStar
        if star != null:
            total += star.points
    return total


func remove_row(row: int) -> void:
    for col in columns:
        var star := get_node_or_null("Star_%d_%d" % [row, col])
        if star != null:
            star.queue_free()

  • ค่าที่ Designer ต้องปรับเอาไปไว้ใน @export ตัวเลขทุกตัวที่เกี่ยวกับรูปร่างของตารางอยู่ใน Inspector ไม่มี Magic Number ซ่อนอยู่ในลูป

  • เข้าใจจังหวะของ queue_free() จำนวนลูกยังเป็น 12 จนกว่า Frame จะจบ ถ้าเกมจริงต้องรู้จำนวนดาวทันทีหลังเก็บ ให้รอด้วย await get_tree().process_frame หรือนับเองในตัวแปร

class_name SaveV2
extends RefCounted
## ระบบบันทึกเวอร์ชัน 2 ของ Star Dodger
## savegame.cfg เก็บ [meta] version, [progress] และ [settings]
## scores.json เก็บคะแนนสูงสุด 5 อันดับ
## ไฟล์เวอร์ชัน 1 (ไม่มี [meta]) จะถูกย้ายมาเป็นเวอร์ชัน 2 โดย High Score ไม่หาย

const VERSION: int = 2
const MAX_TOP: int = 5
const DEFAULT_VOLUME: float = 0.8

var cfg_path: String
var scores_path: String

var high_score: int = 0
var games_played: int = 0
var screen_shake: bool = true
var music_volume: float = DEFAULT_VOLUME
var sfx_volume: float = DEFAULT_VOLUME


# รับ Path จากภายนอกได้ เพื่อให้ Test ใช้ไฟล์ของตัวเองโดยไม่ทับเซฟจริง
func _init(p_cfg_path: String = "user://savegame.cfg", p_scores_path: String = "user://scores.json") -> void:
    cfg_path = p_cfg_path
    scores_path = p_scores_path


# โหลดการตั้งค่า แล้วคืนสถานะ "new", "ok", "migrated" หรือ "corrupt"
func load_settings() -> String:
    _reset_to_defaults()
    if not FileAccess.file_exists(cfg_path):
        return "new"
    var cfg := ConfigFile.new()
    var err := cfg.load(cfg_path)
    if err != OK:
        push_warning("อ่าน %s ไม่ได้ (%s) จึงใช้ค่าเริ่มต้น" % [cfg_path, error_string(err)])
        return "corrupt"
    # ไฟล์เวอร์ชัน 1 ไม่มี [meta] จึงถือว่าเป็น 1
    var version := _read_int(cfg, "meta", "version", 1)
    high_score = maxi(_read_int(cfg, "progress", "high_score", 0), 0)
    games_played = maxi(_read_int(cfg, "progress", "games_played", 0), 0)
    var shake: Variant = cfg.get_value("settings", "screen_shake", true)
    screen_shake = shake if shake is bool else true
    music_volume = _read_volume(cfg, "music_volume")
    sfx_volume = _read_volume(cfg, "sfx_volume")
    if version < VERSION:
        save_settings()
        return "migrated"
    return "ok"


func save_settings() -> Error:
    var cfg := ConfigFile.new()
    cfg.set_value("meta", "version", VERSION)
    cfg.set_value("progress", "high_score", high_score)
    cfg.set_value("progress", "games_played", games_played)
    cfg.set_value("settings", "screen_shake", screen_shake)
    cfg.set_value("settings", "music_volume", music_volume)
    cfg.set_value("settings", "sfx_volume", sfx_volume)
    return cfg.save(cfg_path)


# อ่าน Top 5 จาก JSON ข้ามค่าที่ไม่ใช่ตัวเลข แล้วเรียงจากมากไปน้อย
func load_top_scores() -> Array[int]:
    var result: Array[int] = []
    if not FileAccess.file_exists(scores_path):
        return result
    var json := JSON.new()
    if json.parse(FileAccess.get_file_as_string(scores_path)) != OK:
        push_warning("scores.json เสียที่บรรทัด %d: %s" % [json.get_error_line(), json.get_error_message()])
        return result
    if not (json.data is Array):
        return result
    for item: Variant in json.data:
        # JSON คืนตัวเลขกลับมาเป็น float เสมอ จึงต้องแปลงเป็น int เอง
        if (item is float or item is int) and item >= 0:
            result.append(int(item))
    result.sort()
    result.reverse()
    result.resize(mini(result.size(), MAX_TOP))
    return result


func save_top_scores(scores: Array[int]) -> Error:
    var top: Array[int] = scores.duplicate()
    top.sort()
    top.reverse()
    top.resize(mini(top.size(), MAX_TOP))
    var file := FileAccess.open(scores_path, FileAccess.WRITE)
    if file == null:
        return FileAccess.get_open_error()
    file.store_string(JSON.stringify(top))
    return OK


# บันทึกผลหนึ่งเกม คืน true เมื่อทำ High Score ใหม่
func record_game(score: int) -> bool:
    games_played += 1
    var is_new_high := score > high_score
    if is_new_high:
        high_score = score
    var top := load_top_scores()
    top.append(score)
    save_top_scores(top)
    save_settings()
    return is_new_high


func _reset_to_defaults() -> void:
    high_score = 0
    games_played = 0
    screen_shake = true
    music_volume = DEFAULT_VOLUME
    sfx_volume = DEFAULT_VOLUME


func _read_int(cfg: ConfigFile, section: String, key: String, fallback: int) -> int:
    var value: Variant = cfg.get_value(section, key, fallback)
    if value is int:
        return value
    if value is float:
        return int(value)
    return fallback


func _read_volume(cfg: ConfigFile, key: String) -> float:
    var value: Variant = cfg.get_value("settings", key, DEFAULT_VOLUME)
    if value is int or value is float:
        return clampf(float(value), 0.0, 1.0)
    return DEFAULT_VOLUME

แนวคิด
  • ใส่เลขเวอร์ชันในไฟล์เซฟเสมอ [meta] version ทำให้โค้ดรู้ว่ากำลังอ่านไฟล์รูปแบบไหน ไฟล์ที่ไม่มีเลขเวอร์ชันถือเป็นเวอร์ชัน 1 แล้วย้ายข้อมูล (Migrate) และเขียนกลับเป็นเวอร์ชัน 2 ทันที การโหลดครั้งถัดไปจึงได้สถานะ "ok" และไม่ย้ายซ้ำ
  • เริ่มจากค่าเริ่มต้นแล้วค่อยเชื่อไฟล์ทีละค่า _reset_to_defaults() ทำให้ทุกเส้นทางที่ล้มเหลวยังได้ค่าที่ใช้งานได้ ทุกค่าที่อ่านจากดิสก์ถูกตรวจชนิดและช่วง เพราะผู้เล่นแก้ไฟล์ได้ และไฟล์อาจเสียเมื่อไฟดับระหว่างเขียน
  • คืนสถานะเป็นข้อความ "new", "ok", "migrated" และ "corrupt" ทำให้ Test ตรวจได้ตรงๆ และให้เกมนำไปใช้ต่อได้ เช่น แสดงข้อความ "ไฟล์เซฟเสีย เริ่มใหม่จากศูนย์" เมื่อได้ "corrupt"
  • ฉีด Path เข้ามาทาง _init() เกมจริงใช้ค่าเริ่มต้นใน user:// ส่วน Test ส่ง Path ของโฟลเดอร์ทดสอบเข้ามา เซฟจริงของผู้พัฒนาจึงไม่ถูกทับ เทคนิคนี้เรียกว่า Dependency Injection
  • รู้จักกับดักของ JSON ตัวเลขทุกตัวกลับมาเป็น float จึงต้องแปลงด้วย int() และใช้ JSON.new().parse() เพื่อให้ได้บรรทัดและข้อความของ Error (ในตัวอย่างคือ Expected ']') ไปเขียนเตือน
  • Test ที่บอกผลด้วย Exit Code quit(0) เมื่อผ่านหมดและ quit(1) เมื่อมีข้อที่ไม่ผ่าน ทำให้ PowerShell, Script อัตโนมัติ และระบบ CI รู้ผลได้โดยไม่ต้องอ่านข้อความ
