    data object Loading : NotesUiState
    data object Empty : NotesUiState
    data class Success(val count: Int) : NotesUiState
    data class Error(val message: String) : NotesUiState
    data class Offline(val cached: Int) : NotesUiState
}

fun render(state: NotesUiState): String = when (state) {

    is NotesUiState.Success -> "มี ${state.count} โน้ต"
    is NotesUiState.Error -> "ผิดพลาด: ${state.message}"
}

// โน้ตหนึ่งรายการจากแอปเวอร์ชันเก่า
data class LegacyNote(
    val id: Int,
    val title: String,
    val pinned: Boolean = false,
    val tag: String = "ทั่วไป",
)

// ผลของการแปลงหนึ่งบรรทัด: สำเร็จ หรือ ข้ามพร้อมเหตุผล
sealed interface ImportResult {
    data class Ok(val note: LegacyNote) : ImportResult
    data class Skipped(val lineNo: Int, val reason: String) : ImportResult
}

// แปลงบรรทัดรูปแบบ id;title;pinned;tag
fun parseLine(lineNo: Int, line: String): ImportResult {
    val parts = line.split(";").map { it.trim() }
    if (parts.size < 4) return ImportResult.Skipped(lineNo, "ข้อมูลไม่ครบ 4 ช่อง")
    // toIntOrNull() คืน null เมื่อไม่ใช่ตัวเลข แล้ว ?: ออกจากฟังก์ชันทันที
    val id = parts[0].toIntOrNull()
        ?: return ImportResult.Skipped(lineNo, "id ไม่ใช่ตัวเลข (${parts[0]})")
    val title = parts[1]
    if (title.isBlank()) return ImportResult.Skipped(lineNo, "ไม่มีชื่อโน้ต")
    val note = LegacyNote(
        id = id,
        title = title,
        pinned = parts[2] == "1",
        tag = parts[3].ifBlank { "ทั่วไป" },
    )
    return ImportResult.Ok(note)
}

// แปลงทั้งไฟล์ ข้ามบรรทัดว่างและบรรทัด Comment ที่ขึ้นต้นด้วย #
fun importNotes(raw: String): List<ImportResult> =
    raw.lines()
        .mapIndexed { index, line -> index + 1 to line }
        .filter { (_, line) -> line.isNotBlank() && !line.trimStart().startsWith("#") }
        .map { (lineNo, line) -> parseLine(lineNo, line) }

// Extension function: ย่อชื่อที่ยาวเกินให้พอดีกับการ์ด
fun LegacyNote.preview(max: Int = 20): String =
    if (title.length <= max) title else title.take(max - 3) + "..."

fun main() {
    val raw = """
        # ไฟล์ export จากแอปเวอร์ชัน 1
        1;ซื้อของเข้าบ้าน;1;บ้าน
        2;ประชุมทีม 10:00;0;งาน
        abc;โน้ตเสีย;0;งาน
        3;ทบทวน Kotlin Flow;0;
        4;  ;1;งาน
        5;อ่านเอกสาร Kotlin Coroutines;1;งาน
        6;โน้ตไม่ครบ
    """.trimIndent()

    val results = importNotes(raw)
    val notes = results.filterIsInstance<ImportResult.Ok>().map { it.note }
    val skippedCount = results.count { it is ImportResult.Skipped }
    println("นำเข้าได้ ${notes.size} โน้ต ข้าม $skippedCount บรรทัด")

    // when ครบทุกกรณีของ sealed interface จึงไม่ต้องมี else
    for (result in results) {
        when (result) {
            is ImportResult.Ok -> Unit
            is ImportResult.Skipped -> println("- บรรทัด ${result.lineNo}: ${result.reason}")
        }
    }

    println("รายการโน้ต (ปักหมุดขึ้นก่อน):")
    // sortedByDescending เป็น Stable sort โน้ตที่ค่าเท่ากันจึงคงลำดับเดิม
    notes.sortedByDescending { it.pinned }.forEach { note ->
        val pin = if (note.pinned) "[ปักหมุด] " else ""
        println("$pin#${note.id} ${note.preview()} (${note.tag})")
    }

    val byTag = notes.groupBy { it.tag }.map { (tag, list) -> "$tag=${list.size}" }
    println("จำนวนตาม Tag: ${byTag.joinToString(", ")}")
}

// โน้ตที่มาได้ทั้งจากเครื่องและจากเซิร์ฟเวอร์
data class SyncedNote(
    val id: Int,
    val title: String,
    val pinned: Boolean = false,
    val updatedAt: Long, // เวลาที่แก้ไขล่าสุด หน่วยมิลลิวินาที
)

// แถวของ LazyColumn มี 2 แบบ ทุกแถวต้องมี key ที่ไม่ซ้ำและไม่เปลี่ยนไปมา
sealed interface NoteRow {
    val key: String

    data class Header(val section: String, val title: String) : NoteRow {
        override val key: String get() = "header-$section"
    }

    data class Item(val note: SyncedNote) : NoteRow {
        override val key: String get() = "note-${note.id}"
    }
}

// รวมสองแหล่ง id ซ้ำให้เก็บตัวที่แก้ไขล่าสุด ถ้าเวลาเท่ากันให้ใช้ของเซิร์ฟเวอร์
fun mergeNotes(local: List<SyncedNote>, remote: List<SyncedNote>): List<SyncedNote> =
    (remote + local)
        .groupBy { it.id }
        .map { (_, versions) -> versions.maxBy { it.updatedAt } }

// สร้างแถวสำหรับหน้าจอ: ส่วนปักหมุดอยู่บน ส่วนโน้ตอื่นอยู่ล่าง
fun buildRows(notes: List<SyncedNote>, pinnedOnly: Boolean): List<NoteRow> {
    val (pinned, others) = notes.sortedByDescending { it.updatedAt }.partition { it.pinned }
    val rows = mutableListOf<NoteRow>()
    if (pinned.isNotEmpty()) {
        rows += NoteRow.Header("pinned", "ปักหมุด (${pinned.size})")
        rows += pinned.map { NoteRow.Item(it) }
    }
    if (!pinnedOnly && others.isNotEmpty()) {
        rows += NoteRow.Header("others", "โน้ตอื่น (${others.size})")
        rows += others.map { NoteRow.Item(it) }
    }
    return rows
}

// คืน key ที่ใช้ซ้ำ ถ้าได้ลิสต์ว่างแปลว่าส่งให้ LazyColumn ได้อย่างปลอดภัย
fun duplicateKeys(rows: List<NoteRow>): List<String> =
    rows.groupingBy { it.key }.eachCount().filter { it.value > 1 }.keys.toList()

fun printRows(rows: List<NoteRow>) {
    for (row in rows) {
        when (row) {
            is NoteRow.Header -> println("== ${row.title}  [key=${row.key}]")
            is NoteRow.Item -> println("   ${row.note.title}  [key=${row.key}]")
        }
    }
}

sealed interface SyncUiState {

// ตรรกะของ ViewModel ที่ไม่มี import ของ Android จึง Test บน JVM ได้
class SyncPresenter(
    private val source: NotesRemoteSource,
    private val scope: CoroutineScope,
) {
    private val json = Json { ignoreUnknownKeys = true }
    private val _state = MutableStateFlow<SyncUiState>(SyncUiState.Idle)
    val state: StateFlow<SyncUiState> = _state.asStateFlow()
    private var syncJob: Job? = null

    // คืน true เมื่อเริ่มซิงก์ใหม่ และ false เมื่อกำลังซิงก์อยู่แล้ว
    fun sync(): Boolean {
        if (syncJob?.isActive == true) return false
        _state.value = SyncUiState.Syncing
        syncJob = scope.launch {
            _state.value = try {
                val body = source.fetchNotesJson()
                val notes = json.decodeFromString<List<SyncNoteDto>>(body)
                    .map { CloudNote(it.id, it.title, it.isPinned) }
                    .sortedWith(compareByDescending<CloudNote> { it.pinned }.thenBy { it.id })
                SyncUiState.Success(notes)
            } catch (e: IOException) {
                SyncUiState.Error("เชื่อมต่อเซิร์ฟเวอร์ไม่ได้ ลองใหม่อีกครั้ง")
            } catch (e: ServerException) {
                SyncUiState.Error("เซิร์ฟเวอร์ตอบกลับผิดพลาด (HTTP ${e.code})")
            } catch (e: SerializationException) {
                SyncUiState.Error("ข้อมูลจากเซิร์ฟเวอร์ผิดรูปแบบ")
            }
        }
        return true
    }
}

// แหล่งข้อมูลจริงบน JVM ใช้ HttpURLConnection (ในแอป Android ใช้ Retrofit แทน)
class HttpNotesSource(private val url: String) : NotesRemoteSource {
    override suspend fun fetchNotesJson(): String = withContext(Dispatchers.IO) {
        val conn = URI(url).toURL().openConnection() as HttpURLConnection
        try {
            conn.connectTimeout = 3_000
            conn.readTimeout = 3_000
            val code = conn.responseCode
            if (code != HttpURLConnection.HTTP_OK) throw ServerException(code)
            conn.inputStream.bufferedReader().use { it.readText() }
        } finally {
            conn.disconnect()
        }
    }
}

fun main(args: Array<String>) = runBlocking {
    val url = args.firstOrNull() ?: "http://127.0.0.1:8000/sync.json"
    val presenter = SyncPresenter(HttpNotesSource(url), this)
    println("กดซิงก์ครั้งที่ 1: เริ่มใหม่ = ${presenter.sync()}")
    println("กดซิงก์ครั้งที่ 2: เริ่มใหม่ = ${presenter.sync()}")
    println("สถานะตอนนี้: ${presenter.state.value}")
    // รอจนได้ผลลัพธ์สุดท้าย ไม่ว่าจะสำเร็จหรือผิดพลาด
    when (val result = presenter.state.first { it is SyncUiState.Success || it is SyncUiState.Error }) {
        is SyncUiState.Success -> result.notes.forEach { note ->
            val pin = if (note.pinned) "[ปักหมุด] " else ""
            println("$pin#${note.id} ${note.title}")
        }
        is SyncUiState.Error -> println("ผิดพลาด: ${result.message}")
        else -> Unit
    }
}

// Endpoint ของ API จำลอง: คืน Body ดิบ เพื่อให้ SyncPresenter เป็นผู้แปลง JSON เอง
interface NotesSyncApi {
    @GET("sync.json")
    suspend fun getSyncJson(): ResponseBody
}

// แหล่งข้อมูลจริงของแอป: แปลง Error ของ Retrofit เป็น ServerException ของเราเอง
class RetrofitNotesSource(private val api: NotesSyncApi) : NotesRemoteSource {
    override suspend fun fetchNotesJson(): String =
        try {
            // อ่าน Body บน Dispatchers.IO เพราะ string() อ่านข้อมูลจาก Network
            withContext(Dispatchers.IO) {
                api.getSyncJson().use { body -> body.string() }
            }
        } catch (e: HttpException) {
            throw ServerException(e.code())
        }
}

    private const val BASE_URL = "http://10.0.2.2:8000/"

    val api: NotesSyncApi by lazy {
        Retrofit.Builder()
            .baseUrl(BASE_URL)
            .build()
            .create(NotesSyncApi::class.java)
    }
}
